1//! Money arithmetic shared by every module: one rounding rule and percentages
2//!
3//! Amounts are [`BigDecimal`] in a currency's major unit (rupees, euros, dollars) with
4//! [`MONEY_SCALE`] decimal places once rounded. [`round_money`] is the only rounding rule in the
5//! crate; the Indian tax code re-exports it as `gst::tax::round_to_paise`.
6
7use bigdecimal::{BigDecimal, RoundingMode};
8
9/// Decimal places in a rounded amount: one minor unit (a paisa, a cent) is 0.01
10pub const MONEY_SCALE: i64 = 2;
11
12/// Percent in a whole
13const HUNDRED: i32 = 100;
14
15/// `rate` percent of `amount`, exactly; round the result with [`round_money`]
16///
17/// # Example
18///
19/// ```
20/// use accounting_core::money::percent_of;
21/// use bigdecimal::BigDecimal;
22///
23/// assert_eq!(percent_of(&BigDecimal::from(1500), &BigDecimal::from(18)), BigDecimal::from(270));
24/// ```
25#[must_use]
26pub fn percent_of(amount: &BigDecimal, rate: &BigDecimal) -> BigDecimal {
27    (amount * rate) / BigDecimal::from(HUNDRED)
28}
29
30/// Round an amount to the nearest minor unit, with halves rounded away from zero
31///
32/// 0.025 becomes 0.03 and 0.0249 becomes 0.02. The result always has [`MONEY_SCALE`] decimal
33/// places.
34///
35/// # Example
36///
37/// ```
38/// use accounting_core::money::round_money;
39/// use bigdecimal::BigDecimal;
40/// use std::str::FromStr;
41///
42/// let round = |s| round_money(&BigDecimal::from_str(s).unwrap()).to_string();
43/// assert_eq!(round("0.02475"), "0.02");
44/// assert_eq!(round("0.025"), "0.03");
45/// assert_eq!(round("-0.025"), "-0.03");
46/// assert_eq!(round("90"), "90.00");
47/// ```
48#[must_use]
49pub fn round_money(amount: &BigDecimal) -> BigDecimal {
50    amount.with_scale_round(MONEY_SCALE, RoundingMode::HalfUp)
51}
