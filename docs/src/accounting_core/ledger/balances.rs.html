1//! Pure balance calculations over accounts and transactions
2//!
3//! Nothing here touches storage: callers load the accounts and transactions, and these functions
4//! derive balances from them.
5
6use bigdecimal::BigDecimal;
7use chrono::NaiveDate;
8use std::collections::HashMap;
9
10use crate::types::{Account, AccountBalance, AccountType, EntryType, Transaction, TrialBalance};
11
12/// Balance of `account` from the entries that post to it, measured on its normal side
13pub fn account_balance<'a>(
14    account: &Account,
15    transactions: impl IntoIterator<Item = &'a Transaction>,
16) -> BigDecimal {
17    transactions
18        .into_iter()
19        .flat_map(|transaction| &transaction.entries)
20        .filter(|entry| entry.account_id == account.id)
21        .map(|entry| {
22            account
23                .account_type
24                .balance_effect(entry.entry_type, &entry.amount)
25        })
26        .sum()
27}
28
29/// Debit-positive net movement per account id
30fn net_by_account<'a>(
31    transactions: impl IntoIterator<Item = &'a Transaction>,
32) -> HashMap<&'a str, BigDecimal> {
33    transactions
34        .into_iter()
35        .flat_map(|transaction| &transaction.entries)
36        .fold(HashMap::new(), |mut net, entry| {
37            *net.entry(entry.account_id.as_str()).or_default() +=
38                entry.entry_type.signed(&entry.amount);
39            net
40        })
41}

43/// Trial balance as of a date, from every account and the transactions dated up to then
44///
45/// Transactions dated after `as_of_date` are ignored, so the full history may be passed in.
46pub fn trial_balance(
47    as_of_date: NaiveDate,
48    accounts: impl IntoIterator<Item = Account>,
49    transactions: &[Transaction],
50) -> TrialBalance {
51    let net = net_by_account(transactions.iter().filter(|t| t.date <= as_of_date));
52    let zero = BigDecimal::from(0);
53
54    let balances: HashMap<String, AccountBalance> = accounts
55        .into_iter()
56        .map(|account| {
57            let movement = net.get(account.id.as_str()).unwrap_or(&zero);
58            let balance = account
59                .account_type
60                .balance_effect(EntryType::Debit, movement);
61            (
62                account.id.clone(),
63                AccountBalance::from_balance(account, &balance),
64            )
65        })
66        .collect();
67
68    let total_debits: BigDecimal = balances
69        .values()
70        .filter_map(|b| b.debit_balance.as_ref())
71        .sum();
72    let total_credits: BigDecimal = balances
73        .values()
74        .filter_map(|b| b.credit_balance.as_ref())
75        .sum();
76
77    TrialBalance {
78        as_of_date,
79        balances,
80        is_balanced: total_debits == total_credits,
81        total_debits,
82        total_credits,
83    }
84}

86/// Group account balances by account type
87pub fn group_by_type(
88    balances: impl IntoIterator<Item = AccountBalance>,
89) -> HashMap<AccountType, Vec<AccountBalance>> {
90    balances
91        .into_iter()
92        .fold(HashMap::new(), |mut grouped, balance| {
93            grouped
94                .entry(balance.account.account_type)
95                .or_insert_with(Vec::new)
96                .push(balance);
97            grouped
98        })
