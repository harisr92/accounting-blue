1//! Text normalisation and similarity scoring for transaction descriptions
2//!
3//! Bank narrations wrap the useful part in channel codes, slashes and reference fragments
4//! (`"NEFT/ACME LTD/HDFC0000123"` for what the ledger calls `"Acme Ltd"`). Similarity therefore
5//! combines three views and takes the most generous: token overlap, edit distance, and containment.
6
7use std::collections::HashSet;
8
9/// Longest input considered by the edit-distance comparison
10const MAX_COMPARE_LEN: usize = 256;
11
12/// Shortest string that may win credit purely by being contained in the other
13const MIN_CONTAINMENT_LEN: usize = 4;
14
15/// Credit awarded when one description contains the other
16pub(super) const CONTAINMENT_SCORE: f64 = 0.8;
17
18/// Lowercase `text`, replace every run of non-alphanumeric characters with a single space, and
19/// trim.
20///
21/// ```
22/// use accounting_core::reconciliation::similarity::normalize;
23/// assert_eq!(normalize("NEFT/ACME LTD/HDFC0000123"), "neft acme ltd hdfc0000123");
24/// ```
25#[must_use]
26pub fn normalize(text: &str) -> String {
27    let mut out = String::with_capacity(text.len());
28    let mut pending_space = false;
29
30    for ch in text.chars() {
31        if ch.is_alphanumeric() {
32            if pending_space && !out.is_empty() {
33                out.push(' ');
34            }
35            pending_space = false;
36            out.extend(ch.to_lowercase());
37        } else {
38            pending_space = true;
39        }
40    }
41
42    out
43}

45/// Score how alike two descriptions are, in `0.0..=1.0`.
46///
47/// Two empty descriptions count as identical; one empty description scores zero.
48///
49/// ```
50/// use accounting_core::reconciliation::similarity::similarity;
51/// assert_eq!(similarity("Acme Ltd", "ACME  LTD."), 1.0);
52/// assert!(similarity("NEFT/ACME LTD/0012", "Acme Ltd") >= 0.8);
53/// assert!(similarity("Acme Ltd", "Globex Inc") < 0.4);
54/// ```
55#[must_use]
56pub fn similarity(left: &str, right: &str) -> f64 {
57    let left = normalize(left);
58    let right = normalize(right);
59
60    match (left.is_empty(), right.is_empty()) {
61        (true, true) => return 1.0,
62        (true, false) | (false, true) => return 0.0,
63        (false, false) => {}
64    }
65
66    if left == right {
67        return 1.0;
68    }
69
70    let mut score = token_similarity(&left, &right).max(edit_similarity(&left, &right));
71
72    let shorter_len = left.chars().count().min(right.chars().count());
73    if shorter_len >= MIN_CONTAINMENT_LEN && (left.contains(&right) || right.contains(&left)) {
74        score = score.max(CONTAINMENT_SCORE);
75    }
76
77    score
78}

80/// Jaccard index over whitespace-separated tokens
81#[allow(clippy::cast_precision_loss)] // counts are bounded by input length
82fn token_similarity(left: &str, right: &str) -> f64 {
83    let left_tokens: HashSet<&str> = left.split(' ').collect();
84    let right_tokens: HashSet<&str> = right.split(' ').collect();
85
86    let union = left_tokens.union(&right_tokens).count();
87    if union == 0 {
88        return 0.0;
89    }
90
91    left_tokens.intersection(&right_tokens).count() as f64 / union as f64
92}

94/// Edit distance rescaled against the longer input
95#[allow(clippy::cast_precision_loss)] // counts are bounded by input length
96fn edit_similarity(left: &str, right: &str) -> f64 {
97    let left_chars: Vec<char> = left.chars().take(MAX_COMPARE_LEN).collect();
98    let right_chars: Vec<char> = right.chars().take(MAX_COMPARE_LEN).collect();
99
100    let longest = left_chars.len().max(right_chars.len());
101    if longest == 0 {
102        return 1.0;
103    }
104
105    1.0 - (levenshtein_chars(&left_chars, &right_chars) as f64 / longest as f64)
106}

108/// Levenshtein edit distance between two strings, in characters
109#[must_use]
110pub fn levenshtein(left: &str, right: &str) -> usize {
111    let left_chars: Vec<char> = left.chars().collect();
112    let right_chars: Vec<char> = right.chars().collect();
113    levenshtein_chars(&left_chars, &right_chars)
114}
115
116/// Two-row dynamic programming over characters, so multi-byte text is never split
117fn levenshtein_chars(left: &[char], right: &[char]) -> usize {
118    if left.is_empty() {
119        return right.len();
120    }
121    if right.is_empty() {
122        return left.len();
123    }
124
125    let mut previous: Vec<usize> = (0..=right.len()).collect();
126    let mut current = vec![0usize; right.len() + 1];
127
128    for (i, left_ch) in left.iter().enumerate() {
129        current[0] = i + 1;
130        for (j, right_ch) in right.iter().enumerate() {
131            let substitution = previous[j] + usize::from(left_ch != right_ch);
132            current[j + 1] = substitution.min(previous[j + 1] + 1).min(current[j] + 1);
133        }
134        std::mem::swap(&mut previous, &mut current);
135    }
136
137    previous[right.len()]
138}
