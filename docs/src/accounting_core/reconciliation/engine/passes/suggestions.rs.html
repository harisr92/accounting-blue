1//! Pass 5: report what is left, each with its best still-available counterparts
2
3use super::super::state::{Candidate, MatchState, Sides};
4use crate::reconciliation::config::ReconciliationConfig;
5use crate::reconciliation::types::{PartialMatch, ReconciliationStatus};
6
7/// Suggestion lists, one per row on each side
8struct Suggestions {
9    ledger: Vec<Vec<PartialMatch>>,
10    external: Vec<Vec<PartialMatch>>,
11}
12
13/// Append a suggestion unless the list is already full
14fn push_capped(
15    list: &mut Vec<PartialMatch>,
16    max: usize,
17    counterpart_id: &str,
18    candidate: &Candidate,
19) {
20    if list.len() < max {
21        list.push(PartialMatch {
22            counterpart_id: counterpart_id.to_string(),
23            match_score: candidate.score,
24            differences: candidate.differences.clone(),
25        });
26    }
27}
28
29/// Collect up to `max_suggestions` counterparts per row from the free candidate pairs
30fn suggest(
31    config: &ReconciliationConfig,
32    sides: &Sides,
33    candidates: &[Candidate],
34    state: &MatchState,
35) -> Suggestions {
36    let max = config.max_suggestions;
37    let empty = Suggestions {
38        ledger: vec![Vec::new(); sides.ledger.len()],
39        external: vec![Vec::new(); sides.external.len()],
40    };
41
42    // `candidates` is already sorted by score descending, so each list comes out sorted too.
43    // Never suggest a counterpart that is already spoken for.
44    candidates
45        .iter()
46        .filter(|c| state.is_free(c.ledger_index, c.external_index))
47        .fold(empty, |mut suggestions, c| {
48            let external_id = &sides.external[c.external_index].id;
49            let ledger_id = &sides.ledger[c.ledger_index].id;
50            push_capped(&mut suggestions.ledger[c.ledger_index], max, external_id, c);
51            push_capped(
52                &mut suggestions.external[c.external_index],
53                max,
54                ledger_id,
55                c,
56            );
57            suggestions
58        })
59}
60
61/// Report every unclaimed row as unmatched, carrying its suggestions
62pub(in crate::reconciliation::engine) fn collect_unmatched(
63    config: &ReconciliationConfig,
64    sides: &Sides,
65    candidates: &[Candidate],
66    state: &mut MatchState,
67) {
68    let Suggestions {
69        mut ledger,
70        mut external,
71    } = suggest(config, sides, candidates, state);
72
73    let unmatched_ledger: Vec<ReconciliationStatus> = sides
74        .free_ledger(state)
75        .map(|index| ReconciliationStatus::UnmatchedLedger {
76            ledger_id: sides.ledger[index].id.clone(),
77            possible_matches: std::mem::take(&mut ledger[index]),
78        })
79        .collect();
80    let unmatched_external: Vec<ReconciliationStatus> = sides
81        .free_external(state)
82        .map(|index| ReconciliationStatus::UnmatchedExternal {
83            external_id: sides.external[index].id.clone(),
84            possible_matches: std::mem::take(&mut external[index]),
85        })
86        .collect();
87
88    state.items.extend(unmatched_ledger);
89    state.items.extend(unmatched_external);
90}
