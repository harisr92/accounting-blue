1//! Pass 1: a shared reference number, backed by an equal amount and direction, is conclusive
2
3use std::collections::HashMap;
4
5use super::super::state::{MatchState, Sides};
6use crate::reconciliation::scoring::normalized_reference;
7
8/// Match on reference numbers.
9///
10/// A reference is only acted on when it identifies exactly one available counterpart; a
11/// reference reused across several statement rows is left for the later passes. Dates are
12/// deliberately ignored here, so settlement lag on a row carrying a UTR still matches outright.
13pub(in crate::reconciliation::engine) fn match_by_reference(sides: &Sides, state: &mut MatchState) {
14    let by_reference = index_external_references(sides);
15
16    for &ledger_index in &sides.ledger_order {
17        if let Some(external_index) = unique_counterpart(sides, state, &by_reference, ledger_index)
18        {
19            state.record_exact_match(sides, ledger_index, external_index);
20        }
21    }
22}
23
24/// External indices grouped by normalized reference
25fn index_external_references(sides: &Sides) -> HashMap<String, Vec<usize>> {
26    sides
27        .external
28        .iter()
29        .enumerate()
30        .filter_map(|(index, t)| {
31            normalized_reference(t.reference.as_deref()).map(|key| (key, index))
32        })
33        .fold(HashMap::new(), |mut by_reference, (key, index)| {
34            by_reference.entry(key).or_insert_with(Vec::new).push(index);
35            by_reference
36        })
37}
38
39/// The single free external row sharing this ledger row's reference, amount and direction
40fn unique_counterpart(
41    sides: &Sides,
42    state: &MatchState,
43    by_reference: &HashMap<String, Vec<usize>>,
44    ledger_index: usize,
45) -> Option<usize> {
46    let ledger = &sides.ledger[ledger_index];
47    let key = normalized_reference(ledger.reference.as_deref())?;
48
49    let mut available = by_reference.get(&key)?.iter().copied().filter(|&index| {
50        let external = &sides.external[index];
51        state.is_external_free(index)
52            && external.amount == ledger.amount
53            && external.entry_type == ledger.entry_type
54    });
55
56    let first = available.next()?;
57    // Ambiguous - let the later passes decide
58    available.next().is_none().then_some(first)
59}
