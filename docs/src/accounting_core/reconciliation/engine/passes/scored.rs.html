1//! Passes 3 and 4: score what is left, then assign the best pairs first
2
3use std::cmp::Ordering;
4
5use super::super::state::{Candidate, MatchState, Sides};
6use crate::reconciliation::config::ReconciliationConfig;
7use crate::reconciliation::scoring::{requires_review, score_pair};
8use crate::reconciliation::types::ReconciliationStatus;
9
10/// Score every still-plausible pair once, feeding both the assignment and the suggestion lists.
11///
12/// Pairs dated further apart than `candidate_window_days`, or scoring below
13/// `suggestion_threshold`, are dropped. The result is sorted best-first, ties broken by ids.
14pub(in crate::reconciliation::engine) fn score_remaining(
15    config: &ReconciliationConfig,
16    sides: &Sides,
17    state: &MatchState,
18) -> Vec<Candidate> {
19    let mut candidates: Vec<Candidate> = sides
20        .free_ledger(state)
21        .flat_map(|ledger_index| {
22            sides
23                .free_external(state)
24                .map(move |external_index| (ledger_index, external_index))
25        })
26        .filter(|&(l, e)| {
27            let days_apart = (sides.ledger[l].date - sides.external[e].date).num_days();
28            days_apart.abs() <= config.candidate_window_days
29        })
30        .filter_map(|(ledger_index, external_index)| {
31            let (score, differences) = score_pair(
32                config,
33                &sides.ledger[ledger_index],
34                &sides.external[external_index],
35            );
36            (score >= config.suggestion_threshold).then_some(Candidate {
37                ledger_index,
38                external_index,
39                score,
40                differences,
41            })
42        })
43        .collect();
44
45    candidates.sort_by(|a, b| best_first(sides, a, b));
46    candidates
47}
48
49/// Highest score first, then by ledger id, then by external id
50fn best_first(sides: &Sides, a: &Candidate, b: &Candidate) -> Ordering {
51    b.score
52        .total_cmp(&a.score)
53        .then_with(|| {
54            sides.ledger[a.ledger_index]
55                .id
56                .cmp(&sides.ledger[b.ledger_index].id)
57        })
58        .then_with(|| {
59            sides.external[a.external_index]
60                .id
61                .cmp(&sides.external[b.external_index].id)
62        })
63}

65/// Walk the scored pairs best-first, taking each whose both sides are still free.
66///
67/// Assigning globally rather than per ledger row stops a mediocre pairing from claiming a
68/// counterpart that a stronger pairing later in the input wanted. A pair that disagrees on money
69/// or direction is always reported as a partial match, however high it scores - see
70/// [`requires_review`].
71pub(in crate::reconciliation::engine) fn assign_best_first(
72    config: &ReconciliationConfig,
73    sides: &Sides,
74    candidates: &[Candidate],
75    state: &mut MatchState,
76) {
77    let qualifying = candidates
78        .iter()
79        .take_while(|candidate| candidate.score >= config.partial_match_threshold);
80
81    for candidate in qualifying {
82        if state.is_free(candidate.ledger_index, candidate.external_index) {
83            state.claim(candidate.ledger_index, candidate.external_index);
84            state.items.push(classify(config, sides, candidate));
85        }
86    }
87}

89/// A full match when the score clears `auto_match_threshold` and nothing needs review
90fn classify(
91    config: &ReconciliationConfig,
92    sides: &Sides,
93    candidate: &Candidate,
94) -> ReconciliationStatus {
95    let ledger_id = sides.ledger[candidate.ledger_index].id.clone();
96    let external_id = sides.external[candidate.external_index].id.clone();
97    let needs_review = requires_review(&candidate.differences);
98
99    if candidate.score >= config.auto_match_threshold && !needs_review {
100        ReconciliationStatus::Matched {
101            ledger_id,
102            external_id,
103            match_score: candidate.score,
104        }
105    } else {
106        ReconciliationStatus::PartialMatch {
107            ledger_id,
108            external_id,
109            match_score: candidate.score,
110            differences: candidate.differences.clone(),
111            auto_resolvable: candidate.score >= config.auto_resolve_threshold && !needs_review,
112        }
113    }
114}
