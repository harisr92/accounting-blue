1//! Tuning knobs for the reconciliation engine
2
3/// Thresholds and tolerances that control matching
4///
5/// The defaults suit Indian bank statements, where settlement lag of a day or two is normal and
6/// narrations are noisy. Note that with the defaults a date-shifted pair that carries no reference
7/// number lands as a [`PartialMatch`](crate::reconciliation::ReconciliationStatus::PartialMatch)
8/// rather than an outright match — widen `date_tolerance_days` or lower `auto_match_threshold` if
9/// you would rather accept those automatically. Pairs that agree on reference number, amount and
10/// direction always match outright, whatever their dates.
11#[derive(Debug, Clone, PartialEq)]
12pub struct ReconciliationConfig {
13    /// How many days apart two records may be dated and still score on the date dimension
14    pub date_tolerance_days: i64,
15    /// Largest relative amount gap that still earns credit, as a fraction (`0.01` is 1%)
16    pub amount_tolerance_percentage: f64,
17    /// Description similarity below which a
18    /// [`DescriptionDifference`](crate::reconciliation::MatchDifference::DescriptionDifference)
19    /// is recorded
20    pub description_similarity_threshold: f64,
21    /// Score at or above which a pair is matched without review
22    pub auto_match_threshold: f64,
23    /// Score at or above which a pair is paired up at all
24    pub partial_match_threshold: f64,
25    /// Score at or above which a partial match is flagged `auto_resolvable`
26    pub auto_resolve_threshold: f64,
27    /// Score at or above which a pair is offered as a suggestion for an unmatched transaction
28    pub suggestion_threshold: f64,
29    /// How many suggestions to keep per unmatched transaction
30    pub max_suggestions: usize,
31    /// Pairs dated further apart than this are never scored, bounding the comparison sweep
32    pub candidate_window_days: i64,
33    /// How much each dimension counts towards a pair's score
34    pub weights: ScoringWeights,
35}
36
37impl Default for ReconciliationConfig {
38    fn default() -> Self {
39        Self {
40            date_tolerance_days: 2,
41            amount_tolerance_percentage: 0.01,
42            description_similarity_threshold: 0.8,
43            auto_match_threshold: 0.95,
44            partial_match_threshold: 0.70,
45            auto_resolve_threshold: 0.90,
46            suggestion_threshold: 0.30,
47            max_suggestions: 5,
48            candidate_window_days: 30,
49            weights: ScoringWeights::default(),
50        }
51    }
52}
53
54/// Relative weight of each dimension when scoring a pair
55///
56/// Weights need not sum to one: a score is the weight earned divided by the weight available.
57/// The reference weight only counts when both sides carry a reference, so a missing reference is
58/// neutral rather than a penalty.
59#[derive(Debug, Clone, Copy, PartialEq)]
60pub struct ScoringWeights {
61    /// Agreement on amount
62    pub amount: f64,
63    /// Agreement on date
64    pub date: f64,
65    /// Similarity of the descriptions
66    pub description: f64,
67    /// Agreement on direction (debit or credit)
68    pub entry_type: f64,
69    /// Agreement on reference number, when both sides have one
70    pub reference: f64,
71}
72
73impl Default for ScoringWeights {
74    fn default() -> Self {
75        Self {
76            amount: 0.35,
77            date: 0.30,
78            description: 0.15,
79            entry_type: 0.10,
80            reference: 0.10,
81        }
82    }
83}
