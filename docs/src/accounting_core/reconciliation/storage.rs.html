1//! Extension points for persistence and external data import
2//!
3//! Neither trait is required to run [`ReconciliationEngine`](crate::reconciliation::ReconciliationEngine),
4//! which is pure. They exist so an application can plug its own database and statement formats in
5//! without this crate taking on a database or HTTP dependency.
6
7use async_trait::async_trait;
8use chrono::NaiveDate;
9use uuid::Uuid;
10
11use crate::reconciliation::types::*;
12use crate::types::{ListResponse, PaginationOption};

14/// Persistence for reconciliation runs
15///
16/// Implement this over PostgreSQL, SQLite or anything else; see
17/// [`MemoryReconciliationStorage`](crate::utils::memory_storage::MemoryReconciliationStorage) for a
18/// reference implementation.
19#[async_trait]
20pub trait ReconciliationStorage: Send + Sync {
21    /// Load one account's legs of the transactions booked in a date range, inclusive
22    async fn get_ledger_transactions(
23        &self,
24        account_id: &str,
25        start_date: NaiveDate,
26        end_date: NaiveDate,
27    ) -> ReconciliationResult<Vec<LedgerTransaction>>;
28
29    /// Store a completed report
30    async fn save_reconciliation_report(
31        &mut self,
32        report: &ReconciliationReport,
33    ) -> ReconciliationResult<()>;
34
35    /// Fetch a report by id
36    async fn get_reconciliation_report(
37        &self,
38        report_id: Uuid,
39    ) -> ReconciliationResult<Option<ReconciliationReport>>;
40
41    /// List the reports covering an account, newest first
42    async fn list_reconciliation_reports(
43        &self,
44        account_id: &str,
45        pagination: PaginationOption,
46    ) -> ReconciliationResult<ListResponse<ReconciliationReport>>;
47
48    /// Record that a ledger transaction has been reconciled against an external one
49    async fn mark_transaction_as_reconciled(
50        &mut self,
51        ledger_id: &str,
52        external_id: &str,
53        reconciliation_id: Uuid,
54    ) -> ReconciliationResult<()>;
55}

57/// Turns raw statement data into [`ExternalTransaction`]s
58///
59/// Statement layouts vary by bank and by gateway, so this crate ships no implementations. An
60/// implementation is responsible for one thing beyond parsing: expressing `entry_type` from the
61/// **ledger's** point of view. Money arriving in a bank account is a credit on the bank's
62/// statement but a debit in the ledger's bank account, and the engine compares them directly.
63#[async_trait]
64pub trait ExternalDataParser: Send + Sync {
65    /// Parse a statement, settlement file or API payload
66    async fn parse(&self, data: &[u8]) -> ReconciliationResult<Vec<ExternalTransaction>>;
67}
