1//! Pair scoring: how alike a ledger and an external transaction are, and how they differ
2//!
3//! Each dimension (amount, date, description, direction, reference) is scored on its own; the
4//! pair's score is the weight earned over the weight available.
5
6use bigdecimal::{Signed, ToPrimitive};
7
8use crate::reconciliation::config::ReconciliationConfig;
9use crate::reconciliation::similarity;
10use crate::reconciliation::types::{ExternalTransaction, LedgerTransaction, MatchDifference};
11
12/// Score given to a pair matched by reference or exact agreement
13pub const EXACT_MATCH_SCORE: f64 = 1.0;
14
15/// One dimension's contribution to a pair's score
16#[derive(Debug, Clone, PartialEq)]
17pub(super) struct DimensionScore {
18    /// Weight this dimension carries
19    pub(super) weight: f64,
20    /// Share of the weight earned, in `0.0..=weight`
21    pub(super) earned: f64,
22    /// How the two sides disagree on this dimension, if they do
23    pub(super) difference: Option<MatchDifference>,
24}
25
26impl DimensionScore {
27    fn full(weight: f64) -> Self {
28        Self {
29            weight,
30            earned: weight,
31            difference: None,
32        }
33    }
34}
35
36/// Score how alike two transactions are, in `0.0..=1.0`, along with every way they disagree
37#[must_use]
38pub fn score_pair(
39    config: &ReconciliationConfig,
40    ledger: &LedgerTransaction,
41    external: &ExternalTransaction,
42) -> (f64, Vec<MatchDifference>) {
43    let dimensions = [
44        Some(score_amount(config, ledger, external)),
45        Some(score_date(config, ledger, external)),
46        Some(score_description(config, ledger, external)),
47        Some(score_entry_type(config, ledger, external)),
48        score_reference(config, ledger, external),
49    ];
50
51    let (earned, weight_sum, differences) = dimensions.into_iter().flatten().fold(
52        (0.0, 0.0, Vec::new()),
53        |(earned, weight_sum, mut differences), dimension| {
54            differences.extend(dimension.difference);
55            (
56                earned + dimension.earned,
57                weight_sum + dimension.weight,
58                differences,
59            )
60        },
61    );
62
63    let score = if weight_sum > 0.0 {
64        (earned / weight_sum).clamp(0.0, 1.0)
65    } else {
66        0.0
67    };
68
69    (score, differences)
70}

72/// Full credit for equal amounts, decaying linearly to nothing at the tolerance
73pub(super) fn score_amount(
74    config: &ReconciliationConfig,
75    ledger: &LedgerTransaction,
76    external: &ExternalTransaction,
77) -> DimensionScore {
78    let weight = config.weights.amount;
79    if ledger.amount == external.amount {
80        return DimensionScore::full(weight);
81    }
82
83    let difference = (&ledger.amount - &external.amount).abs();
84    let scale = ledger.amount.abs().max(external.amount.abs());
85    let tolerance = config.amount_tolerance_percentage;
86
87    // A ratio too large for f64 counts as a 100% gap
88    let relative = (scale.is_positive() && tolerance > 0.0)
89        .then(|| (&difference / &scale).to_f64().unwrap_or(1.0))
90        .filter(|&relative| relative < tolerance);
91
92    DimensionScore {
93        weight,
94        earned: relative.map_or(0.0, |relative| weight * (1.0 - relative / tolerance)),
95        difference: Some(MatchDifference::AmountDifference {
96            ledger_amount: ledger.amount.clone(),
97            external_amount: external.amount.clone(),
98            difference,
99        }),
100    }
101}

103/// Full credit on the same day, decaying linearly to nothing at the date tolerance
104#[allow(clippy::cast_precision_loss)] // day counts are tiny
105fn score_date(
106    config: &ReconciliationConfig,
107    ledger: &LedgerTransaction,
108    external: &ExternalTransaction,
109) -> DimensionScore {
110    let weight = config.weights.date;
111    let days_diff = (ledger.date - external.date).num_days().abs();
112    if days_diff == 0 {
113        return DimensionScore::full(weight);
114    }
115
116    let tolerance = config.date_tolerance_days;
117    let earned = if tolerance > 0 {
118        let decay = 1.0 - (days_diff as f64 / tolerance as f64);
119        weight * decay.max(0.0)
120    } else {
121        0.0
122    };
123
124    DimensionScore {
125        weight,
126        earned,
127        difference: Some(MatchDifference::DateDifference {
128            ledger_date: ledger.date,
129            external_date: external.date,
130            days_diff,
131        }),
132    }
133}

135/// Credit in proportion to description similarity
136fn score_description(
137    config: &ReconciliationConfig,
138    ledger: &LedgerTransaction,
139    external: &ExternalTransaction,
140) -> DimensionScore {
141    let weight = config.weights.description;
142    let similarity = similarity::similarity(&ledger.description, &external.description);
143
144    DimensionScore {
145        weight,
146        earned: weight * similarity,
147        difference: (similarity < config.description_similarity_threshold).then(|| {
148            MatchDifference::DescriptionDifference {
149                ledger_description: ledger.description.clone(),
150                external_description: external.description.clone(),
151                similarity,
152            }
153        }),
154    }
155}

157/// All or nothing on direction
158fn score_entry_type(
159    config: &ReconciliationConfig,
160    ledger: &LedgerTransaction,
161    external: &ExternalTransaction,
162) -> DimensionScore {
163    let weight = config.weights.entry_type;
164    if ledger.entry_type == external.entry_type {
165        return DimensionScore::full(weight);
166    }
167
168    DimensionScore {
169        weight,
170        earned: 0.0,
171        difference: Some(MatchDifference::EntryTypeMismatch {
172            ledger_entry_type: ledger.entry_type,
173            external_entry_type: external.entry_type,
174        }),
175    }
176}

178/// All or nothing on reference, and only weighed when both sides have one
179fn score_reference(
180    config: &ReconciliationConfig,
181    ledger: &LedgerTransaction,
182    external: &ExternalTransaction,
183) -> Option<DimensionScore> {
184    let ledger_reference = normalized_reference(ledger.reference.as_deref())?;
185    let external_reference = normalized_reference(external.reference.as_deref())?;
186    let weight = config.weights.reference;
187
188    if ledger_reference == external_reference {
189        return Some(DimensionScore::full(weight));
190    }
191
192    Some(DimensionScore {
193        weight,
194        earned: 0.0,
195        difference: Some(MatchDifference::ReferenceMismatch {
196            ledger_reference: ledger.reference.clone(),
197            external_reference: external.reference.clone(),
198        }),
199    })
200}

202/// Trim and upper-case a reference, treating blank references as absent
203pub(crate) fn normalized_reference(reference: Option<&str>) -> Option<String> {
204    let trimmed = reference?.trim();
205    (!trimmed.is_empty()).then(|| trimmed.to_uppercase())
206}

208/// Whether a pair must be seen by a person regardless of how well it scores.
209///
210/// A gap in the amount means money is unaccounted for, and a flipped direction means the entry
211/// went the wrong way. Both are reconciliation findings in their own right, so neither is ever
212/// swept into an automatic match no matter how well the rest of the record agrees.
213pub(crate) fn requires_review(differences: &[MatchDifference]) -> bool {
214    differences.iter().any(|difference| {
215        matches!(
216            difference,
217            MatchDifference::AmountDifference { .. } | MatchDifference::EntryTypeMismatch { .. }
218        )
219    })
