1//! Credit notes against sales invoices under any tax regime
2//!
3//! A seller issues a [`CreditNote`] to reduce the value of an invoice it issued: goods returned,
4//! a discount after the sale, or an amount over-billed. The note is built from the [`Invoice`] it
5//! corrects, so it bills the same customer under the same regime, and it records the invoice as
6//! an [`OriginalInvoice`]. It may not be dated before the invoice, nor credit more than the
7//! invoice was worth. [`CreditNote::to_entries`] posts it as the reverse of an invoice.
8
9use crate::document::error::DocumentError;
10use crate::document::line::LineItem;
11use crate::document::pricing::Document;
12use crate::document::tax::{TaxError, TaxRegime};
13use crate::sales::invoice::{check_document, Invoice};
14use crate::sales::posting::{sales_entries, SalesAccounts};
15use crate::types::Entry;
16use bigdecimal::BigDecimal;
17use chrono::NaiveDate;
18use serde::{Deserialize, Serialize};

20/// The invoice a credit note corrects
21#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
22pub struct OriginalInvoice {
23    /// Invoice number
24    pub number: String,
25    /// Date of issue
26    pub date: NaiveDate,
27    /// Invoice value, tax included: the most the note may credit
28    pub value: BigDecimal,
29}

31/// A credit note: the seller reduces the value of an invoice it issued
32///
33/// # Example
34///
35/// ```
36/// use accounting_core::document::{FlatRate, LineItem};
37/// use accounting_core::sales::{CreditNote, Invoice, SalesAccounts};
38/// use accounting_core::EntryType;
39/// use bigdecimal::BigDecimal;
40/// use chrono::NaiveDate;
41///
42/// # fn main() -> Result<(), accounting_core::Error> {
43/// let line = |quantity| LineItem::new("Chair", BigDecimal::from(quantity), BigDecimal::from(50),
44///     BigDecimal::from(20));
45/// let date = |day| NaiveDate::from_ymd_opt(2025, 3, day).unwrap();
46/// let invoice = Invoice::new("INV-001", date(1), "Acme Ltd", vec![line(10)?], FlatRate::new("VAT"))?;
47///
48/// // Two chairs come back
49/// let note = CreditNote::new(&invoice, "CN-001", date(5), vec![line(2)?])?;
50/// let accounts = SalesAccounts::new("receivable", "sales returns").with_output_tax("VAT", "vat");
51/// let entries = note.to_entries(&accounts)?;
52/// assert_eq!(entries[0].entry_type, EntryType::Credit);
53/// assert_eq!(entries[0].amount, BigDecimal::from(120));
54/// # Ok(())
55/// # }
56/// ```
57#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
58pub struct CreditNote<R> {
59    /// Note number
60    pub number: String,
61    /// Date of issue
62    pub date: NaiveDate,
63    /// The invoice the note corrects
64    pub original: OriginalInvoice,
65    /// Who is credited, the same as on the invoice
66    pub customer: String,
67    /// What is credited: the value and tax each line takes off the invoice
68    pub lines: Vec<LineItem>,
69    /// The regime the invoice was taxed under
70    pub regime: R,
71}
72
73impl<R: TaxRegime<Error = TaxError> + Clone> CreditNote<R> {
74    /// Create a credit note of `lines` against `invoice`
75    ///
76    /// # Errors
77    ///
78    /// [`DocumentError::EmptyNumber`] or [`DocumentError::NoLines`],
79    /// [`DocumentError::NoteBeforeOriginal`] for a note dated before the invoice,
80    /// [`DocumentError::CreditExceedsInvoice`] for a note worth more than the invoice, or any
81    /// error from [`Document::tax_breakdown`].
82    pub fn new(
83        invoice: &Invoice<R>,
84        number: impl Into<String>,
85        date: NaiveDate,
86        lines: Vec<LineItem>,
87    ) -> Result<Self, DocumentError> {
88        let number = check_document(number.into(), &lines)?;
89        if date < invoice.date {
90            return Err(DocumentError::NoteBeforeOriginal {
91                note_date: date,
92                original_date: invoice.date,
93            });
94        }
95        let note = Self {
96            number,
97            date,
98            original: OriginalInvoice {
99                number: invoice.number.clone(),
100                date: invoice.date,
101                value: invoice.tax_breakdown()?.total,
102            },
103            customer: invoice.customer.clone(),
104            lines,
105            regime: invoice.regime.clone(),
106        };
107        let credited = note.tax_breakdown()?.total;
108        if credited > note.original.value {
109            return Err(DocumentError::CreditExceedsInvoice {
110                credited,
111                invoice_value: note.original.value,
112            });
113        }
114        Ok(note)
115    }
116
117    /// Journal entries that record this note: an invoice's legs on the opposite sides
118    ///
119    /// Pass a sales returns account as `accounts.revenue` to keep returns apart from sales.
120    ///
121    /// # Errors
122    ///
123    /// As [`Invoice::to_entries`].
124    pub fn to_entries(&self, accounts: &SalesAccounts) -> Result<Vec<Entry>, DocumentError> {
125        Ok(sales_entries(&self.tax_breakdown()?, accounts, |leg| {
126            leg.side().opposite()
127        })?)
128    }
129}

133    type Regime = R;

150    }
