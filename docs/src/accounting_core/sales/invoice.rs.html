1//! Sales invoices under any tax regime
2//!
3//! An [`Invoice`] is a numbered, dated bill to a customer: [`LineItem`]s, an optional discount off
4//! the subtotal, and the [`TaxRegime`] its lines are taxed under. Its pricing comes from
5//! [`Document`], and [`Invoice::to_entries`] posts it to the ledger.
6
7use crate::document::discount::Discount;
8use crate::document::error::DocumentError;
9use crate::document::line::LineItem;
10use crate::document::posting::PostingLeg;
11use crate::document::pricing::Document;
12use crate::document::tax::{TaxError, TaxRegime};
13use crate::sales::posting::{sales_entries, SalesAccounts};
14use crate::types::Entry;
15use chrono::NaiveDate;
16use serde::{Deserialize, Serialize};

18/// An invoice taxed under the regime `R`, such as [`FlatRate`](crate::document::FlatRate) for VAT
19///
20/// Deserialising goes through the same checks as [`Invoice::new`] and
21/// [`Invoice::with_discount`].
22///
23/// # Example
24///
25/// ```
26/// use accounting_core::document::{Document, FlatRate, LineItem};
27/// use accounting_core::sales::{Invoice, SalesAccounts};
28/// use bigdecimal::BigDecimal;
29/// use chrono::NaiveDate;
30///
31/// # fn main() -> Result<(), accounting_core::Error> {
32/// let line = LineItem::new("Consulting", BigDecimal::from(10), BigDecimal::from(100),
33///     BigDecimal::from(20))?;
34/// let invoice = Invoice::new("INV-001", NaiveDate::from_ymd_opt(2025, 3, 1).unwrap(),
35///     "Acme Ltd", vec![line], FlatRate::new("VAT"))?;
36/// assert_eq!(invoice.tax_breakdown()?.total, BigDecimal::from(1200));
37///
38/// let accounts = SalesAccounts::new("receivable", "sales").with_output_tax("VAT", "vat-payable");
39/// let entries = invoice.to_entries(&accounts)?;
40/// assert_eq!(entries.len(), 3);
41/// # Ok(())
42/// # }
43/// ```
44#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
45#[serde(
46    try_from = "RawInvoice<R>",
47    bound(deserialize = "R: TaxRegime<Error = TaxError> + Clone + Deserialize<'de>")
48)]
49pub struct Invoice<R> {
50    /// Invoice number
51    pub number: String,
52    /// Date of issue
53    pub date: NaiveDate,
54    /// Who is billed
55    pub customer: String,
56    /// What is billed
57    pub lines: Vec<LineItem>,
58    /// Discount off the subtotal, before tax, shared across the lines by value
59    #[serde(default, skip_serializing_if = "Option::is_none")]
60    pub discount: Option<Discount>,
61    /// The regime the lines are taxed under
62    pub regime: R,
63}
64
65/// Unvalidated shape of an [`Invoice`], used when deserialising
66#[derive(Deserialize)]
67struct RawInvoice<R> {
68    number: String,
69    date: NaiveDate,
70    customer: String,
71    lines: Vec<LineItem>,
72    #[serde(default)]
73    discount: Option<Discount>,
74    regime: R,
75}
76
77impl<R: TaxRegime<Error = TaxError> + Clone> TryFrom<RawInvoice<R>> for Invoice<R> {
78    type Error = DocumentError;
79
80    fn try_from(raw: RawInvoice<R>) -> Result<Self, Self::Error> {
81        let invoice = Self::new(raw.number, raw.date, raw.customer, raw.lines, raw.regime)?;
82        match raw.discount {
83            Some(discount) => invoice.with_discount(discount),
84            None => Ok(invoice),
85        }
86    }
87}

89impl<R: TaxRegime<Error = TaxError> + Clone> Invoice<R> {
90    /// Create an invoice of `lines` to `customer`, taxed under `regime`
91    ///
92    /// The lines were validated when they were built.
93    ///
94    /// # Errors
95    ///
96    /// [`DocumentError::EmptyNumber`] for a blank number, or [`DocumentError::NoLines`].
97    pub fn new(
98        number: impl Into<String>,
99        date: NaiveDate,
100        customer: impl Into<String>,
101        lines: Vec<LineItem>,
102        regime: R,
103    ) -> Result<Self, DocumentError> {
104        let number = check_document(number.into(), &lines)?;
105        Ok(Self {
106            number,
107            date,
108            customer: customer.into(),
109            lines,
110            discount: None,
111            regime,
112        })
113    }
114
115    /// The invoice with `discount` taken off its subtotal, before tax
116    ///
117    /// # Errors
118    ///
119    /// [`DocumentError::InvalidDiscount`] if the discount is out of range or more than the
120    /// [`Document::subtotal`].
121    pub fn with_discount(self, discount: Discount) -> Result<Self, DocumentError> {
122        let invoice = Self {
123            discount: Some(discount),
124            ..self
125        };
126        invoice.check_discount()?;
127        Ok(invoice)
128    }
129
130    /// Journal entries that record this invoice: the receivable debited with the total, sales
131    /// credited with the taxable value, and each tax credited to its account
132    ///
133    /// Pass them to [`TransactionBuilder::entry`](crate::ledger::TransactionBuilder::entry).
134    ///
135    /// # Errors
136    ///
137    /// [`DocumentError::Posting`] if the total is zero or a tax has no account, or any error from
138    /// [`Document::tax_breakdown`].
139    pub fn to_entries(&self, accounts: &SalesAccounts) -> Result<Vec<Entry>, DocumentError> {
140        Ok(sales_entries(
141            &self.tax_breakdown()?,
142            accounts,
143            PostingLeg::side,
144        )?)
145    }
146}

148impl<R: TaxRegime<Error = TaxError> + Clone> Document for Invoice<R> {
149    type Line = LineItem;
150    type Regime = R;
151    type Error = DocumentError;
152
153    fn number(&self) -> &str {
154        &self.number
155    }
156
157    fn date(&self) -> NaiveDate {
158        self.date
159    }
160
161    fn lines(&self) -> &[LineItem] {
162        &self.lines
163    }
164
165    fn document_discount(&self) -> Option<&Discount> {
166        self.discount.as_ref()
167    }
168
169    fn regime(&self) -> R {
170        self.regime.clone()
171    }
172}

174/// `number` trimmed, if it is not blank and `lines` is not empty
175pub(super) fn check_document(number: String, lines: &[LineItem]) -> Result<String, DocumentError> {
176    let number = number.trim().to_string();
177    if number.is_empty() {
178        Err(DocumentError::EmptyNumber)
179    } else if lines.is_empty() {
180        Err(DocumentError::NoLines)
181    } else {
182        Ok(number)
183    }
