1//! HSN/SAC master data and default GST rates
2//!
3//! The master is compiled into the crate from `src/data/hsn_master.json`. It covers common codes
4//! for Indian SMEs under the GST schedule named by [`HsnMaster::schedule`], not the full CBIC
5//! tariff. Rates are defaults: some goods and services carry conditional rates (price thresholds,
6//! input tax credit options), so callers can always pass an explicit rate instead.
7
8use bigdecimal::BigDecimal;
9use chrono::NaiveDate;
10use serde::{Deserialize, Serialize};
11use std::collections::HashMap;
12use std::sync::LazyLock;
13
14/// The embedded master, parsed once on first use
15static MASTER: LazyLock<HsnMaster> = LazyLock::new(|| {
16    serde_json::from_str(include_str!("../data/hsn_master.json"))
17        .expect("embedded HSN/SAC master is valid JSON")
18});
19
20/// Whether a code classifies goods or services
21#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
22#[serde(rename_all = "lowercase")]
23pub enum HsnSacKind {
24    /// Harmonized System of Nomenclature code (goods)
25    Hsn,
26    /// Services Accounting Code (services)
27    Sac,
28}
29
30/// One code in the master
31#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
32pub struct HsnSacEntry {
33    /// HSN or SAC code: 4, 6 or 8 digits
34    pub code: String,
35    /// Goods or services
36    pub kind: HsnSacKind,
37    /// Short description of what the code covers
38    pub description: String,
39    /// Default total GST rate as a percentage (e.g. 18 for 18%)
40    pub gst_rate: BigDecimal,
41}
42
43/// Lookup table of HSN/SAC codes and their default GST rates
44#[derive(Debug, Clone, Deserialize)]
45#[serde(from = "RawMaster")]
46pub struct HsnMaster {
47    schedule: String,
48    effective_from: NaiveDate,
49    entries: Vec<HsnSacEntry>,
50    index: HashMap<String, usize>,
51}
52
53/// On-disk shape of the master, before indexing
54#[derive(Deserialize)]
55struct RawMaster {
56    schedule: String,
57    effective_from: NaiveDate,
58    entries: Vec<HsnSacEntry>,
59}
60
61impl From<RawMaster> for HsnMaster {
62    fn from(raw: RawMaster) -> Self {
63        let index = raw
64            .entries
65            .iter()
66            .enumerate()
67            .map(|(i, entry)| (entry.code.clone(), i))
68            .collect();
69
70        Self {
71            schedule: raw.schedule,
72            effective_from: raw.effective_from,
73            entries: raw.entries,
74            index,
75        }
76    }
77}
78
79impl HsnMaster {
80    /// The master embedded in the crate
81    pub fn global() -> &'static HsnMaster {
82        &MASTER
83    }
84
85    /// Name of the GST rate schedule the default rates follow
86    pub fn schedule(&self) -> &str {
87        &self.schedule
88    }
89
90    /// Date the rate schedule took effect
91    pub fn effective_from(&self) -> NaiveDate {
92        self.effective_from
93    }
94
95    /// All entries, in file order
96    pub fn entries(&self) -> &[HsnSacEntry] {
97        &self.entries
98    }
99
100    /// Find the entry for a code
101    ///
102    /// Tries the exact code first, then its 6- and 4-digit headings, so an 8-digit tariff item
103    /// such as `84713010` resolves to `847130` or `8471`. Malformed codes return `None`.
104    pub fn lookup(&self, code: &str) -> Option<&HsnSacEntry> {
105        if !is_valid_hsn_sac(code) {
106            return None;
107        }
108
109        [code.len(), 6, 4]
110            .into_iter()
111            .filter(|&len| len <= code.len())
112            .find_map(|len| self.index.get(&code[..len]))
113            .map(|&i| &self.entries[i])
114    }
115
116    /// Default GST rate for a code, if the master knows it
117    pub fn default_rate(&self, code: &str) -> Option<BigDecimal> {
118        self.lookup(code).map(|entry| entry.gst_rate.clone())
119    }
120}
121
122/// Whether a code has the shape of an HSN/SAC code: 4, 6 or 8 ASCII digits
123pub(crate) fn is_valid_hsn_sac(code: &str) -> bool {
124    matches!(code.len(), 4 | 6 | 8) && code.bytes().all(|b| b.is_ascii_digit())
125}
