1//! GST invoices under Indian rules
2//!
3//! [`GstInvoice`] models a B2B tax invoice: invoice number, date, seller and buyer [`Gstin`], and
4//! [`GstLineItem`]s carrying an HSN/SAC code. Every identifier is validated on construction, so an
5//! invoice that exists is well-formed. Whether tax is split as CGST + SGST or charged as IGST
6//! follows from the seller and buyer state codes; [`GstInvoice::breakdown`] returns the totals as
7//! a [`GstBreakdown`].
8//!
9//! [`HsnMaster`] holds common HSN/SAC codes with their default GST rates;
10//! [`GstLineItem::with_default_rate`] builds a line at the rate for its code.
11//!
12//! The tax arithmetic comes from [`crate::tax::gst`]. The invoice types are also re-exported at the
13//! crate root.
14//!
15//! # Example
16//!
17//! ```
18//! use accounting_core::invoice::{GstInvoice, GstLineItem, Gstin};
19//! use bigdecimal::BigDecimal;
20//! use chrono::NaiveDate;
21//!
22//! # fn main() -> Result<(), accounting_core::invoice::InvoiceError> {
23//! let item = GstLineItem::new(
24//!     "998314".to_string(),
25//!     "IT consulting".to_string(),
26//!     BigDecimal::from(10),
27//!     BigDecimal::from(1500),
28//!     BigDecimal::from(18),
29//! )?;
30//!
31//! let invoice = GstInvoice::new(
32//!     "INV/2024-25/001".to_string(),
33//!     NaiveDate::from_ymd_opt(2024, 11, 15).unwrap(),
34//!     Gstin::parse("27AAPFU0939F1ZV")?,
35//!     Gstin::parse("27AAPFU0939F2ZU")?,
36//!     vec![item],
37//! )?;
38//!
39//! let breakdown = invoice.breakdown()?;
40//! assert_eq!(breakdown.cgst, BigDecimal::from(1350));
41//! assert_eq!(breakdown.sgst, BigDecimal::from(1350));
42//! assert_eq!(breakdown.total, BigDecimal::from(17700));
43//! # Ok(())
44//! # }
45//! ```
