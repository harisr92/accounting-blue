1//! Ledger posting for a GST invoice
2//!
3//! An issued invoice is one journal entry: the buyer owes the total, the taxable value is sales
4//! revenue, and each tax component is owed to the government. [`posting_legs`] turns a
5//! [`GstBreakdown`] into those legs and [`GstInvoice::to_entries`] maps them onto accounts. The
6//! legs always balance, because the breakdown's total is its taxable value plus its tax.
7//!
8//! A [`CreditNote`] posts the same legs on the opposite sides, reversing the part of the invoice
9//! it credits: [`CreditNote::to_entries`].
10

19/// One leg of an invoice's journal entry
20#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
21pub enum PostingLeg {
22    /// Debit: the total the buyer owes
23    Receivable,
24    /// Credit: the taxable value earned as revenue
25    Sales,
26    /// Credit: central GST owed on an intra-state supply
27    CgstOutput,
28    /// Credit: state GST owed on an intra-state supply
29    SgstOutput,
30    /// Credit: integrated GST owed on an inter-state supply
31    IgstOutput,
32}
33
34impl PostingLeg {
35    /// Every leg, debit first
36    pub const ALL: [PostingLeg; 5] = [
37        Self::Receivable,
38        Self::Sales,
39        Self::CgstOutput,
40        Self::SgstOutput,
41        Self::IgstOutput,
42    ];
43
44    /// Side of the journal entry this leg posts to
45    #[must_use]
46    pub fn side(self) -> EntryType {
47        match self {
48            Self::Receivable => EntryType::Debit,
49            Self::Sales | Self::CgstOutput | Self::SgstOutput | Self::IgstOutput => {
50                EntryType::Credit
51            }
52        }
53    }
54
55    /// The amount of `breakdown` this leg carries
56    #[must_use]
57    pub fn amount(self, breakdown: &GstBreakdown) -> &BigDecimal {
58        match self {
59            Self::Receivable => &breakdown.total,
60            Self::Sales => &breakdown.taxable_value,
61            Self::CgstOutput => &breakdown.cgst,
62            Self::SgstOutput => &breakdown.sgst,
63            Self::IgstOutput => &breakdown.igst,
64        }
65    }
66}
67
68impl fmt::Display for PostingLeg {
69    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
70        f.write_str(match self {
71            Self::Receivable => "Receivable",
72            Self::Sales => "Sales",
73            Self::CgstOutput => "CGST output",
74            Self::SgstOutput => "SGST output",
75            Self::IgstOutput => "IGST output",
76        })
77    }
78}

80/// Legs of the journal entry for `breakdown`, leaving out those with a zero amount
81///
82/// A zero leg would fail [`Transaction::validate`](crate::types::Transaction::validate), and an
83/// invoice only ever carries CGST + SGST or IGST, never both.
84#[must_use]
85pub fn posting_legs(breakdown: &GstBreakdown) -> Vec<(PostingLeg, BigDecimal)> {
86    PostingLeg::ALL
87        .into_iter()
88        .map(|leg| (leg, leg.amount(breakdown)))
89        .filter(|(_, amount)| !amount.is_zero())
90        .map(|(leg, amount)| (leg, amount.clone()))
91        .collect()
92}

94/// Why an invoice could not be posted to the ledger
95#[derive(Debug, thiserror::Error)]
96pub enum PostingError {
97    /// The invoice breaks compliance rules; carries the error-severity issues
98    #[error("invoice is not compliant ({} error(s)), first: {}", .0.len(), .0.first().map_or(String::new(), ToString::to_string))]
99    NotCompliant(Vec<ComplianceIssue>),
100    /// The invoice total is zero, so there is nothing to post
101    #[error("invoice total is zero, so there is nothing to post")]
102    NothingToPost,
103    /// The invoice's tax could not be calculated
104    #[error(transparent)]
105    Invoice(#[from] InvoiceError),
106}

108/// Accounts an invoice posts to, one per [`PostingLeg`]
109#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
110pub struct InvoiceAccounts {
111    /// Debited with the invoice total (accounts receivable or cash)
112    pub receivable: String,
113    /// Credited with the taxable value
114    pub sales: String,
115    /// Credited with CGST
116    pub cgst_output: String,
117    /// Credited with SGST
118    pub sgst_output: String,
119    /// Credited with IGST
120    pub igst_output: String,
121}
122
123impl InvoiceAccounts {
124    /// Name the account for each leg
125    pub fn new(
126        receivable: impl Into<String>,
127        sales: impl Into<String>,
128        cgst_output: impl Into<String>,
129        sgst_output: impl Into<String>,
130        igst_output: impl Into<String>,
131    ) -> Self {
132        Self {
133            receivable: receivable.into(),
134            sales: sales.into(),
135            cgst_output: cgst_output.into(),
136            sgst_output: sgst_output.into(),
137            igst_output: igst_output.into(),
138        }
139    }
140
141    /// Account id `leg` posts to
142    #[must_use]
143    pub fn account_for(&self, leg: PostingLeg) -> &str {
144        match leg {
145            PostingLeg::Receivable => &self.receivable,
146            PostingLeg::Sales => &self.sales,
147            PostingLeg::CgstOutput => &self.cgst_output,
148            PostingLeg::SgstOutput => &self.sgst_output,
149            PostingLeg::IgstOutput => &self.igst_output,
150        }
151    }
152}

154impl GstInvoice {
155    /// Journal entries that record this invoice in the ledger
156    ///
157    /// The invoice must pass the error-severity compliance rules of
158    /// [`validate_invoice`](super::validate_invoice), checked as of its own date: every error rule
159    /// applies except the future-date check, which depends on when the caller posts it. Warnings
160    /// don't block the posting. The entries balance: the receivable debit equals the sales and tax
161    /// credits. Pass them to [`TransactionBuilder::entry`](crate::ledger::TransactionBuilder::entry)
162    /// to build the transaction.
163    ///
164    /// # Errors
165    ///
166    /// [`PostingError::NotCompliant`] with the error-severity issues if the invoice breaks a
167    /// compliance rule, [`PostingError::NothingToPost`] if its total is zero, or
168    /// [`PostingError::Invoice`] for any error from [`GstInvoice::breakdown`].
169    ///
170    /// # Example
171    ///
172    /// ```
173    /// use accounting_core::invoice::{GstInvoice, GstLineItem, Gstin, InvoiceAccounts};
174    /// use accounting_core::EntryType;
175    /// use bigdecimal::BigDecimal;
176    /// use chrono::NaiveDate;
177    ///
178    /// # fn main() -> Result<(), accounting_core::Error> {
179    /// let item = GstLineItem::new("998314", "IT consulting", BigDecimal::from(1),
180    ///     BigDecimal::from(1000), BigDecimal::from(18))?;
181    /// let invoice = GstInvoice::new("INV-001", NaiveDate::from_ymd_opt(2024, 11, 15).unwrap(),
182    ///     Gstin::parse("27AAPFU0939F1ZV")?, Gstin::parse("29AAPFU0939F1ZR")?, vec![item])?;
183    ///
184    /// let accounts = InvoiceAccounts::new("receivable", "sales", "cgst", "sgst", "igst");
185    /// let entries = invoice.to_entries(&accounts)?;
186    ///
187    /// // Inter-state: receivable 1180 = sales 1000 + IGST 180
188    /// assert_eq!(entries.len(), 3);
189    /// assert_eq!(entries[0].entry_type, EntryType::Debit);
190    /// assert_eq!(entries[0].amount, BigDecimal::from(1180));
191    /// assert_eq!(entries[2].account_id, "igst");
192    /// # Ok(())
193    /// # }
194    /// ```
195    pub fn to_entries(&self, accounts: &InvoiceAccounts) -> Result<Vec<Entry>, PostingError> {
196        let errors = compliance_errors(self, self.invoice_date);
197        document_entries(self, errors, accounts, PostingLeg::side)
198    }
199}

201impl CreditNote {
202    /// Journal entries that record this credit note in the ledger
203    ///
204    /// The legs are an invoice's, on the opposite sides: sales and the output tax are debited,
205    /// and the receivable is credited with the note's total. Pass a sales returns account as
206    /// `accounts.sales` to keep returns apart from sales. The note must pass the error-severity
207    /// rules of [`validate_credit_note`](super::validate_credit_note) as of its own date.
208    ///
209    /// # Errors
210    ///
211    /// As [`GstInvoice::to_entries`].
212    ///
213    /// # Example
214    ///
215    /// ```
216    /// use accounting_core::invoice::{CreditNote, GstInvoice, GstLineItem, Gstin, InvoiceAccounts};
217    /// use accounting_core::EntryType;
218    /// use bigdecimal::BigDecimal;
219    /// use chrono::NaiveDate;
220    ///
221    /// # fn main() -> Result<(), accounting_core::Error> {
222    /// let item = GstLineItem::new("998314", "IT consulting", BigDecimal::from(1),
223    ///     BigDecimal::from(1000), BigDecimal::from(18))?;
224    /// let invoice = GstInvoice::new("INV-001", NaiveDate::from_ymd_opt(2024, 11, 15).unwrap(),
225    ///     Gstin::parse("27AAPFU0939F1ZV")?, Gstin::parse("29AAPFU0939F1ZR")?, vec![item.clone()])?;
226    /// let note = CreditNote::new(&invoice, "CN-001", NaiveDate::from_ymd_opt(2024, 11, 20).unwrap(),
227    ///     vec![item])?;
228    ///
229    /// let accounts = InvoiceAccounts::new("receivable", "sales returns", "cgst", "sgst", "igst");
230    /// let entries = note.to_entries(&accounts)?;
231    ///
232    /// // Inter-state: sales returns 1000 + IGST 180 = receivable 1180, reversed
233    /// assert_eq!(entries[0].entry_type, EntryType::Credit);
234    /// assert_eq!(entries[0].amount, BigDecimal::from(1180));
235    /// assert_eq!(entries[1].entry_type, EntryType::Debit);
236    /// assert_eq!(entries[2].account_id, "igst");
237    /// # Ok(())
238    /// # }
239    /// ```
240    pub fn to_entries(&self, accounts: &InvoiceAccounts) -> Result<Vec<Entry>, PostingError> {
241        let errors = credit_note_errors(self, self.note_date);
242        document_entries(self, errors, accounts, |leg| leg.side().opposite())
243    }
244}

246/// The entries for `document`'s legs, each on the side `side` gives it, unless it has compliance
247/// `errors` or nothing to post
248fn document_entries(
249    document: &dyn GstDocument,
250    errors: Vec<ComplianceIssue>,
251    accounts: &InvoiceAccounts,
252    side: impl Fn(PostingLeg) -> EntryType,
253) -> Result<Vec<Entry>, PostingError> {
254    if !errors.is_empty() {
255        return Err(PostingError::NotCompliant(errors));
256    }
257
258    let legs = posting_legs(&document.breakdown()?);
259    if legs.is_empty() {
260        return Err(PostingError::NothingToPost);
261    }
262
263    Ok(legs
264        .into_iter()
265        .map(|(leg, amount)| {
266            Entry::new(
267                accounts.account_for(leg),
268                side(leg),
269                amount,
270                Some(leg.to_string()),
271            )
272        })
273        .collect())
