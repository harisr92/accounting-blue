1//! Drawing primitives in millimetres measured from the top-left corner of an A4 page
2
3use super::fonts::Face;
4use printpdf::{
5    Color, Greyscale, Line, LinePoint, Mm, Op, Point, Pt, TextItem, XObjectId, XObjectTransform,
6};
7
8/// A4 width in millimetres
9pub(super) const PAGE_WIDTH: f32 = 210.0;
10/// A4 height in millimetres
11pub(super) const PAGE_HEIGHT: f32 = 297.0;
12/// Thickness of rules, in points
13const RULE_WIDTH_PT: f32 = 0.5;
14/// Ink for text and rules
15const BLACK: f32 = 0.0;
16/// Millimetres in an inch, to turn a scale in millimetres per pixel into a DPI
17const MM_PER_INCH: f32 = 25.4;
18
19/// An image added to the document, with its size in pixels
20pub(super) struct Image {
21    /// Handle that draws the image
22    pub(super) id: XObjectId,
23    /// Width in pixels
24    pub(super) width_px: usize,
25    /// Height in pixels
26    pub(super) height_px: usize,
27}
28
29/// How text sits relative to its x position
30#[derive(Debug, Clone, Copy, PartialEq, Eq)]
31pub(super) enum Align {
32    /// Starts at x
33    Left,
34    /// Centred on x
35    Center,
36    /// Ends at x
37    Right,
38}
39
40/// A PDF point for a position measured from the top-left corner
41fn point(x: f32, y: f32) -> Point {
42    Point::new(Mm(x), Mm(PAGE_HEIGHT - y))
43}
44
45/// Ops that reset the pen at the start of a page
46pub(super) fn page_setup() -> Vec<Op> {
47    vec![
48        Op::SetOutlineThickness {
49            pt: Pt(RULE_WIDTH_PT),
50        },
51        Op::SetOutlineColor { col: grey(BLACK) },
52        Op::SetFillColor { col: grey(BLACK) },
53    ]
54}
55
56/// A grey level from 0 (black) to 1 (white)
57fn grey(level: f32) -> Color {
58    Color::Greyscale(Greyscale::new(level, None))
59}
60
61/// `text` in `face` at `size` points, on the baseline `y`, aligned to `x`
62pub(super) fn text(face: &Face, size: f32, x: f32, y: f32, align: Align, text: &str) -> Vec<Op> {
63    let x = match align {
64        Align::Left => x,
65        Align::Center => x - face.width_mm(text, size) / 2.0,
66        Align::Right => x - face.width_mm(text, size),
67    };
68    vec![
69        Op::StartTextSection,
70        Op::SetFont {
71            font: face.handle.clone(),
72            size: Pt(size),
73        },
74        Op::SetTextCursor { pos: point(x, y) },
75        Op::ShowText {
76            items: vec![TextItem::Text(text.to_string())],
77        },
78        Op::EndTextSection,
79    ]
80}
81
82/// Like [`text`], in grey
83pub(super) fn grey_text(
84    face: &Face,
85    size: f32,
86    level: f32,
87    (x, y): (f32, f32),
88    align: Align,
89    content: &str,
90) -> Vec<Op> {
91    std::iter::once(Op::SetFillColor { col: grey(level) })
92        .chain(text(face, size, x, y, align, content))
93        .chain(std::iter::once(Op::SetFillColor { col: grey(BLACK) }))
94        .collect()
95}
96
97/// A horizontal rule from `x1` to `x2` at `y`
98pub(super) fn rule(x1: f32, x2: f32, y: f32) -> Op {
99    Op::DrawLine {
100        line: Line {
101            points: [x1, x2]
102                .into_iter()
103                .map(|x| LinePoint {
104                    p: point(x, y),
105                    bezier: false,
106                })
107                .collect(),
108            is_closed: false,
109        },
110    }
111}
112
113/// `image` scaled to fit a `width` by `height` box whose top-left corner is at (`x`, `y`),
114/// keeping its aspect ratio, and the height it is drawn at; nothing for an empty image
115pub(super) fn image(
116    image: &Image,
117    (x, y): (f32, f32),
118    (width, height): (f32, f32),
119) -> (Vec<Op>, f32) {
120    if image.width_px == 0 || image.height_px == 0 {
121        return (Vec::new(), 0.0);
122    }
123    let mm_per_px = (width / image.width_px as f32).min(height / image.height_px as f32);
124    let drawn_height = mm_per_px * image.height_px as f32;
125    let lower_left = point(x, y + drawn_height);
126    let op = Op::UseXobject {
127        id: image.id.clone(),
128        transform: XObjectTransform {
129            translate_x: Some(lower_left.x),
130            translate_y: Some(lower_left.y),
131            dpi: Some(MM_PER_INCH / mm_per_px),
132            ..XObjectTransform::default()
133        },
134    };
135    (vec![op], drawn_height)
136}
