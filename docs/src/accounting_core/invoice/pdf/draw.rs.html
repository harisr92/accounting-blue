1//! Drawing primitives in millimetres measured from the top-left corner of an A4 page
2
3use super::fonts::Face;
4use printpdf::{Color, Greyscale, Line, LinePoint, Mm, Op, PaintMode, Point, Pt, Rect, TextItem};
5
6/// A4 width in millimetres
7pub(super) const PAGE_WIDTH: f32 = 210.0;
8/// A4 height in millimetres
9pub(super) const PAGE_HEIGHT: f32 = 297.0;
10/// Thickness of rules and frames, in points
11const RULE_WIDTH_PT: f32 = 0.5;
12/// Ink for text and rules
13const BLACK: f32 = 0.0;
14
15/// How text sits relative to its x position
16#[derive(Debug, Clone, Copy, PartialEq, Eq)]
17pub(super) enum Align {
18    /// Starts at x
19    Left,
20    /// Centred on x
21    Center,
22    /// Ends at x
23    Right,
24}
25
26/// A PDF point for a position measured from the top-left corner
27fn point(x: f32, y: f32) -> Point {
28    Point::new(Mm(x), Mm(PAGE_HEIGHT - y))
29}
30
31/// Ops that reset the pen at the start of a page
32pub(super) fn page_setup() -> Vec<Op> {
33    vec![
34        Op::SetOutlineThickness {
35            pt: Pt(RULE_WIDTH_PT),
36        },
37        Op::SetOutlineColor { col: grey(BLACK) },
38        Op::SetFillColor { col: grey(BLACK) },
39    ]
40}
41
42/// A grey level from 0 (black) to 1 (white)
43fn grey(level: f32) -> Color {
44    Color::Greyscale(Greyscale::new(level, None))
45}
46
47/// `text` in `face` at `size` points, on the baseline `y`, aligned to `x`
48pub(super) fn text(face: &Face, size: f32, x: f32, y: f32, align: Align, text: &str) -> Vec<Op> {
49    let x = match align {
50        Align::Left => x,
51        Align::Center => x - face.width_mm(text, size) / 2.0,
52        Align::Right => x - face.width_mm(text, size),
53    };
54    vec![
55        Op::StartTextSection,
56        Op::SetFont {
57            font: face.handle.clone(),
58            size: Pt(size),
59        },
60        Op::SetTextCursor { pos: point(x, y) },
61        Op::ShowText {
62            items: vec![TextItem::Text(text.to_string())],
63        },
64        Op::EndTextSection,
65    ]
66}
67
68/// Like [`text`], in grey
69pub(super) fn grey_text(
70    face: &Face,
71    size: f32,
72    level: f32,
73    (x, y): (f32, f32),
74    align: Align,
75    content: &str,
76) -> Vec<Op> {
77    std::iter::once(Op::SetFillColor { col: grey(level) })
78        .chain(text(face, size, x, y, align, content))
79        .chain(std::iter::once(Op::SetFillColor { col: grey(BLACK) }))
80        .collect()
81}
82
83/// A horizontal rule from `x1` to `x2` at `y`
84pub(super) fn rule(x1: f32, x2: f32, y: f32) -> Op {
85    Op::DrawLine {
86        line: Line {
87            points: [x1, x2]
88                .into_iter()
89                .map(|x| LinePoint {
90                    p: point(x, y),
91                    bezier: false,
92                })
93                .collect(),
94            is_closed: false,
95        },
96    }
97}
98
99/// The outline of a box whose top-left corner is at (`x`, `y`)
100pub(super) fn frame(x: f32, y: f32, width: f32, height: f32) -> Op {
101    let lower_left = point(x, y + height);
102    Op::DrawRectangle {
103        rectangle: Rect {
104            x: lower_left.x,
105            y: lower_left.y,
106            width: Mm(width).into(),
107            height: Mm(height).into(),
108            mode: Some(PaintMode::Stroke),
109            winding_order: None,
110        },
111    }
112}
