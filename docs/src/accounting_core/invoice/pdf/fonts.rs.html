1//! Font faces for the PDF renderer, with the metrics to measure text
2
3use super::{FontFace, PdfError, PdfFont};
4use printpdf::{BuiltinFont, ParsedFont, PdfDocument, PdfFontHandle};
5
6/// Points per millimetre
7const PT_PER_MM: f32 = 72.0 / 25.4;
8/// Width, in ems, of a character with no glyph in the font. Rendering refuses such characters
9/// unless they are whitespace, and printpdf's built-in Helvetica has no space glyph, so this is
10/// the width of the standard Helvetica space.
11const MISSING_GLYPH_EM: f32 = 0.278;
12
13/// One face: the handle the page ops use and the metrics to measure text with
14#[derive(Debug, Clone)]
15pub(super) struct Face {
16    pub(super) handle: PdfFontHandle,
17    metrics: ParsedFont,
18}
19
20impl Face {
21    /// Whether the face can draw `c`
22    pub(super) fn has_glyph(&self, c: char) -> bool {
23        self.metrics.lookup_glyph_index(u32::from(c)).is_some()
24    }
25
26    /// Width of `text` set at `size` points, in millimetres
27    pub(super) fn width_mm(&self, text: &str, size: f32) -> f32 {
28        let units_per_em = f32::from(self.metrics.units_per_em.max(1));
29        let ems: f32 = text
30            .chars()
31            .map(|c| {
32                self.metrics
33                    .lookup_glyph_index(u32::from(c))
34                    .and_then(|glyph| self.metrics.get_glyph_width(glyph))
35                    .map_or(MISSING_GLYPH_EM, |width| f32::from(width) / units_per_em)
36            })
37            .sum();
38        ems * size / PT_PER_MM
39    }
40}
41
42/// The regular and bold faces an invoice is set in
43#[derive(Debug, Clone)]
44pub(super) struct Fonts {
45    pub(super) regular: Face,
46    pub(super) bold: Face,
47}
48
49impl Fonts {
50    /// The face a page op refers to by `handle`, if it is one of these
51    pub(super) fn face(&self, handle: &PdfFontHandle) -> Option<&Face> {
52        [&self.regular, &self.bold]
53            .into_iter()
54            .find(|face| &face.handle == handle)
55    }
56
57    /// Load the faces `font` asks for, registering custom ones with `document`
58    pub(super) fn load(document: &mut PdfDocument, font: &PdfFont) -> Result<Self, PdfError> {
59        match font {
60            PdfFont::Builtin => Ok(Self {
61                regular: builtin(BuiltinFont::Helvetica)?,
62                bold: builtin(BuiltinFont::HelveticaBold)?,
63            }),
64            PdfFont::Custom { regular, bold } => {
65                let regular_face = custom(document, regular, FontFace::Regular)?;
66                let bold_face = match bold {
67                    Some(bytes) => custom(document, bytes, FontFace::Bold)?,
68                    None => regular_face.clone(),
69                };
70                Ok(Self {
71                    regular: regular_face,
72                    bold: bold_face,
73                })
74            }
75        }
76    }
77}
78
79/// A standard PDF font, measured with the metrics printpdf ships for it
80fn builtin(font: BuiltinFont) -> Result<Face, PdfError> {
81    let metrics = font.get_parsed_font().ok_or(PdfError::BuiltinFontMetrics)?;
82    Ok(Face {
83        handle: PdfFontHandle::Builtin(font),
84        metrics,
85    })
86}
87
88/// A TrueType or OpenType font from `bytes`, embedded in `document`
89fn custom(document: &mut PdfDocument, bytes: &[u8], face: FontFace) -> Result<Face, PdfError> {
90    let metrics =
91        ParsedFont::from_bytes(bytes, 0, &mut Vec::new()).ok_or(PdfError::InvalidFont { face })?;
92    let id = document.add_font(&metrics);
93    Ok(Face {
94        handle: PdfFontHandle::External(id),
95        metrics,
96    })
