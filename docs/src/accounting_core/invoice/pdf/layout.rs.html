69/// How a cell too wide for its column is made to fit

74    /// Set in a smaller size, so a number is never printed cut short

203    /// How many item rows fit on each page, and the row slots the totals and footer need
204    pub(super) fn capacity(&self) -> RowCapacity {
205        let rows_below =
206            |top: f32| row_count(BODY_BOTTOM - top - ROW_HEIGHT, f32::floor, ROW_HEIGHT);
207        let (_, totals_bottom) = self.totals(BLOCK_GAP);
208        let (_, closing_bottom) = self.closing(totals_bottom);
209        RowCapacity {
210            first_page: rows_below(self.header().1),
211            other_pages: rows_below(self.continuation_header().1),
212            totals: row_count(closing_bottom, f32::ceil, ROW_HEIGHT),
213        }
214    }

216    /// Check the closing column (notes and terms) fits below the totals on a page of its own
217    ///
218    /// The totals, notes, terms and signature always go on one page, so a closing column taller
219    /// than the room left there would run over the page foot.
220    pub(super) fn check_terms_fit(&self) -> Result<(), PdfError> {
221        let top = self.continuation_header().1 + BLOCK_GAP;
222        let (_, totals_bottom) = self.totals(top);
223        let (_, bottom) = self.closing_column(totals_bottom);
224        if bottom <= BODY_BOTTOM + FIT_TOLERANCE {
225            return Ok(());
226        }
227        let lines = self.wrapped(WORDS_WIDTH, &self.print.notes).len()
228            + self.wrapped(WORDS_WIDTH, &self.options.terms).len();
229        let over = row_count(bottom - BODY_BOTTOM - FIT_TOLERANCE, f32::ceil, LINE_HEIGHT);
230        Err(PdfError::TermsTooLong {
231            lines,
232            max_lines: lines.saturating_sub(over),
233        })
234    }

236    /// Ops for page `number` of `count`, showing `rows`, with the totals on the last page
237    pub(super) fn page(&self, number: usize, count: usize, rows: &[PrintRow]) -> Vec<Op> {
238        let (head, top) = if number == 1 {
239            self.header()
240        } else {
241            self.continuation_header()
242        };
243        let (table, below_table) = if rows.is_empty() {
244            (Vec::new(), top)
245        } else {
246            self.table(top, rows)
247        };
248        let (closing, _) = if number == count {
249            let (totals, bottom) = self.totals(below_table + BLOCK_GAP);
250            let (footer, bottom) = self.closing(bottom);
251            ([totals, footer].concat(), bottom)
252        } else {
253            (Vec::new(), below_table)
254        };
255        [
256            page_setup(),
257            head,
258            table,
259            closing,
260            self.page_strip(number, count),
261        ]
262        .concat()
263    }

265    /// Logo if given, seller, invoice details, then the buyer, on the first page
266    fn header(&self) -> Block {
267        let (logo, logo_height) = self.logo.map_or((Vec::new(), 0.0), |logo| {
268            image(logo, (LEFT, MARGIN), (LOGO_WIDTH, LOGO_HEIGHT))
269        });
270        let (seller, seller_bottom) = self.seller(MARGIN + 5.0);
271        let (details, details_bottom) = self.details(MARGIN + 6.0);
272        let top_bottom = (MARGIN + logo_height)
273            .max(seller_bottom)
274            .max(details_bottom)
275            + BLOCK_GAP;
276        let (buyer, bottom) = self.buyer(top_bottom + 5.0);
277        let ops = [
278            logo,
279            seller,
280            details,
281            vec![
282                rule(LEFT, RIGHT, top_bottom),
283                rule(LEFT, RIGHT, bottom + BLOCK_GAP),
284            ],
285            buyer,
286        ]
287        .concat();
288        (ops, bottom + BLOCK_GAP + 2.0)
289    }

311    fn details(&self, y: f32) -> Block {
312        let print = self.print;
313        let supply = if print.is_inter_state {
314            "Inter-state (IGST)"
315        } else {
316            "Intra-state (CGST + SGST)"
317        };

329        .into_iter()
330        .flatten()
331        .collect();
332        let lines = self.wrapped(DETAILS_WIDTH, &lines);
333        let title = text(
334            &self.fonts.bold,

341        let detail_ops = lines.iter().enumerate().flat_map(|(index, line)| {
342            let baseline = y + 6.0 + LINE_HEIGHT * index as f32;

347                baseline,
348                Align::Right,
349                line,

351        });
352        let bottom = y + 6.0 + LINE_HEIGHT * (lines.len() - 1) as f32;
353        (title.into_iter().chain(detail_ops).collect(), bottom)
354    }

356    /// "Bill To" with the buyer's name, or [`WALK_IN_BUYER_LABEL`] when it has none, address,
357    /// state and GSTIN; beside it "Ship To" with the delivery address when there is one
358    fn buyer(&self, y: f32) -> Block {
359        let print = self.print;
360        let buyer = &print.buyer;
361        let shown_name = if buyer.has_name() {
362            buyer.name.as_str()
363        } else {
364            WALK_IN_BUYER_LABEL
365        };
366        let bill_lines = party_lines(buyer, Some(&print.buyer_state));
367        let Some(ship_to) = &print.ship_to else {
368            return self.party_block((LEFT, y), RIGHT - LEFT, "BILL TO", shown_name, &bill_lines);
369        };
370        let width = (RIGHT - LEFT - COLUMN_GAP) / 2.0;
371        let (bill, bill_bottom) =
372            self.party_block((LEFT, y), width, "BILL TO", shown_name, &bill_lines);
373        let (ship, ship_bottom) = self.party_block(
374            (LEFT + width + COLUMN_GAP, y),
375            width,
376            "SHIP TO",
377            &ship_to.name,
378            &ship_to_lines(ship_to),
379        );
380        ([bill, ship].concat(), bill_bottom.max(ship_bottom))
381    }

383    /// A party block `width` wide at `(x, y)`: a small grey `label`, the name in bold, then
384    /// `lines`
385    fn party_block(
386        &self,
387        (x, y): (f32, f32),
388        width: f32,
389        label: &str,
390        name: &str,
391        lines: &[String],
392    ) -> Block {
393        let bold = &self.fonts.bold;
394        let label = grey_text(bold, SMALL_SIZE, MUTED, (x, y), Align::Left, label);
395        let name = self.fit(bold, PARTY_SIZE, name, width);
396        let name_ops = text(bold, PARTY_SIZE, x, y + 5.0, Align::Left, &name);
397        let (lines, bottom) = self.lines(x, y + 5.0 + LINE_HEIGHT + 0.5, width, lines);
398        ([label, name_ops, lines].concat(), bottom)
399    }

431    /// Column headings, then one line per row, between rules
432    fn table(&self, top: f32, rows: &[PrintRow]) -> Block {
433        let columns = self.columns();
434        let with_discount = columns.len() == COLUMNS_WITH_DISCOUNT.len();
435        let headings: Vec<String> = columns
436            .iter()
437            .map(|column| column.heading.to_string())
438            .collect();
439        let heading = self.row(columns, &self.fonts.bold, top, &headings);
440        let body = rows.iter().enumerate().flat_map(|(index, row)| {
441            let row_top = top + ROW_HEIGHT * (index + 1) as f32;
442            self.row(
443                columns,
444                &self.fonts.regular,
445                row_top,
446                &cells(row, with_discount),
447            )
448        });
449        let bottom = top + ROW_HEIGHT * (rows.len() + 1) as f32;
450        let rules = [top, top + ROW_HEIGHT, bottom].map(|y| rule(LEFT, RIGHT, y));
451        (
452            heading.into_iter().chain(body).chain(rules).collect(),
453            bottom,
454        )
455    }

457    /// One table row of `cells` whose top edge is at `top`, each cut to its one of `columns`
458    fn row(&self, columns: &[Column], face: &Face, top: f32, cells: &[String]) -> Vec<Op> {
459        let lefts = columns.iter().scan(LEFT, |x, column| {
460            let left = *x;
461            *x += column.width;
462            Some(left)
463        });
464        columns
465            .iter()
466            .zip(lefts)
467            .zip(cells)
468            .flat_map(|((column, left), cell)| {
469                let inner = column.width - 2.0 * CELL_PADDING;
470                let (size, content) = match column.overflow {
471                    Overflow::Ellipsis => (BODY_SIZE, self.fit(face, BODY_SIZE, cell, inner)),
472                    Overflow::Shrink => (shrink_to_fit(face, BODY_SIZE, cell, inner), cell.clone()),
473                };
474                let x = match column.align {
475                    Align::Left => left + CELL_PADDING,
476                    Align::Center => left + column.width / 2.0,
477                    Align::Right => left + column.width - CELL_PADDING,
478                };
479                text(face, size, x, top + ROW_BASELINE, column.align, &content)
480            })
481            .collect()
482    }

484    /// Amount in words on the left; subtotal and discount when there is one, taxable value, tax
485    /// lines and total on the right
486    fn totals(&self, top: f32) -> Block {
487        let print = self.print;
488        let lines = summary_lines(print);
489        let amounts = lines
490            .iter()
491            .enumerate()
492            .flat_map(|(index, (label, amount))| {
493                let y = top + LINE_HEIGHT * (index + 1) as f32;
494                self.amount_line(&self.fonts.regular, BODY_SIZE, y, label, amount)
495            });
496        let rule_y = top + LINE_HEIGHT * lines.len() as f32 + 1.5;
497        let total_y = rule_y + LINE_HEIGHT + 1.0;
498        let total = self.amount_line(&self.fonts.bold, TOTAL_SIZE, total_y, "Total", &print.total);
499        let (words, words_bottom) = self.words(top);
500        let ops = amounts
501            .chain(std::iter::once(rule(TOTALS_LABEL_X, RIGHT, rule_y)))
502            .chain(total)
503            .chain(words)
504            .collect();
505        (ops, total_y.max(words_bottom) + BLOCK_GAP)
506    }

544    /// The left column below the totals, top to bottom: notes, payment details, then terms
545    fn closing_column(&self, top: f32) -> Block {
546        let (notes, bottom) = self.section(top, "Notes", &self.print.notes, WORDS_WIDTH);
547        let (payment, bottom) = self.payment(next_top(&notes, bottom));
548        let (terms, bottom) = self.section(
549            next_top(&payment, bottom),
550            "Terms & Conditions",
551            &self.options.terms,
552            WORDS_WIDTH,
553        );
554        ([notes, payment, terms].concat(), bottom)
555    }

557    /// "Payment details" with the bank account and UPI ID, and the UPI QR code beside them when
558    /// there is one; nothing when there are no payment details
559    fn payment(&self, top: f32) -> Block {
560        let Some(payment) = &self.print.payment else {
561            return (Vec::new(), top);
562        };
563        let lines = payment_lines(payment);
564        let Some(modules) = self.qr else {
565            return self.section(top, "Payment details", &lines, WORDS_WIDTH);
566        };
567        let (details, details_bottom) =
568            self.section(top, "Payment details", &lines, WORDS_WIDTH - QR_COLUMN);
569        let (code, code_bottom) = self.qr_code(modules, top);
570        ([details, code].concat(), details_bottom.max(code_bottom))
571    }

573    /// The QR code centred in the right of the payment block, with [`QR_CAPTION`] under it
574    fn qr_code(&self, modules: &[Vec<bool>], top: f32) -> Block {
575        let centre = LEFT + WORDS_WIDTH - QR_COLUMN / 2.0;
576        let code_top = top + 1.0;
577        let quiet_zone = QR_QUIET_MODULES * QR_SIZE / modules.len().max(1) as f32;
578        let caption_y = code_top + QR_SIZE + quiet_zone + LINE_HEIGHT / 2.0;
579        let caption = grey_text(
580            &self.fonts.regular,
581            SMALL_SIZE,
582            MUTED,
583            (centre, caption_y),
584            Align::Center,
585            QR_CAPTION,
586        );
587        let code = qr_ops(modules, (centre - QR_SIZE / 2.0, code_top), QR_SIZE);
588        ([code, caption].concat(), caption_y)
589    }

591    /// A bold `heading` and `lines` wrapped to `width` below it, or nothing when `lines` is
592    /// empty
593    fn section(&self, top: f32, heading: &str, lines: &[String], width: f32) -> Block {
594        if lines.is_empty() {
595            return (Vec::new(), top);
596        }
597        let heading = text(
598            &self.fonts.bold,
599            BODY_SIZE,
600            LEFT,
601            top + LINE_HEIGHT,
602            Align::Left,
603            heading,
604        );
605        let (lines, bottom) = self.lines(LEFT, top + 2.0 * LINE_HEIGHT, width, lines);
606        ([heading, lines].concat(), bottom)
607    }

614            &format!("For {}", self.print.seller.name),

621                BODY_SIZE,
622                RIGHT,

629                &self.fonts.regular,

660    /// Body-size lines from baseline `y` down, each wrapped to `width`
661    fn lines(&self, x: f32, y: f32, width: f32, lines: &[String]) -> Block {
662        let face = &self.fonts.regular;
663        let wrapped = self.wrapped(width, lines);
664        let ops = wrapped
665            .iter()
666            .enumerate()
667            .flat_map(|(index, line)| {
668                text(
669                    face,
670                    BODY_SIZE,
671                    x,
672                    y + LINE_HEIGHT * index as f32,
673                    Align::Left,
674                    line,
675                )
676            })
677            .collect();
678        (
679            ops,
680            y + LINE_HEIGHT * wrapped.len().saturating_sub(1) as f32,
681        )
682    }
683
684    /// Body-size `lines`, each wrapped to `width`
685    fn wrapped(&self, width: f32, lines: &[String]) -> Vec<String> {
686        let face = &self.fonts.regular;
687        lines
688            .iter()
689            .flat_map(|line| wrap(line, width, |s| face.width_mm(s, BODY_SIZE)))
690            .collect()
691    }
692
693    /// `text` cut to fit `width` in `face` at `size`
694    fn fit(&self, face: &Face, size: f32, text: &str, width: f32) -> String {
695        fit(text, width, |s| face.width_mm(s, size))
696    }

757/// The totals lines above the rule, as label and amount: the subtotal and the invoice discount
758/// when there are discounts, then the taxable value, each tax and the total tax
759fn summary_lines(print: &InvoicePrint) -> Vec<(&str, String)> {
760    let subtotal = print
761        .shows_subtotal()
762        .then(|| (SUBTOTAL_LABEL, print.subtotal.clone()));
763    let discount = print
764        .discount
765        .as_ref()
766        .zip(print.discount_label.as_ref())
767        .map(|(amount, label)| (label.as_str(), format!("-{amount}")));
768    subtotal
769        .into_iter()
770        .chain(discount)
771        .chain(std::iter::once((
772            "Taxable Value",
773            print.taxable_value.clone(),
774        )))
775        .chain(
776            print
777                .tax_lines
778                .iter()
779                .map(|line| (line.label.as_str(), line.amount.clone())),
780        )
781        .chain(std::iter::once(("Total Tax", print.total_tax.clone())))
782        .collect()
783}

806/// The size, at most `size`, at which `text` in `face` fits in `width` millimetres
807fn shrink_to_fit(face: &Face, size: f32, text: &str, width: f32) -> f32 {
808    let natural = face.width_mm(text, size);
809    if natural <= width {
810        size
811    } else {
812        // Text width is proportional to its size
813        size * width / natural
814    }
815}

817/// Rows of `pitch` millimetres in `height` millimetres, rounded by `round`
818fn row_count(height: f32, round: fn(f32) -> f32, pitch: f32) -> usize {
819    // Non-negative and bounded by the page height over the pitch, so the cast is exact
820    round(height / pitch).max(0.0) as usize
