1//! PDF rendering of GST invoices (needs the `pdf` feature)
2//!
3//! [`render_pdf`] lays an [`InvoicePrint`] out on A4 pages and returns the PDF bytes;
4//! [`GstInvoice::to_pdf`] builds the print model and renders it in one call. The page shows a
5//! logo placeholder, the seller and invoice details, the buyer, the item table, the tax summary
6//! with the total in words, any terms, a signature line and a footer note with page numbers. Item
7//! tables longer than a page continue on the next one, and descriptions too long for their
8//! column are cut short with an ellipsis.
9//!
10//! By default the invoice is set in the standard Helvetica font and amounts are labelled `Rs.`.
11//! Helvetica only covers Western European characters, so an invoice with the `₹` sign or with
12//! names in an Indian script needs a [`PdfFont::Custom`] TrueType font that has those glyphs.
13//! Rendering refuses any character its font can't draw with [`PdfError::MissingGlyph`], whether it
14//! comes from the invoice or is one of the renderer's own labels. The terms print on the same page
15//! as the totals, so terms longer than that page holds are refused with
16//! [`PdfError::TermsTooLong`].
17//!
18//! # Example
19//!
20//! ```
21//! use accounting_core::invoice::{
22//!     GstInvoice, GstLineItem, Gstin, InvoiceParties, InvoiceParty, PdfOptions,
23//! };
24//! use bigdecimal::BigDecimal;
25//! use chrono::NaiveDate;
26//!
27//! # fn main() -> Result<(), accounting_core::Error> {
28//! let seller = Gstin::parse("27AAPFU0939F1ZV")?;
29//! let buyer = Gstin::parse("27AAPFU0939F2ZU")?;
30//! let item = GstLineItem::new(
31//!     "998314",
32//!     "IT consulting",
33//!     BigDecimal::from(10),
34//!     BigDecimal::from(1500),
35//!     BigDecimal::from(18),
36//! )?;
37//! let invoice = GstInvoice::new(
38//!     "INV/2024-25/001",
39//!     NaiveDate::from_ymd_opt(2024, 11, 15).unwrap(),
40//!     seller.clone(),
41//!     buyer.clone(),
42//!     vec![item],
43//! )?;
44//! let parties = InvoiceParties::new(
45//!     InvoiceParty::new("Acme Services", vec!["Mumbai".into()], seller),
46//!     InvoiceParty::new("Globex Ltd", vec!["Pune".into()], buyer),
47//! );
48//!
49//! let pdf = invoice.to_pdf(&parties, &PdfOptions::default())?;
50//! assert!(pdf.starts_with(b"%PDF-"));
51//! # Ok(())
52//! # }
53//! ```

68/// Currency label printed before amounts when the built-in font is used
69pub const DEFAULT_CURRENCY_LABEL: &str = "Rs.";
70/// Note printed at the foot of every page by default
71pub const DEFAULT_FOOTER_NOTE: &str = "E-Invoice Ready";

73/// The font an invoice is set in
74#[derive(Debug, Clone, Default, PartialEq, Eq)]
75pub enum PdfFont {
76    /// Standard Helvetica and Helvetica Bold, not embedded; Western European characters only
77    #[default]
78    Builtin,
79    /// TrueType or OpenType font files, embedded in the PDF
80    Custom {
81        /// Regular face
82        regular: Vec<u8>,
83        /// Bold face; the regular face is used when absent
84        bold: Option<Vec<u8>>,
85    },
86}
87
88/// How an invoice is rendered
89#[derive(Debug, Clone, PartialEq, Eq)]
90pub struct PdfOptions {
91    /// Font to set the invoice in
92    pub font: PdfFont,
93    /// Printed before each total, e.g. `Rs.` or `₹`; empty for none
94    pub currency_label: String,
95    /// Terms and conditions, one paragraph each, printed above the page foot on the last page
96    pub terms: Vec<String>,
97    /// Note printed at the foot of every page
98    pub footer_note: String,
99}
100
101impl Default for PdfOptions {
102    fn default() -> Self {
103        Self {
104            font: PdfFont::Builtin,
105            currency_label: DEFAULT_CURRENCY_LABEL.to_string(),
106            terms: Vec::new(),
107            footer_note: DEFAULT_FOOTER_NOTE.to_string(),
108        }
109    }
110}

112/// Which face of a [`PdfFont::Custom`] font
113#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
114pub enum FontFace {
115    /// The regular face
116    Regular,
117    /// The bold face
118    Bold,
119}
120
121impl fmt::Display for FontFace {
122    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
123        f.write_str(match self {
124            Self::Regular => "regular",
125            Self::Bold => "bold",
126        })
127    }
128}

130/// Why an invoice could not be rendered to PDF
131#[derive(Debug, thiserror::Error)]
132pub enum PdfError {
133    /// The invoice or its parties are invalid
134    #[error(transparent)]
135    Invoice(#[from] InvoiceError),
136    /// The bytes given for a custom font are not a TrueType or OpenType font
137    #[error("the {face} font is not a valid TrueType or OpenType font")]
138    InvalidFont {
139        /// Which face was rejected
140        face: FontFace,
141    },
142    /// The metrics printpdf ships for the built-in fonts could not be loaded
143    #[error("could not load the metrics of the built-in font")]
144    BuiltinFontMetrics,
145    /// The text contains a character the font has no glyph for
146    #[error("the font has no glyph for {ch:?}; use a font that covers it")]
147    MissingGlyph {
148        /// The first character that can't be drawn
149        ch: char,
150    },
151    /// The terms, once wrapped, don't fit below the totals on one page
152    #[error("the terms take {lines} lines but at most {max_lines} fit below the totals")]
153    TermsTooLong {
154        /// Lines the terms wrap to
155        lines: usize,
156        /// Lines that fit
157        max_lines: usize,
158    },
159}

161/// Render the printable view of an invoice as A4 PDF bytes
162///
163/// # Errors
164///
165/// [`PdfError::InvalidFont`] if custom font bytes don't parse, [`PdfError::MissingGlyph`] if the
166/// font can't draw a character on the page, [`PdfError::TermsTooLong`] if the terms don't fit on
167/// the last page, and
168/// [`PdfError::BuiltinFontMetrics`] if the built-in font metrics are unavailable.
169pub fn render_pdf(print: &InvoicePrint, options: &PdfOptions) -> Result<Vec<u8>, PdfError> {
170    let mut document = PdfDocument::new(&format!("{} {}", print.title, print.invoice_number));
171    let fonts = Fonts::load(&mut document, &options.font)?;
172    let layout = Layout::new(&fonts, print, options);
173    layout.check_terms_fit()?;
174
175    let pages = paginate_rows(&print.rows, layout.capacity());
176    let count = pages.len();
177    let page_ops: Vec<Vec<Op>> = pages
178        .iter()
179        .enumerate()
180        .map(|(index, rows)| layout.page(index + 1, count, rows))
181        .collect();
182    check_glyphs(&fonts, &page_ops)?;
183    let pages = page_ops
184        .into_iter()
185        .map(|ops| PdfPage::new(Mm(PAGE_WIDTH), Mm(PAGE_HEIGHT), ops))
186        .collect();
187
188    Ok(document
189        .with_pages(pages)
190        .save(&PdfSaveOptions::default(), &mut Vec::new()))
191}

193impl GstInvoice {
194    /// Render this invoice, issued by and to `parties`, as A4 PDF bytes
195    ///
196    /// # Errors
197    ///
198    /// [`PdfError::Invoice`] from [`InvoicePrint::from_invoice`], or any error from
199    /// [`render_pdf`].
200    pub fn to_pdf(
201        &self,
202        parties: &InvoiceParties,
203        options: &PdfOptions,
204    ) -> Result<Vec<u8>, PdfError> {
205        render_pdf(&InvoicePrint::from_invoice(self, parties)?, options)
206    }
207}

209/// Refuse text the fonts can't draw, rather than print blank boxes
210///
211/// Walks the page ops as drawn, so labels, numbers and caller text are all checked against the
212/// face they are set in.
213fn check_glyphs(fonts: &Fonts, pages: &[Vec<Op>]) -> Result<(), PdfError> {
214    let missing = pages
215        .iter()
216        .flatten()
217        .scan(None::<&Face>, |face, op| {
218            match op {
219                Op::SetFont { font, .. } => *face = fonts.face(font),
220                Op::ShowText { items } => {
221                    let unknown = face.and_then(|face| first_missing(face, items));
222                    return Some(unknown);
223                }
224                _ => {}
225            }
226            Some(None)
227        })
228        .flatten()
229        .next();
230    missing.map_or(Ok(()), |ch| Err(PdfError::MissingGlyph { ch }))
231}
232
233/// The first non-whitespace character in `items` that `face` has no glyph for
234fn first_missing(face: &Face, items: &[TextItem]) -> Option<char> {
235    items
236        .iter()
237        .filter_map(|item| match item {
238            TextItem::Text(text) => Some(text),
239            _ => None,
240        })
241        .flat_map(|text| text.chars())
242        .find(|c| !c.is_whitespace() && !face.has_glyph(*c))
243}
