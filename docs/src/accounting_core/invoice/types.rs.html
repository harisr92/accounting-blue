11/// Characters a GSTIN is built from, in the order the checksum algorithm assigns them values
12const GSTIN_CHARSET: &[u8; 36] = b"0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ";
13
14/// Length of a GSTIN
15const GSTIN_LEN: usize = 15;
16
17/// Maximum length of an invoice number (Rule 46 of the CGST Rules)
18const INVOICE_NUMBER_MAX_LEN: usize = 16;
19
20/// A validated GST Identification Number
21///
22/// A GSTIN is 15 characters, laid out as:
23///
24/// | Position | Meaning                                          |
25/// |----------|--------------------------------------------------|
26/// | 1-2      | State code (`01`-`38`, `97` or `99`)             |
27/// | 3-12     | PAN of the taxpayer (`AAAAA9999A`)               |
28/// | 13       | Entity number for the same PAN (`1`-`9`, `A`-`Z`) |
29/// | 14       | `Z` by default                                   |
30/// | 15       | Mod-36 checksum over the first 14 characters     |
31///
32/// Lowercase input is accepted and normalised to uppercase. Deserialising goes through the same
33/// validation, so a `Gstin` value is always well-formed.
34#[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize, Deserialize)]
35#[serde(try_from = "String", into = "String")]
36pub struct Gstin(String);
37
38impl Gstin {
39    /// Parse and validate a GSTIN
40    pub fn parse(value: &str) -> Result<Self, InvoiceError> {
41        let gstin = value.to_ascii_uppercase();
42        let bytes = gstin.as_bytes();
43
44        if bytes.len() != GSTIN_LEN {
45            return Err(InvoiceError::InvalidGstin(format!(
46                "{value}: must be exactly {GSTIN_LEN} characters"
47            )));
48        }
49
50        let state_code = &gstin[0..2];
51        if !is_valid_state_code(state_code) {
52            return Err(InvoiceError::InvalidGstin(format!(
53                "{value}: invalid state code '{state_code}'"
54            )));
55        }
56
57        let pan = &bytes[2..12];
58        let pan_is_valid = pan[0..5].iter().all(u8::is_ascii_uppercase)
59            && pan[5..9].iter().all(u8::is_ascii_digit)
60            && pan[9].is_ascii_uppercase();
61        if !pan_is_valid {
62            return Err(InvoiceError::InvalidGstin(format!(
63                "{value}: characters 3-12 must be a PAN (AAAAA9999A)"
64            )));
65        }
66
67        let entity = bytes[12];
68        if !(matches!(entity, b'1'..=b'9') || entity.is_ascii_uppercase()) {
69            return Err(InvoiceError::InvalidGstin(format!(
70                "{value}: entity number must be 1-9 or A-Z"
71            )));
72        }
73
74        if bytes[13] != b'Z' {
75            return Err(InvoiceError::InvalidGstin(format!(
76                "{value}: character 14 must be 'Z'"
77            )));
78        }
79
80        let expected = gstin_checksum(&bytes[..14]);
81        if bytes[14] != expected {
82            return Err(InvoiceError::InvalidGstin(format!(
83                "{value}: checksum mismatch, expected '{}'",
84                expected as char
85            )));
86        }
87
88        Ok(Self(gstin))
89    }
90
91    /// The GSTIN as a string
92    pub fn as_str(&self) -> &str {
93        &self.0
94    }
95
96    /// Two-digit state code the taxpayer is registered in
97    pub fn state_code(&self) -> &str {
98        &self.0[0..2]
99    }
100
101    /// PAN embedded in the GSTIN
102    pub fn pan(&self) -> &str {
103        &self.0[2..12]
104    }
105}
106
107/// State and union territory codes, plus `97` (other territory) and `99` (centre jurisdiction)
108fn is_valid_state_code(code: &str) -> bool {
109    matches!(code.parse::<u8>(), Ok(1..=38 | 97 | 99))
110}

112/// Compute the GSTIN check character for the first 14 characters
113///
114/// Callers must pass characters from [`GSTIN_CHARSET`] only.
115fn gstin_checksum(body: &[u8]) -> u8 {
116    let sum: u32 = body
117        .iter()
118        .enumerate()
119        .map(|(i, c)| {
120            let value = GSTIN_CHARSET.iter().position(|x| x == c).unwrap_or(0) as u32;
121            let product = value * if i % 2 == 0 { 1 } else { 2 };
122            product / 36 + product % 36
123        })
124        .sum();
125
126    GSTIN_CHARSET[((36 - sum % 36) % 36) as usize]
127}

211/// A line on a GST invoice
212#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
213pub struct GstLineItem {
214    /// HSN code (goods) or SAC code (services): 4, 6 or 8 digits
215    pub hsn_sac: String,
216    /// Item description
217    pub description: String,
218    /// Quantity supplied
219    pub quantity: BigDecimal,
220    /// Unit price before GST
221    pub unit_price: BigDecimal,
222    /// Total GST rate as a percentage (e.g. 18 for 18%)
223    pub gst_rate: BigDecimal,
224}
225
226impl GstLineItem {
227    /// Create a validated line item
228    pub fn new(
229        hsn_sac: String,
230        description: String,
231        quantity: BigDecimal,
232        unit_price: BigDecimal,
233        gst_rate: BigDecimal,
234    ) -> Result<Self, InvoiceError> {
235        if !is_valid_hsn_sac(&hsn_sac) {
236            return Err(InvoiceError::InvalidHsnSac(hsn_sac));
237        }
238
239        if description.trim().is_empty() {
240            return Err(InvoiceError::InvalidLineItem(
241                "description cannot be empty".to_string(),
242            ));
243        }
244
245        if quantity <= *crate::ZERO {
246            return Err(InvoiceError::InvalidLineItem(
247                "quantity must be positive".to_string(),
248            ));
249        }
250
251        if unit_price < *crate::ZERO {
252            return Err(InvoiceError::InvalidLineItem(
253                "unit price cannot be negative".to_string(),
254            ));
255        }
256
257        if gst_rate < *crate::ZERO || gst_rate > 100 {
258            return Err(InvoiceError::InvalidLineItem(format!(
259                "GST rate must be between 0 and 100, got {gst_rate}"
260            )));
261        }
262
263        Ok(Self {
264            hsn_sac,
265            description,
266            quantity,
267            unit_price,
268            gst_rate,
269        })
270    }
271
272    /// Create a validated line item at the default GST rate for its HSN/SAC code
273    ///
274    /// The rate comes from [`HsnMaster::global`], falling back from the exact code to its 6- and
275    /// 4-digit headings. Use [`GstLineItem::new`] to charge a different rate.
276    pub fn with_default_rate(
277        hsn_sac: String,
278        description: String,
279        quantity: BigDecimal,
280        unit_price: BigDecimal,
281    ) -> Result<Self, InvoiceError> {
282        if !is_valid_hsn_sac(&hsn_sac) {
283            return Err(InvoiceError::InvalidHsnSac(hsn_sac));
284        }
285
286        let gst_rate = HsnMaster::global()
287            .default_rate(&hsn_sac)
288            .ok_or_else(|| InvoiceError::UnknownHsnSac(hsn_sac.clone()))?;
289
290        Self::new(hsn_sac, description, quantity, unit_price, gst_rate)
291    }
292
293    /// Taxable value of the line (quantity x unit price)
294    pub fn taxable_value(&self) -> BigDecimal {
295        &self.quantity * &self.unit_price
296    }
297
298    /// Tax breakdown for this line, split as IGST for inter-state supply or CGST + SGST otherwise
299    pub fn breakdown(&self, is_inter_state: bool) -> Result<GstBreakdown, InvoiceError> {
300        let rate = if is_inter_state {
301            GstRate::inter_state(self.gst_rate.clone())
302        } else {
303            GstRate::intra_state(self.gst_rate.clone())
304        };
305
306        Ok(GstCalculation::calculate(self.taxable_value(), rate)?.into())
307    }
308}

310/// A B2B tax invoice under Indian GST
311///
312/// Whether the supply is inter-state (IGST) or intra-state (CGST + SGST) is derived from the
313/// state codes of the seller and buyer GSTINs.
314#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
315pub struct GstInvoice {
316    /// Invoice number: at most 16 characters of letters, digits, `-` and `/`
317    pub invoice_number: String,
318    /// Date of issue
319    pub invoice_date: NaiveDate,
320    /// Supplier's GSTIN
321    pub seller_gstin: Gstin,
322    /// Recipient's GSTIN
323    pub buyer_gstin: Gstin,
324    /// Invoice lines
325    pub line_items: Vec<GstLineItem>,
326}
327
328impl GstInvoice {
329    /// Create a validated invoice
330    pub fn new(
331        invoice_number: String,
332        invoice_date: NaiveDate,
333        seller_gstin: Gstin,
334        buyer_gstin: Gstin,
335        line_items: Vec<GstLineItem>,
336    ) -> Result<Self, InvoiceError> {
337        validate_invoice_number(&invoice_number)?;
338
339        if line_items.is_empty() {
340            return Err(InvoiceError::InvalidLineItem(
341                "an invoice needs at least one line item".to_string(),
342            ));
343        }
344
345        Ok(Self {
346            invoice_number,
347            invoice_date,
348            seller_gstin,
349            buyer_gstin,
350            line_items,
351        })
352    }
353
354    /// Whether seller and buyer are registered in different states
355    pub fn is_inter_state(&self) -> bool {
356        self.seller_gstin.state_code() != self.buyer_gstin.state_code()
357    }
358
359    /// Tax breakdown for each line, in order
360    pub fn line_breakdowns(&self) -> Result<Vec<GstBreakdown>, InvoiceError> {
361        let is_inter_state = self.is_inter_state();
362        self.line_items
363            .iter()
364            .map(|item| item.breakdown(is_inter_state))
365            .collect()
366    }
367
368    /// Tax breakdown summed across all lines
369    pub fn breakdown(&self) -> Result<GstBreakdown, InvoiceError> {
370        let mut total = GstBreakdown::zero();
371        for line in self.line_breakdowns()? {
372            total += &line;
373        }
374        Ok(total)
375    }
376}

378/// Check an invoice number against Rule 46: unique per financial year (not checked here), at
379/// most 16 characters, and only letters, digits, `-` and `/`
380fn validate_invoice_number(invoice_number: &str) -> Result<(), InvoiceError> {
381    if invoice_number.is_empty() || invoice_number.len() > INVOICE_NUMBER_MAX_LEN {
382        return Err(InvoiceError::InvalidInvoiceNumber(format!(
383            "{invoice_number}: must be 1-{INVOICE_NUMBER_MAX_LEN} characters"
384        )));
385    }
386
387    if !invoice_number
388        .chars()
389        .all(|c| c.is_ascii_alphanumeric() || c == '-' || c == '/')
390    {
391        return Err(InvoiceError::InvalidInvoiceNumber(format!(
392            "{invoice_number}: only letters, digits, '-' and '/' are allowed"
393        )));
394    }
395
396    Ok(())
397}

399/// Invoice-related errors
400#[derive(Debug, thiserror::Error)]
401pub enum InvoiceError {
402    #[error("Invalid GSTIN: {0}")]
403    InvalidGstin(String),
404    #[error("Invalid invoice number: {0}")]
405    InvalidInvoiceNumber(String),
406    #[error("Invalid HSN/SAC code: {0}")]
407    InvalidHsnSac(String),
408    #[error("HSN/SAC code not in the master data: {0}")]
409    UnknownHsnSac(String),
410    #[error("Invalid line item: {0}")]
411    InvalidLineItem(String),
412    #[error(transparent)]
413    Gst(#[from] GstError),
414}
