11/// Characters a GSTIN is built from, in the order the checksum algorithm assigns them values
12const GSTIN_CHARSET: &[u8; 36] = b"0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ";
13
14/// Length of a GSTIN
15const GSTIN_LEN: usize = 15;
16
17/// Maximum length of an invoice number (Rule 46 of the CGST Rules)
18const INVOICE_NUMBER_MAX_LEN: usize = 16;
19
20/// Largest GST rate a line item may carry, as a percentage
21const MAX_GST_RATE: u32 = 100;

23/// A validated GST Identification Number
24///
25/// A GSTIN is 15 characters, laid out as:
26///
27/// | Position | Meaning                                          |
28/// |----------|--------------------------------------------------|
29/// | 1-2      | State code (`01`-`38`, `97` or `99`)             |
30/// | 3-12     | PAN of the taxpayer (`AAAAA9999A`)               |
31/// | 13       | Entity number for the same PAN (`1`-`9`, `A`-`Z`) |
32/// | 14       | `Z` by default                                   |
33/// | 15       | Mod-36 checksum over the first 14 characters     |
34///
35/// Lowercase input is accepted and normalised to uppercase. Deserialising goes through the same
36/// validation, so a `Gstin` value is always well-formed.
37#[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize, Deserialize)]
38#[serde(try_from = "String", into = "String")]
39pub struct Gstin(String);
40
41impl Gstin {
42    /// Parse and validate a GSTIN
43    ///
44    /// # Errors
45    ///
46    /// [`InvoiceError::InvalidGstin`] naming the first rule the value breaks.
47    pub fn parse(value: &str) -> Result<Self, InvoiceError> {
48        let gstin = value.to_ascii_uppercase();
49        check_gstin(gstin.as_bytes()).map_err(|reason| InvoiceError::InvalidGstin {
50            value: value.to_string(),
51            reason,
52        })?;
53        Ok(Self(gstin))
54    }
55
56    /// The GSTIN as a string
57    #[must_use]
58    pub fn as_str(&self) -> &str {
59        &self.0
60    }
61
62    /// Two-digit state code the taxpayer is registered in
63    #[must_use]
64    pub fn state_code(&self) -> &str {
65        &self.0[0..2]
66    }
67
68    /// PAN embedded in the GSTIN
69    #[must_use]
70    pub fn pan(&self) -> &str {
71        &self.0[2..12]
72    }
73}

75/// Why a GSTIN was rejected
76#[derive(Debug, Clone, Copy, PartialEq, Eq, thiserror::Error)]
77pub enum GstinError {
78    /// Not exactly 15 characters
79    #[error("must be exactly {GSTIN_LEN} characters")]
80    Length,
81    /// The first two characters are not a known state code
82    #[error("invalid state code")]
83    StateCode,
84    /// Characters 3-12 are not shaped like a PAN
85    #[error("characters 3-12 must be a PAN (AAAAA9999A)")]
86    Pan,
87    /// Character 13 is not 1-9 or A-Z
88    #[error("entity number must be 1-9 or A-Z")]
89    EntityNumber,
90    /// Character 14 is not `Z`
91    #[error("character 14 must be 'Z'")]
92    DefaultZ,
93    /// The check character does not match
94    #[error("checksum mismatch, expected '{expected}'")]
95    Checksum {
96        /// The check character the first 14 characters call for
97        expected: char,
98    },
99}
100
101/// Run every GSTIN rule over an uppercased value
102fn check_gstin(bytes: &[u8]) -> Result<(), GstinError> {
103    let (Some(body), Some(&check), GSTIN_LEN) = (bytes.get(..14), bytes.get(14), bytes.len())
104    else {
105        return Err(GstinError::Length);
106    };
107
108    check_rule(is_valid_state_code(&body[0..2]), GstinError::StateCode)?;
109    check_rule(is_pan(&body[2..12]), GstinError::Pan)?;
110    check_rule(
111        matches!(body[12], b'1'..=b'9') || body[12].is_ascii_uppercase(),
112        GstinError::EntityNumber,
113    )?;
114    check_rule(body[13] == b'Z', GstinError::DefaultZ)?;
115    match gstin_checksum(body) {
116        Some(expected) if expected == check => Ok(()),
117        Some(expected) => Err(GstinError::Checksum {
118            expected: char::from(expected),
119        }),
120        // Unreachable after the checks above, which admit only charset characters
121        None => Err(GstinError::Pan),
122    }
123}
124
125fn check_rule(holds: bool, error: GstinError) -> Result<(), GstinError> {
126    if holds {
127        Ok(())
128    } else {
129        Err(error)
130    }
131}

133/// State and union territory codes, plus `97` (other territory) and `99` (centre jurisdiction)
134fn is_valid_state_code(code: &[u8]) -> bool {
135    code.iter().all(u8::is_ascii_digit)
136        && matches!(
137            code.iter().fold(0u32, |n, d| n * 10 + u32::from(d - b'0')),
138            1..=38 | 97 | 99
139        )
140}
141
142/// Whether ten characters have the shape of a PAN: five letters, four digits, one letter
143fn is_pan(pan: &[u8]) -> bool {
144    pan.len() == 10
145        && pan[0..5].iter().all(u8::is_ascii_uppercase)
146        && pan[5..9].iter().all(u8::is_ascii_digit)
147        && pan[9].is_ascii_uppercase()
148}

150/// Compute the GSTIN check character for the first 14 characters
151///
152/// Returns `None` if any character is outside [`GSTIN_CHARSET`].
153pub(super) fn gstin_checksum(body: &[u8]) -> Option<u8> {
154    let radix = GSTIN_CHARSET.len();
155    let sum = body
156        .iter()
157        .enumerate()
158        .map(|(i, c)| {
159            let value = GSTIN_CHARSET.iter().position(|x| x == c)?;
160            let product = value * if i % 2 == 0 { 1 } else { 2 };
161            Some(product / radix + product % radix)
162        })
163        .sum::<Option<usize>>()?;
164
165    GSTIN_CHARSET.get((radix - sum % radix) % radix).copied()
166}

288impl GstLineItem {
289    /// Create a validated line item
290    ///
291    /// # Errors
292    ///
293    /// [`InvoiceError::InvalidHsnSac`] for a malformed code, or [`InvoiceError::InvalidLineItem`]
294    /// naming the first rule the other fields break.
295    pub fn new(
296        hsn_sac: impl Into<String>,
297        description: impl Into<String>,
298        quantity: BigDecimal,
299        unit_price: BigDecimal,
300        gst_rate: BigDecimal,
301    ) -> Result<Self, InvoiceError> {
302        let hsn_sac = hsn_sac.into();
303        if !is_valid_hsn_sac(&hsn_sac) {
304            return Err(InvoiceError::InvalidHsnSac(hsn_sac));
305        }
306
307        let item = Self {
308            hsn_sac,
309            description: description.into(),
310            quantity,
311            unit_price,
312            gst_rate,
313        };
314        item.check().map_err(InvoiceError::InvalidLineItem)?;
315        Ok(item)
316    }
317
318    /// The field rules other than the HSN/SAC shape
319    pub(super) fn check(&self) -> Result<(), LineItemError> {
320        if self.description.trim().is_empty() {
321            Err(LineItemError::EmptyDescription)
322        } else if !self.quantity.is_positive() {
323            Err(LineItemError::NonPositiveQuantity)
324        } else if self.unit_price.is_negative() {
325            Err(LineItemError::NegativeUnitPrice)
326        } else if self.gst_rate.is_negative() || self.gst_rate > MAX_GST_RATE {
327            Err(LineItemError::RateOutOfRange(self.gst_rate.clone()))
328        } else {
329            Ok(())
330        }
331    }
332
333    /// Create a validated line item at the default GST rate for its HSN/SAC code
334    ///
335    /// The rate comes from [`HsnMaster::global`], falling back from the exact code to its 6- and
336    /// 4-digit headings. Use [`GstLineItem::new`] to charge a different rate.
337    ///
338    /// # Errors
339    ///
340    /// [`InvoiceError::InvalidHsnSac`] for a malformed code, [`InvoiceError::UnknownHsnSac`] for a
341    /// code the master does not know, or any error from [`GstLineItem::new`].
342    pub fn with_default_rate(
343        hsn_sac: impl Into<String>,
344        description: impl Into<String>,
345        quantity: BigDecimal,
346        unit_price: BigDecimal,
347    ) -> Result<Self, InvoiceError> {
348        let hsn_sac = hsn_sac.into();
349        let Some(gst_rate) = HsnMaster::global().default_rate(&hsn_sac) else {
350            return Err(if is_valid_hsn_sac(&hsn_sac) {
351                InvoiceError::UnknownHsnSac(hsn_sac)
352            } else {
353                InvoiceError::InvalidHsnSac(hsn_sac)
354            });
355        };
356
357        Self::new(hsn_sac, description, quantity, unit_price, gst_rate)
358    }
359
360    /// Taxable value of the line (quantity x unit price), exactly
361    ///
362    /// This is not rounded; [`GstLineItem::breakdown`] rounds it to paise, so its
363    /// `taxable_value` is the amount that is invoiced and posted.
364    #[must_use]
365    pub fn taxable_value(&self) -> BigDecimal {
366        &self.quantity * &self.unit_price
367    }
368
369    /// Tax breakdown for this line, split as IGST for inter-state supply or CGST + SGST otherwise
370    ///
371    /// # Errors
372    ///
373    /// [`InvoiceError::Gst`] if the rate fails [`GstRate::validate`].
374    pub fn breakdown(&self, is_inter_state: bool) -> Result<GstBreakdown, InvoiceError> {
375        let rate = GstRate::for_supply(self.gst_rate.clone(), is_inter_state);
376        Ok(GstCalculation::calculate(self.taxable_value(), rate)?.into())
377    }
378}

380/// A B2B tax invoice under Indian GST
381///
382/// Whether the supply is inter-state (IGST) or intra-state (CGST + SGST) is derived from the
383/// state codes of the seller and buyer GSTINs. Deserialising goes through [`GstInvoice::new`],
384/// so the same rules apply.
385#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
386#[serde(try_from = "RawGstInvoice")]
387pub struct GstInvoice {
388    /// Invoice number: at most 16 characters of letters, digits, `-` and `/`
389    pub invoice_number: String,
390    /// Date of issue
391    pub invoice_date: NaiveDate,
392    /// Supplier's GSTIN
393    pub seller_gstin: Gstin,
394    /// Recipient's GSTIN
395    pub buyer_gstin: Gstin,
396    /// Invoice lines
397    pub line_items: Vec<GstLineItem>,
398}

400/// Unvalidated shape of a [`GstInvoice`], used when deserialising
401#[derive(Deserialize)]
402struct RawGstInvoice {
403    invoice_number: String,
404    invoice_date: NaiveDate,
405    seller_gstin: Gstin,
406    buyer_gstin: Gstin,
407    line_items: Vec<GstLineItem>,
408}

410impl TryFrom<RawGstInvoice> for GstInvoice {
411    type Error = InvoiceError;

413    fn try_from(raw: RawGstInvoice) -> Result<Self, Self::Error> {
414        Self::new(
415            raw.invoice_number,
416            raw.invoice_date,
417            raw.seller_gstin,
418            raw.buyer_gstin,
419            raw.line_items,

421    }

424impl GstInvoice {
425    /// Create a validated invoice
426    ///
427    /// # Errors
428    ///
429    /// [`InvoiceError::InvalidInvoiceNumber`] or [`InvoiceError::EmptyInvoice`].
430    pub fn new(
431        invoice_number: impl Into<String>,
432        invoice_date: NaiveDate,
433        seller_gstin: Gstin,
434        buyer_gstin: Gstin,
435        line_items: Vec<GstLineItem>,
436    ) -> Result<Self, InvoiceError> {
437        let invoice_number = invoice_number.into();
438        validate_invoice_number(&invoice_number)?;
439
440        if line_items.is_empty() {
441            return Err(InvoiceError::EmptyInvoice);
442        }
443
444        Ok(Self {
445            invoice_number,
446            invoice_date,
447            seller_gstin,
448            buyer_gstin,
449            line_items,
450        })
451    }
452
453    /// Whether seller and buyer are registered in different states
454    #[must_use]
455    pub fn is_inter_state(&self) -> bool {
456        self.seller_gstin.state_code() != self.buyer_gstin.state_code()
457    }
458
459    /// Tax breakdown for each line, in order
460    ///
461    /// # Errors
462    ///
463    /// The first error from [`GstLineItem::breakdown`].
464    pub fn line_breakdowns(&self) -> Result<Vec<GstBreakdown>, InvoiceError> {
465        let is_inter_state = self.is_inter_state();
466        self.line_items
467            .iter()
468            .map(|item| item.breakdown(is_inter_state))
469            .collect()
470    }
471
472    /// Tax breakdown summed across all lines
473    ///
474    /// # Errors
475    ///
476    /// The first error from [`GstLineItem::breakdown`].
477    pub fn breakdown(&self) -> Result<GstBreakdown, InvoiceError> {
478        Ok(self.line_breakdowns()?.iter().sum())
479    }
480}

482/// Check an invoice number against Rule 46: unique per financial year (not checked here), at
483/// most 16 characters, and only letters, digits, `-` and `/`
484pub(super) fn validate_invoice_number(invoice_number: &str) -> Result<(), InvoiceError> {
485    invoice_number_error(invoice_number).map_or(Ok(()), |reason| {
486        Err(InvoiceError::InvalidInvoiceNumber {
487            value: invoice_number.to_string(),
488            reason,
489        })
490    })
491}
492
493/// The first Rule 46 check an invoice number fails, if any
494pub(super) fn invoice_number_error(invoice_number: &str) -> Option<InvoiceNumberError> {
495    if invoice_number.is_empty() || invoice_number.len() > INVOICE_NUMBER_MAX_LEN {
496        Some(InvoiceNumberError::Length)
497    } else if !invoice_number
498        .chars()
499        .all(|c| c.is_ascii_alphanumeric() || c == '-' || c == '/')
500    {
501        Some(InvoiceNumberError::Characters)
502    } else {
503        None
504    }
505}

507/// Why an invoice number was rejected
508#[derive(Debug, Clone, Copy, PartialEq, Eq, thiserror::Error)]
509pub enum InvoiceNumberError {
510    /// Empty or longer than 16 characters
511    #[error("must be 1-{INVOICE_NUMBER_MAX_LEN} characters")]
512    Length,
513    /// Contains something other than letters, digits, `-` and `/`
514    #[error("only letters, digits, '-' and '/' are allowed")]
515    Characters,
516}

518/// Why a line item was rejected
519#[derive(Debug, Clone, PartialEq, thiserror::Error)]
520pub enum LineItemError {
521    /// The description is empty or only whitespace
522    #[error("description cannot be empty")]
523    EmptyDescription,
524    /// The quantity is zero or negative
525    #[error("quantity must be positive")]
526    NonPositiveQuantity,
527    /// The unit price is negative
528    #[error("unit price cannot be negative")]
529    NegativeUnitPrice,
530    /// The GST rate is outside 0-100
531    #[error("GST rate must be between 0 and {MAX_GST_RATE}, got {0}")]
532    RateOutOfRange(BigDecimal),
533}

535/// Invoice-related errors
536#[derive(Debug, thiserror::Error)]
537pub enum InvoiceError {
538    /// The GSTIN is malformed
539    #[error("invalid GSTIN {value}: {reason}")]
540    InvalidGstin {
541        /// The value as given
542        value: String,
543        /// The first rule it breaks
544        reason: GstinError,
545    },
546    /// The invoice number breaks Rule 46
547    #[error("invalid invoice number {value}: {reason}")]
548    InvalidInvoiceNumber {
549        /// The value as given
550        value: String,
551        /// The first rule it breaks
552        reason: InvoiceNumberError,
553    },
554    /// The HSN/SAC code is not 4, 6 or 8 digits
555    #[error("invalid HSN/SAC code: {0}")]
556    InvalidHsnSac(String),
557    /// The HSN/SAC code is well-formed but not in the master data
558    #[error("HSN/SAC code not in the master data: {0}")]
559    UnknownHsnSac(String),
560    /// A line item field is invalid
561    #[error("invalid line item: {0}")]
562    InvalidLineItem(LineItemError),
563    /// The invoice has no line items
564    #[error("an invoice needs at least one line item")]
565    EmptyInvoice,
566    /// The GST rate is inconsistent
567    #[error(transparent)]
568    Gst(#[from] GstError),
