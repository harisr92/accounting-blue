1//! Compliance checks for a GST invoice
2//!
3//! [`GstInvoice::new`] rejects malformed input, but an invoice's fields are public and can be
4//! edited afterwards, and some rules depend on things construction can't see: the date the
5//! invoice is checked on and the HSN/SAC master. [`validate_invoice`] runs every
6//! rule and returns an [`InvoiceValidationReport`] listing each [`ComplianceIssue`] found, so a
7//! caller sees all problems at once rather than the first.
8//!
9//! Issues are [`Severity::Error`] when the invoice must not be issued or posted, and
10//! [`Severity::Warning`] when it is valid but worth a second look.
11//!
12//! [`validate_credit_note`] runs the same rules over a [`CreditNote`], which is a
13//! [`GstDocument`] too, plus the rules only a note has.
14

23/// How serious a [`ComplianceIssue`] is
24#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
25pub enum Severity {
26    /// The invoice must not be issued or posted
27    Error,
28    /// The invoice is valid, but something looks unusual
29    Warning,
30}
31
32/// One rule an invoice breaks
33///
34/// `line` fields are indices into [`GstInvoice::line_items`]; messages number lines from 1.
35#[derive(Debug, Clone, PartialEq, thiserror::Error)]
36pub enum ComplianceIssue {
37    /// The invoice number breaks Rule 46
38    #[error("invoice number {value:?}: {reason}")]
39    InvalidInvoiceNumber {
40        /// The invoice number
41        value: String,
42        /// The first rule it breaks
43        reason: InvoiceNumberError,
44    },
45    /// The invoice has no line items
46    #[error("an invoice needs at least one line item")]
47    NoLineItems,
48    /// A line's HSN/SAC code is missing or not 4, 6 or 8 digits
49    #[error("line {}: HSN/SAC code {code:?} must be 4, 6 or 8 digits", .line + 1)]
50    InvalidHsnSac {
51        /// Index of the line
52        line: usize,
53        /// The code as given
54        code: String,
55    },
56    /// A line's description, quantity, price or rate is invalid
57    #[error("line {}: {reason}", .line + 1)]
58    InvalidLineItem {
59        /// Index of the line
60        line: usize,
61        /// The first rule the line breaks
62        reason: LineItemError,
63    },
64    /// The invoice is dated after the day it is checked
65    #[error("invoice date {invoice_date} is after {as_of}")]
66    FutureDate {
67        /// Date on the invoice
68        invoice_date: NaiveDate,
69        /// Date the invoice was checked against
70        as_of: NaiveDate,
71    },
72    /// Seller and buyer have the same GSTIN
73    #[error("seller and buyer have the same GSTIN {0}")]
74    SameSellerAndBuyer(String),
75    /// A line's HSN/SAC code is well-formed but not in the master data
76    #[error("line {}: HSN/SAC code {code} is not in the master data", .line + 1)]
77    UnknownHsnSac {
78        /// Index of the line
79        line: usize,
80        /// The code as given
81        code: String,
82    },
83    /// A line's rate differs from the default rate for its HSN/SAC code
84    #[error("line {}: rate {rate}% differs from the {default}% default for {code}", .line + 1)]
85    RateDiffersFromHsnDefault {
86        /// Index of the line
87        line: usize,
88        /// The line's HSN/SAC code
89        code: String,
90        /// Rate charged on the line
91        rate: BigDecimal,
92        /// Default rate in the master
93        default: BigDecimal,
94    },
95    /// A line's taxable value rounds to zero paise, so it adds nothing to the invoice
96    #[error("line {}: taxable value rounds to zero", .line + 1)]
97    ZeroValueLine {
98        /// Index of the line
99        line: usize,
100    },
101    /// A credit note is dated before the invoice it corrects
102    #[error("credit note dated {note_date} is before its invoice dated {original_date}")]
103    NoteBeforeOriginal {
104        /// Date of the note
105        note_date: NaiveDate,
106        /// Date of the invoice it corrects
107        original_date: NaiveDate,
108    },
109    /// A credit note's original supply kind can't apply to its buyer, such as B2B for an
110    /// unregistered buyer
111    #[error("a {kind} invoice can't have been issued to {buyer}")]
112    OriginalKindMismatch {
113        /// The supply kind the note records for its invoice
114        kind: SupplyKind,
115        /// The note's buyer
116        buyer: Recipient,
117    },
118    /// A credit note credits more than the invoice it corrects was worth
119    #[error("credit note total {credited} is more than its invoice's value {invoice_value}")]
120    CreditExceedsInvoice {
121        /// The note's total, tax included
122        credited: BigDecimal,
123        /// The invoice's value, tax included
124        invoice_value: BigDecimal,
125    },
126    /// A credit note is issued after the Section 34(2) deadline for its invoice
127    #[error("credit note dated {note_date} is after the {deadline} deadline for its invoice")]
128    CreditNoteTooLate {
129        /// Date of the note
130        note_date: NaiveDate,
131        /// Last day it could be issued
132        deadline: NaiveDate,
133    },
134}

136impl ComplianceIssue {
137    /// How serious this issue is
138    #[must_use]
139    pub fn severity(&self) -> Severity {
140        match self {
141            Self::InvalidInvoiceNumber { .. }
142            | Self::NoLineItems
143            | Self::InvalidHsnSac { .. }
144            | Self::InvalidLineItem { .. }
145            | Self::FutureDate { .. }
146            | Self::SameSellerAndBuyer(_)
147            | Self::NoteBeforeOriginal { .. }
148            | Self::OriginalKindMismatch { .. }
149            | Self::CreditExceedsInvoice { .. }
150            | Self::CreditNoteTooLate { .. } => Severity::Error,
151            Self::UnknownHsnSac { .. }
152            | Self::RateDiffersFromHsnDefault { .. }
153            | Self::ZeroValueLine { .. } => Severity::Warning,
154        }
155    }
156}

158/// Every issue [`validate_invoice`] found, in rule order
159#[derive(Debug, Clone, Default, PartialEq)]
160pub struct InvoiceValidationReport {
161    issues: Vec<ComplianceIssue>,
162}
163
164impl InvoiceValidationReport {
165    /// All issues
166    #[must_use]
167    pub fn issues(&self) -> &[ComplianceIssue] {
168        &self.issues
169    }
170
171    /// Issues of severity [`Severity::Error`]
172    pub fn errors(&self) -> impl Iterator<Item = &ComplianceIssue> {
173        self.of_severity(Severity::Error)
174    }
175
176    /// Issues of severity [`Severity::Warning`]
177    pub fn warnings(&self) -> impl Iterator<Item = &ComplianceIssue> {
178        self.of_severity(Severity::Warning)
179    }
180
181    fn of_severity(&self, severity: Severity) -> impl Iterator<Item = &ComplianceIssue> {
182        self.issues
183            .iter()
184            .filter(move |issue| issue.severity() == severity)
185    }
186
187    /// Whether the invoice has no errors (warnings are allowed)
188    #[must_use]
189    pub fn is_compliant(&self) -> bool {
190        self.errors().next().is_none()
191    }
192
193    /// Consume the report, returning its issues
194    #[must_use]
195    pub fn into_issues(self) -> Vec<ComplianceIssue> {
196        self.issues
197    }
198}

200/// A rule whose issues are errors: every issue it finds on a document checked on a given date
201type ErrorRule = fn(&dyn GstDocument, NaiveDate) -> Vec<ComplianceIssue>;
202
203/// A rule whose issues are warnings: every issue it finds on a document, given the HSN/SAC master
204type WarningRule = fn(&dyn GstDocument, &HsnMaster) -> Vec<ComplianceIssue>;
205
206/// A rule only credit notes have, whose issues are errors
207type NoteRule = fn(&CreditNote) -> Vec<ComplianceIssue>;
208
209/// The rules that block an invoice, in report order
210const ERROR_RULES: &[ErrorRule] = &[
211    invoice_number_rule,
212    line_items_present_rule,
213    hsn_sac_shape_rule,
214    line_fields_rule,
215    date_rule,
216    parties_rule,
217];
218
219/// The rules that only flag an invoice for a second look, reported after the errors
220const WARNING_RULES: &[WarningRule] = &[hsn_master_rule, zero_value_rule];
221
222/// The rules a credit note must meet on top of [`ERROR_RULES`], reported after them
223const NOTE_RULES: &[NoteRule] = &[
224    note_date_rule,
225    note_kind_rule,
226    note_value_rule,
227    note_deadline_rule,
228];

230/// Check an invoice against every compliance rule
231///
232/// `as_of` is the day the invoice is checked, usually today; an invoice dated after it is
233/// flagged. `master` supplies the HSN/SAC codes and default rates the lines are compared with,
234/// usually [`HsnMaster::global`]. The function is pure, so the caller supplies both.
235///
236/// # Example
237///
238/// ```
239/// use accounting_core::invoice::{validate_invoice, ComplianceIssue, GstInvoice, GstLineItem, Gstin, HsnMaster};
240/// use bigdecimal::BigDecimal;
241/// use chrono::NaiveDate;
242///
243/// # fn main() -> Result<(), accounting_core::Error> {
244/// let item = GstLineItem::new("998314", "IT consulting", BigDecimal::from(1),
245///     BigDecimal::from(1000), BigDecimal::from(18))?;
246/// let invoice = GstInvoice::new("INV-001", NaiveDate::from_ymd_opt(2024, 11, 15).unwrap(),
247///     Gstin::parse("27AAPFU0939F1ZV")?, Gstin::parse("27AAPFU0939F2ZU")?, vec![item])?;
248/// let master = HsnMaster::global();
249///
250/// let checked_on = NaiveDate::from_ymd_opt(2024, 11, 30).unwrap();
251/// assert!(validate_invoice(&invoice, checked_on, master).is_compliant());
252///
253/// let report = validate_invoice(&invoice, NaiveDate::from_ymd_opt(2024, 11, 1).unwrap(), master);
254/// assert!(!report.is_compliant());
255/// assert!(matches!(report.issues(), [ComplianceIssue::FutureDate { .. }]));
256/// # Ok(())
257/// # }
258/// ```
259#[must_use]
260pub fn validate_invoice(
261    invoice: &GstInvoice,
262    as_of: NaiveDate,
263    master: &HsnMaster,
264) -> InvoiceValidationReport {
265    report(compliance_errors(invoice, as_of), invoice, master)
266}

268/// Check a credit note against every compliance rule
269///
270/// The note meets the same rules as an invoice, as [`validate_invoice`] describes, and four of
271/// its own: it is not dated before the invoice it corrects, its buyer is one the invoice's supply
272/// kind can apply to, its total is no more than the invoice's value, and it is issued by
273/// [`CreditNote::deadline`] (Section 34(2)). All four are errors.
274///
275/// # Example
276///
277/// ```
278/// use accounting_core::invoice::{validate_credit_note, ComplianceIssue, CreditNote, GstInvoice,
279///     GstLineItem, Gstin, HsnMaster};
280/// use bigdecimal::BigDecimal;
281/// use chrono::NaiveDate;
282///
283/// # fn main() -> Result<(), accounting_core::Error> {
284/// let item = GstLineItem::new("998314", "IT consulting", BigDecimal::from(1),
285///     BigDecimal::from(1000), BigDecimal::from(18))?;
286/// let invoice = GstInvoice::new("INV-001", NaiveDate::from_ymd_opt(2024, 11, 15).unwrap(),
287///     Gstin::parse("27AAPFU0939F1ZV")?, Gstin::parse("27AAPFU0939F2ZU")?, vec![item.clone()])?;
288///
289/// // The deadline for an invoice of FY 2024-25 is 30 November 2025
290/// let late = NaiveDate::from_ymd_opt(2025, 12, 1).unwrap();
291/// let note = CreditNote::new(&invoice, "CN-001", late, vec![item])?;
292/// let report = validate_credit_note(&note, late, HsnMaster::global());
293/// assert!(matches!(report.issues(), [ComplianceIssue::CreditNoteTooLate { .. }]));
294/// # Ok(())
295/// # }
296/// ```
297#[must_use]
298pub fn validate_credit_note(
299    note: &CreditNote,
300    as_of: NaiveDate,
301    master: &HsnMaster,
302) -> InvoiceValidationReport {
303    report(credit_note_errors(note, as_of), note, master)
304}

306/// A report of `errors` followed by the warnings for `document`
307fn report(
308    errors: Vec<ComplianceIssue>,
309    document: &dyn GstDocument,
310    master: &HsnMaster,
311) -> InvoiceValidationReport {
312    let warnings = WARNING_RULES.iter().flat_map(|rule| rule(document, master));
313    InvoiceValidationReport {
314        issues: errors.into_iter().chain(warnings).collect(),
315    }
316}

318/// Only the error-severity issues, which need no HSN/SAC master
319pub(crate) fn compliance_errors(
320    document: &dyn GstDocument,
321    as_of: NaiveDate,
322) -> Vec<ComplianceIssue> {
323    ERROR_RULES
324        .iter()
325        .flat_map(|rule| rule(document, as_of))
326        .collect()
327}
328
329/// Only the error-severity issues of a credit note: the document rules, then its own
330pub(crate) fn credit_note_errors(note: &CreditNote, as_of: NaiveDate) -> Vec<ComplianceIssue> {
331    let note_issues = NOTE_RULES.iter().flat_map(|rule| rule(note));
332    compliance_errors(note, as_of)
333        .into_iter()
334        .chain(note_issues)
335        .collect()
336}

338/// Issues found on each line by `check`, which is given the line's index and item
339fn per_line(
340    document: &dyn GstDocument,
341    check: impl Fn(usize, &GstLineItem) -> Option<ComplianceIssue>,
342) -> Vec<ComplianceIssue> {
343    document
344        .line_items()
345        .iter()
346        .enumerate()
347        .filter_map(|(line, item)| check(line, item))
348        .collect()
349}

351fn invoice_number_rule(document: &dyn GstDocument, _as_of: NaiveDate) -> Vec<ComplianceIssue> {
352    invoice_number_error(document.number())
353        .map(|reason| ComplianceIssue::InvalidInvoiceNumber {
354            value: document.number().to_string(),
355            reason,
356        })
357        .into_iter()
358        .collect()
359}

361fn line_items_present_rule(document: &dyn GstDocument, _as_of: NaiveDate) -> Vec<ComplianceIssue> {
362    if document.line_items().is_empty() {
363        vec![ComplianceIssue::NoLineItems]
364    } else {
365        Vec::new()
366    }
367}

369/// The HSN/SAC shape rule [`GstLineItem::new`] applies to each line
370fn hsn_sac_shape_rule(document: &dyn GstDocument, _as_of: NaiveDate) -> Vec<ComplianceIssue> {
371    per_line(document, |line, item| {
372        (!is_valid_hsn_sac(&item.hsn_sac)).then(|| ComplianceIssue::InvalidHsnSac {
373            line,
374            code: item.hsn_sac.clone(),
375        })
376    })
377}

379/// The other construction rules for each line: description, quantity, price and rate
380fn line_fields_rule(document: &dyn GstDocument, _as_of: NaiveDate) -> Vec<ComplianceIssue> {
381    per_line(document, |line, item| {
382        item.check()
383            .err()
384            .map(|reason| ComplianceIssue::InvalidLineItem { line, reason })
385    })
386}

388fn date_rule(document: &dyn GstDocument, as_of: NaiveDate) -> Vec<ComplianceIssue> {
389    if document.date() > as_of {
390        vec![ComplianceIssue::FutureDate {
391            invoice_date: document.date(),
392            as_of,
393        }]
394    } else {
395        Vec::new()
396    }
397}

399fn parties_rule(document: &dyn GstDocument, _as_of: NaiveDate) -> Vec<ComplianceIssue> {
400    if document.buyer().gstin() == Some(document.seller_gstin()) {
401        vec![ComplianceIssue::SameSellerAndBuyer(
402            document.seller_gstin().to_string(),
403        )]
404    } else {
405        Vec::new()
406    }
407}

409/// Warn about well-formed codes the master doesn't know, and rates that differ from its default
410///
411/// A rate is only compared when the master has the line's exact code: a fallback heading's rate
412/// may not apply to every tariff item under it. It is also not compared on a line that already
413/// breaks a field rule, on an exempt or non-GST line (whose rate is 0 by definition), or on an
414/// invoice dated before [`HsnMaster::effective_from`], when the schedule didn't apply yet. A credit
415/// note goes by its original invoice's date ([`GstDocument::supply_date`]), since it reverses tax
416/// at the rates that invoice charged.
417fn hsn_master_rule(document: &dyn GstDocument, master: &HsnMaster) -> Vec<ComplianceIssue> {
418    let schedule_applies = document.supply_date() >= master.effective_from();
419    per_line(document, |line, item| {
420        if !is_valid_hsn_sac(&item.hsn_sac) {
421            return None;
422        }
423        let Some(entry) = master.lookup(&item.hsn_sac) else {
424            return Some(ComplianceIssue::UnknownHsnSac {
425                line,
426                code: item.hsn_sac.clone(),
427            });
428        };
429        let compare_rate = schedule_applies
430            && item.treatment.is_taxable()
431            && entry.code == item.hsn_sac
432            && item.check().is_ok();
433        (compare_rate && entry.gst_rate != item.gst_rate).then(|| {
434            ComplianceIssue::RateDiffersFromHsnDefault {
435                line,
436                code: item.hsn_sac.clone(),
437                rate: item.gst_rate.clone(),
438                default: entry.gst_rate.clone(),
439            }
440        })
441    })
442}

444fn zero_value_rule(document: &dyn GstDocument, _master: &HsnMaster) -> Vec<ComplianceIssue> {
445    per_line(document, |line, item| {
446        round_to_paise(&item.taxable_value())
447            .is_zero()
448            .then_some(ComplianceIssue::ZeroValueLine { line })
449    })
450}

452fn note_date_rule(note: &CreditNote) -> Vec<ComplianceIssue> {
453    if note.predates_original() {
454        vec![ComplianceIssue::NoteBeforeOriginal {
455            note_date: note.note_date,
456            original_date: note.original.date,
457        }]
458    } else {
459        Vec::new()
460    }
461}

463/// The note's buyer must be one its original invoice's supply kind can apply to, or GSTR-1 would
464/// report it in the wrong table or drop it from Table 9B
465fn note_kind_rule(note: &CreditNote) -> Vec<ComplianceIssue> {
466    if note.original_kind_fits_buyer() {
467        Vec::new()
468    } else {
469        vec![ComplianceIssue::OriginalKindMismatch {
470            kind: note.original.kind,
471            buyer: note.buyer.clone(),
472        }]
473    }
474}

476/// A note may credit at most its invoice's value; a note whose tax can't be computed is left to
477/// [`line_fields_rule`]
478fn note_value_rule(note: &CreditNote) -> Vec<ComplianceIssue> {
479    match note.credit_over_value() {
480        Ok(Some(credited)) => vec![ComplianceIssue::CreditExceedsInvoice {
481            credited,
482            invoice_value: note.original.value.clone(),
483        }],
484        Ok(None) | Err(_) => Vec::new(),
485    }
486}
487
488fn note_deadline_rule(note: &CreditNote) -> Vec<ComplianceIssue> {
489    let deadline = note.deadline();
490    if note.note_date > deadline {
491        vec![ComplianceIssue::CreditNoteTooLate {
492            note_date: note.note_date,
493            deadline,
494        }]
495    } else {
496        Vec::new()
497    }
