1//! Compliance checks for a GST invoice
2//!
3//! [`GstInvoice::new`] rejects malformed input, but an invoice's fields are public and can be
4//! edited afterwards, and some rules depend on things construction can't see: the date the
5//! invoice is checked on and the HSN/SAC master. [`validate_invoice`] runs every
6//! rule and returns an [`InvoiceValidationReport`] listing each [`ComplianceIssue`] found, so a
7//! caller sees all problems at once rather than the first.
8//!
9//! Issues are [`Severity::Error`] when the invoice must not be issued or posted, and
10//! [`Severity::Warning`] when it is valid but worth a second look.
11
12use super::hsn_lookup::{is_valid_hsn_sac, HsnMaster};
13use super::types::{invoice_number_error, GstInvoice, GstLineItem};
14use super::types::{InvoiceNumberError, LineItemError};
15use crate::tax::round_to_paise;
16use bigdecimal::{BigDecimal, Zero};
17use chrono::NaiveDate;

19/// How serious a [`ComplianceIssue`] is
20#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
21pub enum Severity {
22    /// The invoice must not be issued or posted
23    Error,
24    /// The invoice is valid, but something looks unusual
25    Warning,
26}
27
28/// One rule an invoice breaks
29///
30/// `line` fields are indices into [`GstInvoice::line_items`]; messages number lines from 1.
31#[derive(Debug, Clone, PartialEq, thiserror::Error)]
32pub enum ComplianceIssue {
33    /// The invoice number breaks Rule 46
34    #[error("invoice number {value:?}: {reason}")]
35    InvalidInvoiceNumber {
36        /// The invoice number
37        value: String,
38        /// The first rule it breaks
39        reason: InvoiceNumberError,
40    },
41    /// The invoice has no line items
42    #[error("an invoice needs at least one line item")]
43    NoLineItems,
44    /// A line's HSN/SAC code is missing or not 4, 6 or 8 digits
45    #[error("line {}: HSN/SAC code {code:?} must be 4, 6 or 8 digits", .line + 1)]
46    InvalidHsnSac {
47        /// Index of the line
48        line: usize,
49        /// The code as given
50        code: String,
51    },
52    /// A line's description, quantity, price or rate is invalid
53    #[error("line {}: {reason}", .line + 1)]
54    InvalidLineItem {
55        /// Index of the line
56        line: usize,
57        /// The first rule the line breaks
58        reason: LineItemError,
59    },
60    /// The invoice is dated after the day it is checked
61    #[error("invoice date {invoice_date} is after {as_of}")]
62    FutureDate {
63        /// Date on the invoice
64        invoice_date: NaiveDate,
65        /// Date the invoice was checked against
66        as_of: NaiveDate,
67    },
68    /// Seller and buyer have the same GSTIN
69    #[error("seller and buyer have the same GSTIN {0}")]
70    SameSellerAndBuyer(String),
71    /// A line's HSN/SAC code is well-formed but not in the master data
72    #[error("line {}: HSN/SAC code {code} is not in the master data", .line + 1)]
73    UnknownHsnSac {
74        /// Index of the line
75        line: usize,
76        /// The code as given
77        code: String,
78    },
79    /// A line's rate differs from the default rate for its HSN/SAC code
80    #[error("line {}: rate {rate}% differs from the {default}% default for {code}", .line + 1)]
81    RateDiffersFromHsnDefault {
82        /// Index of the line
83        line: usize,
84        /// The line's HSN/SAC code
85        code: String,
86        /// Rate charged on the line
87        rate: BigDecimal,
88        /// Default rate in the master
89        default: BigDecimal,
90    },
91    /// A line's taxable value rounds to zero paise, so it adds nothing to the invoice
92    #[error("line {}: taxable value rounds to zero", .line + 1)]
93    ZeroValueLine {
94        /// Index of the line
95        line: usize,
96    },
97}
98
99impl ComplianceIssue {
100    /// How serious this issue is
101    #[must_use]
102    pub fn severity(&self) -> Severity {
103        match self {
104            Self::InvalidInvoiceNumber { .. }
105            | Self::NoLineItems
106            | Self::InvalidHsnSac { .. }
107            | Self::InvalidLineItem { .. }
108            | Self::FutureDate { .. }
109            | Self::SameSellerAndBuyer(_) => Severity::Error,
110            Self::UnknownHsnSac { .. }
111            | Self::RateDiffersFromHsnDefault { .. }
112            | Self::ZeroValueLine { .. } => Severity::Warning,
113        }
114    }
115}

117/// Every issue [`validate_invoice`] found, in rule order
118#[derive(Debug, Clone, Default, PartialEq)]
119pub struct InvoiceValidationReport {
120    issues: Vec<ComplianceIssue>,
121}
122
123impl InvoiceValidationReport {
124    /// All issues
125    #[must_use]
126    pub fn issues(&self) -> &[ComplianceIssue] {
127        &self.issues
128    }
129
130    /// Issues of severity [`Severity::Error`]
131    pub fn errors(&self) -> impl Iterator<Item = &ComplianceIssue> {
132        self.of_severity(Severity::Error)
133    }
134
135    /// Issues of severity [`Severity::Warning`]
136    pub fn warnings(&self) -> impl Iterator<Item = &ComplianceIssue> {
137        self.of_severity(Severity::Warning)
138    }
139
140    fn of_severity(&self, severity: Severity) -> impl Iterator<Item = &ComplianceIssue> {
141        self.issues
142            .iter()
143            .filter(move |issue| issue.severity() == severity)
144    }
145
146    /// Whether the invoice has no errors (warnings are allowed)
147    #[must_use]
148    pub fn is_compliant(&self) -> bool {
149        self.errors().next().is_none()
150    }
151
152    /// Consume the report, returning its issues
153    #[must_use]
154    pub fn into_issues(self) -> Vec<ComplianceIssue> {
155        self.issues
156    }
157}

159/// A rule whose issues are errors: every issue it finds on an invoice checked on a given date
160type ErrorRule = fn(&GstInvoice, NaiveDate) -> Vec<ComplianceIssue>;
161
162/// A rule whose issues are warnings: every issue it finds on an invoice, given the HSN/SAC master
163type WarningRule = fn(&GstInvoice, &HsnMaster) -> Vec<ComplianceIssue>;
164
165/// The rules that block an invoice, in report order
166const ERROR_RULES: &[ErrorRule] = &[
167    invoice_number_rule,
168    line_items_present_rule,
169    hsn_sac_shape_rule,
170    line_fields_rule,
171    date_rule,
172    parties_rule,
173];
174
175/// The rules that only flag an invoice for a second look, reported after the errors
176const WARNING_RULES: &[WarningRule] = &[hsn_master_rule, zero_value_rule];
177
178/// Check an invoice against every compliance rule
179///
180/// `as_of` is the day the invoice is checked, usually today; an invoice dated after it is
181/// flagged. `master` supplies the HSN/SAC codes and default rates the lines are compared with,
182/// usually [`HsnMaster::global`]. The function is pure, so the caller supplies both.
183///
184/// # Example
185///
186/// ```
187/// use accounting_core::invoice::{validate_invoice, ComplianceIssue, GstInvoice, GstLineItem, Gstin, HsnMaster};
188/// use bigdecimal::BigDecimal;
189/// use chrono::NaiveDate;
190///
191/// # fn main() -> Result<(), accounting_core::Error> {
192/// let item = GstLineItem::new("998314", "IT consulting", BigDecimal::from(1),
193///     BigDecimal::from(1000), BigDecimal::from(18))?;
194/// let invoice = GstInvoice::new("INV-001", NaiveDate::from_ymd_opt(2024, 11, 15).unwrap(),
195///     Gstin::parse("27AAPFU0939F1ZV")?, Gstin::parse("27AAPFU0939F2ZU")?, vec![item])?;
196/// let master = HsnMaster::global();
197///
198/// let checked_on = NaiveDate::from_ymd_opt(2024, 11, 30).unwrap();
199/// assert!(validate_invoice(&invoice, checked_on, master).is_compliant());
200///
201/// let report = validate_invoice(&invoice, NaiveDate::from_ymd_opt(2024, 11, 1).unwrap(), master);
202/// assert!(!report.is_compliant());
203/// assert!(matches!(report.issues(), [ComplianceIssue::FutureDate { .. }]));
204/// # Ok(())
205/// # }
206/// ```
207#[must_use]
208pub fn validate_invoice(
209    invoice: &GstInvoice,
210    as_of: NaiveDate,
211    master: &HsnMaster,
212) -> InvoiceValidationReport {
213    let warnings = WARNING_RULES.iter().flat_map(|rule| rule(invoice, master));
214    InvoiceValidationReport {
215        issues: compliance_errors(invoice, as_of)
216            .into_iter()
217            .chain(warnings)
218            .collect(),
219    }
220}

222/// Only the error-severity issues, which need no HSN/SAC master
223pub(crate) fn compliance_errors(invoice: &GstInvoice, as_of: NaiveDate) -> Vec<ComplianceIssue> {
224    ERROR_RULES
225        .iter()
226        .flat_map(|rule| rule(invoice, as_of))
227        .collect()
228}

230/// Issues found on each line by `check`, which is given the line's index and item
231fn per_line(
232    invoice: &GstInvoice,
233    check: impl Fn(usize, &GstLineItem) -> Option<ComplianceIssue>,
234) -> Vec<ComplianceIssue> {
235    invoice
236        .line_items
237        .iter()
238        .enumerate()
239        .filter_map(|(line, item)| check(line, item))
240        .collect()
241}

243fn invoice_number_rule(invoice: &GstInvoice, _as_of: NaiveDate) -> Vec<ComplianceIssue> {
244    invoice_number_error(&invoice.invoice_number)
245        .map(|reason| ComplianceIssue::InvalidInvoiceNumber {
246            value: invoice.invoice_number.clone(),
247            reason,
248        })
249        .into_iter()
250        .collect()
251}
252
253fn line_items_present_rule(invoice: &GstInvoice, _as_of: NaiveDate) -> Vec<ComplianceIssue> {
254    if invoice.line_items.is_empty() {
255        vec![ComplianceIssue::NoLineItems]
256    } else {
257        Vec::new()
258    }
259}

261/// The HSN/SAC shape rule [`GstLineItem::new`] applies to each line
262fn hsn_sac_shape_rule(invoice: &GstInvoice, _as_of: NaiveDate) -> Vec<ComplianceIssue> {
263    per_line(invoice, |line, item| {
264        (!is_valid_hsn_sac(&item.hsn_sac)).then(|| ComplianceIssue::InvalidHsnSac {
265            line,
266            code: item.hsn_sac.clone(),
267        })
268    })
269}
270
271/// The other construction rules for each line: description, quantity, price and rate
272fn line_fields_rule(invoice: &GstInvoice, _as_of: NaiveDate) -> Vec<ComplianceIssue> {
273    per_line(invoice, |line, item| {
274        item.check()
275            .err()
276            .map(|reason| ComplianceIssue::InvalidLineItem { line, reason })
277    })
278}

280fn date_rule(invoice: &GstInvoice, as_of: NaiveDate) -> Vec<ComplianceIssue> {
281    if invoice.invoice_date > as_of {
282        vec![ComplianceIssue::FutureDate {
283            invoice_date: invoice.invoice_date,
284            as_of,
285        }]
286    } else {
287        Vec::new()
288    }
289}

291fn parties_rule(invoice: &GstInvoice, _as_of: NaiveDate) -> Vec<ComplianceIssue> {
292    if invoice.seller_gstin == invoice.buyer_gstin {
293        vec![ComplianceIssue::SameSellerAndBuyer(
294            invoice.seller_gstin.to_string(),
295        )]
296    } else {
297        Vec::new()
298    }
299}

301/// Warn about well-formed codes the master doesn't know, and rates that differ from its default
302///
303/// A rate is only compared when the master has the line's exact code: a fallback heading's rate
304/// may not apply to every tariff item under it. It is also not compared on a line that already
305/// breaks a field rule, or on an invoice dated before [`HsnMaster::effective_from`], when the
306/// schedule didn't apply yet.
307fn hsn_master_rule(invoice: &GstInvoice, master: &HsnMaster) -> Vec<ComplianceIssue> {
308    let schedule_applies = invoice.invoice_date >= master.effective_from();
309    per_line(invoice, |line, item| {
310        if !is_valid_hsn_sac(&item.hsn_sac) {
311            return None;
312        }
313        let Some(entry) = master.lookup(&item.hsn_sac) else {
314            return Some(ComplianceIssue::UnknownHsnSac {
315                line,
316                code: item.hsn_sac.clone(),
317            });
318        };
319        let compare_rate = schedule_applies && entry.code == item.hsn_sac && item.check().is_ok();
320        (compare_rate && entry.gst_rate != item.gst_rate).then(|| {
321            ComplianceIssue::RateDiffersFromHsnDefault {
322                line,
323                code: item.hsn_sac.clone(),
324                rate: item.gst_rate.clone(),
325                default: entry.gst_rate.clone(),
326            }
327        })
328    })
329}

331fn zero_value_rule(invoice: &GstInvoice, _master: &HsnMaster) -> Vec<ComplianceIssue> {
332    per_line(invoice, |line, item| {
333        round_to_paise(&item.taxable_value())
334            .is_zero()
335            .then_some(ComplianceIssue::ZeroValueLine { line })
336    })
