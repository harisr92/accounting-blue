1//! GST state codes and their names
2//!
3//! [`GST_STATES`] lists every state and union territory code the GST portal uses, with `97`
4//! (other territory) and `99` (centre jurisdiction). An invoice prints a place of supply as
5//! [`state_label`], such as `29 - Karnataka`, rather than the bare code.
6//!
7//! # Example
8//!
9//! ```
10//! use accounting_core::invoice::{state_label, state_name};
11//!
12//! assert_eq!(state_name("29"), Some("Karnataka"));
13//! assert_eq!(state_label("29"), "29 - Karnataka");
14//! assert_eq!(state_label("00"), "00");
15//! ```
16

61/// The name of the state or territory with GST code `code`, if there is one
62#[must_use]
63pub fn state_name(code: impl Into<String>) -> Option<&'static str> {
64    let code = code.into();
65    GST_STATES
66        .iter()
67        .find(|(known, _)| *known == code)
68        .map(|(_, name)| *name)
69}
70
71/// `code` with its name, as in `29 - Karnataka`; the bare code when it names no state
72#[must_use]
73pub fn state_label(code: impl Into<String>) -> String {
74    let code = code.into();
75    match state_name(code.as_str()) {
76        Some(name) => format!("{code} - {name}"),
77        None => code,
78    }
79}
