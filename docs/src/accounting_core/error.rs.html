1//! Error types
2//!
3//! Each module has its own error enum; [`Error`] wraps all of them so an application can use a
4//! single `?`-friendly type across ledger, GST, invoice and reconciliation calls.

15/// Boxed error from a storage backend, kept as the `source` of a storage failure
16pub type BoxError = Box<dyn std::error::Error + Send + Sync>;

18/// Any error raised by this crate
19#[derive(Debug, thiserror::Error)]
20pub enum Error {
21    /// Ledger, account or transaction failure
22    #[error(transparent)]
23    Ledger(#[from] LedgerError),
24    /// GST rate or calculation failure
25    #[error(transparent)]
26    Gst(#[from] GstError),
27    /// GST invoice failure
28    #[error(transparent)]
29    Invoice(#[from] InvoiceError),
30    /// An invoice could not be posted to the ledger
31    #[error(transparent)]
32    Posting(#[from] PostingError),
33    /// A GSTR-1 return could not be built or written
34    #[error(transparent)]
35    Gstr1(#[from] Gstr1Error),
36    /// Reconciliation storage or import failure
37    #[error(transparent)]
38    Reconciliation(#[from] ReconciliationError),
39    /// An invoice could not be rendered to PDF
40    #[cfg(feature = "pdf")]
41    #[error(transparent)]
42    Pdf(#[from] PdfError),
43}

45/// Result type using the crate-level [`Error`]
46pub type Result<T> = std::result::Result<T, Error>;

48/// Why a text field was rejected
49#[derive(Debug, Clone, Copy, PartialEq, Eq, thiserror::Error)]
50pub enum FieldError {
51    /// The field is empty or only whitespace
52    #[error("cannot be empty")]
53    Empty,
54    /// The field is longer than allowed
55    #[error("cannot exceed {max} characters")]
56    TooLong {
57        /// Maximum length in bytes
58        max: usize,
59    },
60    /// The field contains characters outside the allowed set
61    #[error("can only contain alphanumeric characters, dashes, and underscores")]
62    InvalidCharacters,
63}

65/// Why pagination parameters were rejected
66#[derive(Debug, Clone, Copy, PartialEq, Eq, thiserror::Error)]
67pub enum PaginationError {
68    /// Pages are numbered from 1
69    #[error("page must be 1 or greater")]
70    PageOutOfRange,
71    /// Page size is outside the accepted range
72    #[error("page size must be between 1 and {max}")]
73    PageSizeOutOfRange {
74        /// Largest accepted page size
75        max: u32,
76    },
77}

79/// Errors raised by the ledger and its storage
80#[derive(Debug, thiserror::Error)]
81pub enum LedgerError {
82    /// The storage backend failed
83    #[error("storage error: {0}")]
84    Storage(#[source] BoxError),
85    /// No account with this id exists
86    #[error("account not found: {0}")]
87    AccountNotFound(String),
88    /// No transaction with this id exists
89    #[error("transaction not found: {0}")]
90    TransactionNotFound(String),
91    /// An account with this id already exists
92    #[error("account with ID '{0}' already exists")]
93    DuplicateAccount(String),
94    /// The parent account named on a new account does not exist
95    #[error("parent account '{0}' does not exist")]
96    ParentNotFound(String),
97    /// Following parent links from this account leads back to it
98    #[error("account hierarchy has a cycle through '{0}'")]
99    AccountCycle(String),
100    /// The account has transactions posted to it, so deleting it would orphan them
101    #[error("account '{0}' has posted transactions and cannot be deleted")]
102    AccountHasTransactions(String),
103    /// The account is the parent of other accounts, so deleting it would orphan them
104    #[error("account '{0}' has child accounts and cannot be deleted")]
105    AccountHasChildren(String),
106    /// A transaction needs a debit and a credit leg at least
107    #[error("transaction must have at least two entries for double-entry bookkeeping")]
108    TooFewEntries,
109    /// Debits and credits differ
110    #[error("transaction is not balanced: debits = {debits}, credits = {credits}")]
111    Unbalanced {
112        /// Sum of debit entries
113        debits: BigDecimal,
114        /// Sum of credit entries
115        credits: BigDecimal,
116    },
117    /// An entry amount is zero or negative
118    #[error("entry amounts must be positive")]
119    NonPositiveAmount,
120    /// The same account appears twice on the same side of a transaction
121    #[error("account '{0}' appears multiple times with the same entry type in transaction")]
122    DuplicateEntry(String),
123    /// A text field failed validation
124    #[error("{field} {error}")]
125    InvalidField {
126        /// Which field was rejected
127        field: &'static str,
128        /// Why it was rejected
129        error: FieldError,
130    },
131    /// Pagination parameters were rejected
132    #[error("invalid pagination: {0}")]
133    InvalidPagination(#[from] PaginationError),
134}
135
136impl LedgerError {
137    /// Wrap a backend error as a [`LedgerError::Storage`]
138    pub fn storage(error: impl Into<BoxError>) -> Self {
139        Self::Storage(error.into())
140    }
141}

143/// Result type for ledger operations
144pub type LedgerResult<T> = std::result::Result<T, LedgerError>;
