1//! # Accounting Core
2//!
3//! A country-neutral accounting library providing double-entry bookkeeping, financial
4//! reporting and reconciliation. Indian GST support is an opt-in Cargo feature.
5//!
6//! ## Features
7//!
8//! - **Double-entry bookkeeping**: Complete transaction validation and balance tracking
9//! - **Invoicing**: Invoices, credit notes and discounts under a pluggable tax regime (none, a flat
10//!   VAT or sales tax, or your own), posted to the ledger
11//! - **Account management**: Support for Assets, Liabilities, Equity, Income, and Expense accounts
12//! - **Paginated responses**: Efficient pagination for large datasets with comprehensive metadata
13//! - **Financial reporting**: Balance sheets, income statements, and trial balance generation
14//! - **Reconciliation**: Bank statement and payment gateway reconciliation
15//! - **Storage abstraction**: Database-agnostic design with trait-based storage
16//!
17//! ## Optional features
18//!
19//! | Feature | Adds |
20//! |---------|------|
21//! | `india` | Indian GST on top of the invoicing core: rates and CGST/SGST/IGST maths (`gst::tax`), GSTIN, HSN/SAC master data, GST invoices and credit notes (`gst::invoice`), GSTR-1 returns (`gst::returns`), and Indian amount formatting (`utils::formatting`) |
22//! | `pdf` | Renders GST invoices to PDF; turns on `india` |
23//!
24//! ```toml
25//! [dependencies]
26//! accounting-core = { version = "0.4", features = ["india"] }
27//! ```
28//!
29//! ## Quick Start
30//!
31//! ```rust
32//! use accounting_core::{Ledger, AccountType, PaginationOption, PaginationParams};
33//! use accounting_core::utils::memory_storage::MemoryStorage;
34//! use bigdecimal::BigDecimal;
35//! use chrono::NaiveDate;
36//!
37//! # #[tokio::main]
38//! # async fn main() -> Result<(), Box<dyn std::error::Error>> {
39//! // Create a ledger with in-memory storage (for production, use a database storage)
40//! let storage = MemoryStorage::new();
41//! let mut ledger = Ledger::new(storage);
42//!
43//! // Create a cash account
44//! let cash_account = ledger.create_account(
45//!     "cash".to_string(),
46//!     "Cash Account".to_string(),
47//!     AccountType::Asset,
48//!     None,
49//! ).await?;
50//!
51//! // List all accounts with pagination (default: page 1, 50 items per page)
52//! let pagination = PaginationParams::new(1, 50)?;
53//! let result = ledger.list_accounts(PaginationOption::Paginated(pagination)).await?;
54//! let accounts = result.into_paginated_response();
55//! println!("Total accounts: {}, Current page: {} of {}",
56//!          accounts.total_count,
57//!          accounts.page,
58//!          accounts.total_pages);
59//! # Ok(())
60//! # }
61//! ```
62//!
63//! ## Pagination Support
64//!
65//! The library provides comprehensive pagination support for both accounts and transactions:
66//!
67//! ### Basic Pagination
68//!
69//! ```rust
70//! use accounting_core::{Ledger, PaginationOption, PaginationParams, AccountType};
71//! use accounting_core::utils::memory_storage::MemoryStorage;
72//!
73//! # #[tokio::main]
74//! # async fn main() -> Result<(), Box<dyn std::error::Error>> {
75//! let storage = MemoryStorage::new();
76//! let ledger = Ledger::new(storage);
77//!
78//! // Get first page with 10 accounts per page
79//! let pagination = PaginationParams::new(1, 10)?;
80//! let result = ledger.list_accounts(PaginationOption::Paginated(pagination)).await?;
81//! let result = result.into_paginated_response();
82//!
83//! println!("Page {} of {} (showing {} of {} total accounts)",
84//!          result.page,
85//!          result.total_pages,
86//!          result.items.len(),
87//!          result.total_count);
88//!
89//! // Check if there are more pages
90//! if result.has_next {
91//!     let next_page = PaginationParams::new(2, 10)?;
92//!     let next_result = ledger.list_accounts(PaginationOption::Paginated(next_page)).await?;
93//!     // Process next page...
94//! }
95//! # Ok(())
96//! # }
97//! ```
98//!
99//! ### Filtered Pagination
100//!
101//! ```rust
102//! use accounting_core::{Ledger, PaginationOption, PaginationParams, AccountType};
103//! use accounting_core::utils::memory_storage::MemoryStorage;
104//! use chrono::NaiveDate;
105//!
106//! # #[tokio::main]
107//! # async fn main() -> Result<(), Box<dyn std::error::Error>> {
108//! let storage = MemoryStorage::new();
109//! let ledger = Ledger::new(storage);
110//!
111//! // Get only Asset accounts with pagination
112//! let pagination = PaginationParams::new(1, 20)?;
113//! let assets = ledger.list_accounts_by_type(AccountType::Asset, PaginationOption::Paginated(pagination)).await?;
114//!
115//! // Get transactions for a specific date range with pagination
116//! let start_date = NaiveDate::from_ymd_opt(2024, 1, 1);
117//! let end_date = NaiveDate::from_ymd_opt(2024, 12, 31);
118//! let transactions = ledger.list_transactions(start_date, end_date, PaginationOption::Paginated(pagination)).await?;
119//!
120//! // Get transactions for a specific account with pagination
121//! let account_txns = ledger.list_account_transactions(
122//!     "cash",
123//!     start_date,
124//!     end_date,
125//!     PaginationOption::Paginated(pagination)
126//! ).await?;
127//! # Ok(())
128//! # }
129//! ```
130//!
131//! ### Pagination Metadata
132//!
133//! All paginated responses include comprehensive metadata for building user interfaces:
134//!
135//! ```rust
136//! # use accounting_core::{Ledger, PaginationOption, PaginationParams};
137//! # use accounting_core::utils::memory_storage::MemoryStorage;
138//! # #[tokio::main]
139//! # async fn main() -> Result<(), Box<dyn std::error::Error>> {
140//! # let storage = MemoryStorage::new();
141//! # let ledger = Ledger::new(storage);
142//! let pagination = PaginationParams::new(2, 10)?;
143//! let response = ledger.list_accounts(PaginationOption::Paginated(pagination)).await?;
144//! let result = response.into_paginated_response();
145//!
146//! // Access pagination metadata
147//! println!("Current page: {}", result.page);           // 2
148//! println!("Page size: {}", result.page_size);         // 10
149//! println!("Total items: {}", result.total_count);     // e.g., 25
150//! println!("Total pages: {}", result.total_pages);     // e.g., 3
151//! println!("Has next page: {}", result.has_next);      // true
152//! println!("Has previous page: {}", result.has_previous); // true
153//!
154//! // Use metadata to build navigation
155//! if result.has_previous {
156//!     println!("Previous page available");
157//! }
158//! if result.has_next {
159//!     println!("Next page available");
160//! }
161//! # Ok(())
162//! # }
163//! ```

165//! ## Invoicing
166//!
167//! [`sales`] builds invoices and credit notes on the shared [`document`] machinery, works out
168//! their tax under a [`TaxRegime`] and posts them as balanced journal entries.
169//!
170//! ```rust
171//! use accounting_core::{FlatRate, Invoice, LineItem, SalesAccounts, Document};
172//! use bigdecimal::BigDecimal;
173//! use chrono::NaiveDate;
174//!
175//! # fn main() -> Result<(), accounting_core::Error> {
176//! let line = LineItem::new("Consulting", BigDecimal::from(10), BigDecimal::from(100),
177//!     BigDecimal::from(20))?;
178//! let invoice = Invoice::new("INV-001", NaiveDate::from_ymd_opt(2025, 3, 1).unwrap(),
179//!     "Acme Ltd", vec![line], FlatRate::new("VAT"))?;
180//! assert_eq!(invoice.tax_breakdown()?.total, BigDecimal::from(1200));
181//!
182//! let accounts = SalesAccounts::new("1200", "4000").with_output_tax("VAT", "2200");
183//! let entries = invoice.to_entries(&accounts)?; // Dr 1200 1200, Cr 4000 1000, Cr 2200 200
184//! assert_eq!(entries.len(), 3);
185//! # Ok(())
186//! # }
187//! ```
188//!
189//! ## Reconciliation
190//!
191//! [`reconciliation::ReconciliationEngine`] matches ledger records against bank statements and
192//! payment gateway settlements, classifying every record as matched, partially matched or
193//! unmatched and suggesting counterparts for the leftovers. It is pure and synchronous; see the
194//! [`reconciliation`] module docs for the matching passes and the direction conventions.
195//!
196//! ```rust
197//! use accounting_core::reconciliation::{
198//!     ExternalSource, ExternalTransaction, LedgerTransaction, ReconciliationEngine,
199//! };
200//! use accounting_core::EntryType;
201//! use bigdecimal::BigDecimal;
202//! use chrono::NaiveDate;
203//!
204//! let date = NaiveDate::from_ymd_opt(2024, 11, 15).unwrap();
205//! let source = ExternalSource::BankStatement {
206//!     bank_name: "SBI".to_string(),
207//!     account_number: "12345678901".to_string(),
208//! };
209//!
210//! let ledger = vec![LedgerTransaction::new(
211//!     "txn-1".to_string(),
212//!     date,
213//!     BigDecimal::from(1000),
214//!     "Payment from Acme Ltd".to_string(),
215//!     EntryType::Debit,
216//!     "bank".to_string(),
217//! )];
218//! let external = vec![ExternalTransaction::new(
219//!     "stmt-1".to_string(),
220//!     date,
221//!     BigDecimal::from(1000),
222//!     "NEFT/ACME LTD/0012".to_string(),
223//!     EntryType::Debit,
224//!     source.clone(),
225//! )];
226//!
227//! let report = ReconciliationEngine::default().reconcile(&ledger, &external, source);
228//! assert_eq!(report.matched_count, 1);
229//! println!("Balance difference: {}", report.summary.difference);
230//! ```
