1//! HSN/SAC master data and default GST rates
2//!
3//! The master is compiled into the crate from two files:
4//!
5//! - `src/gst/data/hsn_master.json`: the codes whose default rate has been checked against the GST
6//!   schedule named by [`HsnMaster::schedule`], with their rates. The notifications and references
7//!   it was checked against are listed under `sources` there. Always compiled in with `india`.
8//! - `src/gst/data/hsn_codes.json`: every 4-, 6- and 8-digit HSN code and 6-digit SAC code with
9//!   its description, from the GST portal's HSN/SAC master (built by
10//!   `scripts/build_hsn_codes.py`, which names the source and the day it was downloaded). It adds
11//!   about 1.8 MB to the binary, so it is only compiled in with the `hsn-catalog` feature.
12//!
13//! Without `hsn-catalog` the embedded master holds only the checked codes, so
14//! [`HsnMaster::lists_every_code`] is `false` and validation does not warn about codes it lacks.
15//!
16//! The portal file has no rates, so most codes have none: [`HsnMaster::default_rate`] falls back
17//! to the nearest heading that has one. Rates are defaults: some goods and services carry
18//! conditional rates (price thresholds, input tax credit options), so callers can always pass an
19//! explicit rate instead.
20
21use bigdecimal::BigDecimal;
22use chrono::NaiveDate;
23use serde::{Deserialize, Serialize};
24use std::collections::HashMap;

27/// The embedded master, parsed once on first use
28///
29/// The JSON is compiled into the crate, so a parse failure is a build defect rather than a
30/// runtime condition; `test_embedded_master_parses` catches it in CI.
31static MASTER: LazyLock<HsnMaster> =
32    LazyLock::new(|| HsnMaster::from_parts(embedded_rates(), embedded_codes()));

34/// The embedded checked rates
35#[allow(clippy::expect_used)]
36fn embedded_rates() -> RawRates {
37    serde_json::from_str(include_str!("../data/hsn_master.json"))
38        .expect("embedded HSN/SAC rates are valid JSON")
39}

57/// Whether a code classifies goods or services
58#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
59#[serde(rename_all = "lowercase")]
60pub enum HsnSacKind {
61    /// Harmonized System of Nomenclature code (goods)
62    Hsn,
63    /// Services Accounting Code (services)
64    Sac,
65}
66
67/// Chapter of the classification that holds every SAC code; all other chapters are goods
68const SAC_CHAPTER: &str = "99";
69
70impl HsnSacKind {
71    /// Whether a code classifies goods or services, from its chapter: SAC codes start with `99`
72    ///
73    /// The code's shape is not checked; see [`HsnMaster::lookup`] for well-formed codes.
74    #[must_use]
75    pub fn of_code(code: &str) -> Self {
76        if code.starts_with(SAC_CHAPTER) {
77            Self::Sac
78        } else {
79            Self::Hsn
80        }
81    }
82}

84/// One code in the master
85#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
86pub struct HsnSacEntry {
87    /// HSN or SAC code: 4, 6 or 8 digits
88    pub code: String,
89    /// Goods or services
90    pub kind: HsnSacKind,
91    /// Short description of what the code covers
92    pub description: String,
93    /// Default total GST rate as a percentage (e.g. 18 for 18%), when it has been checked for
94    /// this code; `None` for most codes, which the portal's list gives no rate
95    pub gst_rate: Option<BigDecimal>,
96}

98/// Lookup table of HSN/SAC codes and their default GST rates
99///
100/// Deserializes from the shape of `hsn_master.json` (`schedule`, `effective_from` and `entries`
101/// with rates), so a caller can check documents against a master of its own.
102#[derive(Debug, Clone, Deserialize)]
103#[serde(from = "RawRates")]
104pub struct HsnMaster {
105    schedule: String,
106    effective_from: NaiveDate,
107    /// Ordered by code
108    entries: Vec<HsnSacEntry>,
109    index: HashMap<String, usize>,
110    /// Each entry's description in lower case, in `entries` order, for [`HsnMaster::search`]
111    search_text: Vec<String>,
112    /// Whether every valid code is listed, so a code missing from it is unknown
113    lists_every_code: bool,
114}
115
116/// On-disk shape of the checked rates
117#[derive(Deserialize)]
118struct RawRates {
119    schedule: String,
120    effective_from: NaiveDate,
121    entries: Vec<RawRate>,
122}
123
124/// One checked code: its description and default rate
125#[derive(Deserialize)]
126struct RawRate {
127    code: String,
128    description: String,
129    gst_rate: BigDecimal,
130}
131
132/// On-disk shape of the portal's code list: `[code, description]` pairs
133#[derive(Deserialize)]
134struct RawCodes {
135    codes: Vec<(String, String)>,
136}
137
138/// A caller's own master is taken as the whole list of codes it accepts
139impl From<RawRates> for HsnMaster {
140    fn from(rates: RawRates) -> Self {
141        let mut master = Self::from_parts(rates, None);
142        master.lists_every_code = true;
143        master
144    }
145}
146
147/// Headings a code falls back to, longest first: the code itself, then its 6- and 4-digit headings
148const HEADING_LENGTHS: [usize; 2] = [6, 4];
149
150impl HsnMaster {
151    /// The master over `codes`, with the checked `rates` (and their wording) laid over them. A
152    /// checked code missing from the list is added. Without `codes` it holds only the checked
153    /// codes and does not list every code.
154    fn from_parts(rates: RawRates, codes: Option<RawCodes>) -> Self {
155        let lists_every_code = codes.is_some();
156        let checked: HashMap<String, RawRate> = rates
157            .entries
158            .into_iter()
159            .map(|rate| (rate.code.clone(), rate))
160            .collect();
161        let listed: HashMap<String, String> = codes
162            .map(|codes| codes.codes.into_iter().collect())
163            .unwrap_or_default();
164        let mut all: Vec<&String> = listed.keys().chain(checked.keys()).collect();
165        all.sort();
166        all.dedup();
167        let entries: Vec<HsnSacEntry> = all
168            .into_iter()
169            .map(|code| match checked.get(code) {
170                Some(rate) => entry(code, &rate.description, Some(rate.gst_rate.clone())),
171                None => entry(code, listed.get(code).map_or("", String::as_str), None),
172            })
173            .collect();
174        Self::indexed(
175            rates.schedule,
176            rates.effective_from,
177            entries,
178            lists_every_code,
179        )
180    }
181
182    fn indexed(
183        schedule: String,
184        effective_from: NaiveDate,
185        entries: Vec<HsnSacEntry>,
186        lists_every_code: bool,
187    ) -> Self {
188        let index = entries
189            .iter()
190            .enumerate()
191            .map(|(i, e)| (e.code.clone(), i))
192            .collect();
193        let search_text = entries
194            .iter()
195            .map(|e| e.description.to_lowercase())
196            .collect();
197        Self {
198            schedule,
199            effective_from,
200            entries,
201            index,
202            search_text,
203            lists_every_code,
204        }
205    }
206
207    /// The master embedded in the crate
208    #[must_use]
209    pub fn global() -> &'static HsnMaster {
210        &MASTER
211    }
212
213    /// Name of the GST rate schedule the default rates follow
214    #[must_use]
215    pub fn schedule(&self) -> &str {
216        &self.schedule
217    }
218
219    /// Date the rate schedule took effect
220    #[must_use]
221    pub fn effective_from(&self) -> NaiveDate {
222        self.effective_from
223    }
224
225    /// Whether the master lists every valid code, so that a code it lacks is unknown
226    ///
227    /// `true` for a master deserialized by the caller, and for [`HsnMaster::global`] with the
228    /// `hsn-catalog` feature. Without that feature the embedded master holds only the codes with
229    /// checked rates, and this is `false`.
230    #[must_use]
231    pub fn lists_every_code(&self) -> bool {
232        self.lists_every_code
233    }
234
235    /// All entries, ordered by code
236    #[must_use]
237    pub fn entries(&self) -> &[HsnSacEntry] {
238        &self.entries
239    }
240
241    /// The code and the headings it falls back to that the master holds, longest first
242    fn candidates(&self, code: &str) -> Vec<&HsnSacEntry> {
243        std::iter::once(code.len())
244            .chain(HEADING_LENGTHS.into_iter().filter(|&len| len < code.len()))
245            .filter_map(|len| self.index.get(&code[..len]))
246            .map(|&i| &self.entries[i])
247            .collect()
248    }
249
250    /// Find the entry for a code
251    ///
252    /// Tries the exact code first, then its 6- and 4-digit headings, so an 8-digit tariff item
253    /// such as `84713010` resolves to `847130` or `8471` when the master lacks it. A code is known
254    /// when it or one of its headings is. Malformed codes return `None`.
255    #[must_use]
256    pub fn lookup(&self, code: &str) -> Option<&HsnSacEntry> {
257        if !is_valid_hsn_sac(code) {
258            return None;
259        }
260        self.candidates(code).into_iter().next()
261    }
262
263    /// Default GST rate for a code: that of the code, or of the nearest heading over it that has
264    /// one. `None` when neither the code nor its headings has a checked rate.
265    #[must_use]
266    pub fn default_rate(&self, code: &str) -> Option<BigDecimal> {
267        if !is_valid_hsn_sac(code) {
268            return None;
269        }
270        self.candidates(code)
271            .into_iter()
272            .find_map(|entry| entry.gst_rate.clone())
273    }
274
275    /// Up to `limit` entries matching `query`, for picking a code
276    ///
277    /// Digits match codes starting with them; anything else matches descriptions containing every
278    /// word of it, ignoring case. Headings come before the tariff items under them (shorter codes
279    /// first), then codes in order. A blank query matches nothing.
280    #[must_use]
281    pub fn search(&self, query: &str, limit: usize) -> Vec<&HsnSacEntry> {
282        let query = query.trim();
283        if query.is_empty() {
284            return Vec::new();
285        }
286        let words: Vec<String> = query
287            .to_lowercase()
288            .split_whitespace()
289            .map(String::from)
290            .collect();
291        let by_code = query.bytes().all(|b| b.is_ascii_digit());
292        let mut found: Vec<&HsnSacEntry> = self
293            .entries
294            .iter()
295            .zip(&self.search_text)
296            .filter(|(entry, text)| {
297                if by_code {
298                    entry.code.starts_with(query)
299                } else {
300                    words.iter().all(|word| text.contains(word.as_str()))
301                }
302            })
303            .map(|(entry, _)| entry)
304            .collect();
305        // Stable: entries are already in code order.
306        found.sort_by_key(|entry| entry.code.len());
307        found.truncate(limit);
308        found
309    }
310}

312/// An entry for `code`, its kind read from its chapter
313fn entry(code: &str, description: &str, gst_rate: Option<BigDecimal>) -> HsnSacEntry {
314    HsnSacEntry {
315        code: code.to_string(),
316        kind: HsnSacKind::of_code(code),
317        description: description.to_string(),
318        gst_rate,
319    }
320}

322/// Whether a code has the shape of an HSN/SAC code: 4, 6 or 8 ASCII digits
323pub(crate) fn is_valid_hsn_sac(code: &str) -> bool {
324    matches!(code.len(), 4 | 6 | 8) && code.bytes().all(|b| b.is_ascii_digit())
