1//! Ledger posting for a GST invoice
2//!
3//! An issued invoice is one journal entry: the buyer owes the total, the taxable value is sales
4//! revenue, and each tax component is owed to the government. [`posting_legs`] turns a
5//! [`GstBreakdown`] into those legs and [`GstInvoice::to_entries`] maps them onto accounts. The
6//! legs always balance, because the breakdown's total is its taxable value plus its tax.
7//!
8//! A [`GstCreditNote`] posts the same legs on the opposite sides, reversing the part of the invoice
9//! it credits: [`GstCreditNote::to_entries`].
10//!
11//! A bill of supply charges no tax, so it posts only the receivable and revenue. A composition
12//! seller's also posts the tax it owes on its turnover
13//! ([`GstDocument::composition_tax`]): debited to an expense and credited to a payable, from
14//! [`InvoiceAccounts::with_composition_tax`].
15

28/// Memo of the entries that post a composition seller's tax on its turnover
29pub const COMPOSITION_TAX_MEMO: &str = "Composition tax";

31/// Every leg a GST invoice can post, debit first: receivable, revenue, then CGST, SGST and IGST
32#[must_use]
33pub fn gst_legs() -> [PostingLeg; 5] {
34    [
35        PostingLeg::Receivable,
36        PostingLeg::Revenue,
37        PostingLeg::output_tax(CGST_CODE),
38        PostingLeg::output_tax(SGST_CODE),
39        PostingLeg::output_tax(IGST_CODE),
40    ]
41}
42
43/// Legs of the journal entry for `breakdown`, leaving out those with a zero amount
44///
45/// An invoice only ever carries CGST + SGST or IGST, never both. See
46/// [`sales_legs`].
47#[must_use]
48pub fn posting_legs(breakdown: &GstBreakdown) -> Vec<(PostingLeg, BigDecimal)> {
49    sales_legs(&TaxBreakdown::from(breakdown))
50}

52/// Why an invoice could not be posted to the ledger
53#[derive(Debug, thiserror::Error)]
54pub enum PostingError {
55    /// The invoice breaks compliance rules; carries the error-severity issues
56    #[error("invoice is not compliant ({} error(s)), first: {}", .0.len(), .0.first().map_or(String::new(), ToString::to_string))]
57    NotCompliant(Vec<ComplianceIssue>),
58    /// The invoice total is zero, so there is nothing to post
59    #[error("invoice total is zero, so there is nothing to post")]
60    NothingToPost,
61    /// The invoice's tax could not be calculated
62    #[error(transparent)]
63    Invoice(#[from] InvoiceError),
64    /// The breakdown could not be mapped onto the accounts
65    #[error(transparent)]
66    Posting(document::PostingError),
67    /// A composition seller's document owes composition tax, but the accounts name no expense
68    /// and payable account for it
69    #[error("composition tax of {amount} needs an expense and a payable account")]
70    MissingCompositionAccounts {
71        /// The composition tax to post
72        amount: BigDecimal,
73    },
74}
75
76impl From<document::PostingError> for PostingError {
77    /// [`PostingError::NothingToPost`] for a zero total, [`PostingError::Posting`] otherwise
78    fn from(error: document::PostingError) -> Self {
79        match error {
80            document::PostingError::NothingToPost => Self::NothingToPost,
81            other @ document::PostingError::NoTaxAccount { .. } => Self::Posting(other),
82        }
83    }
84}

86/// Accounts a GST invoice posts to, one per leg in [`gst_legs`]
87#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
88pub struct InvoiceAccounts {
89    /// Debited with the invoice total (accounts receivable or cash)
90    pub receivable: String,
91    /// Credited with the taxable value
92    pub revenue: String,
93    /// Credited with CGST
94    pub cgst_output: String,
95    /// Credited with SGST
96    pub sgst_output: String,
97    /// Credited with IGST
98    pub igst_output: String,
99    /// Debited with a composition seller's tax on its turnover; needed only to post a
100    /// composition seller's documents
101    #[serde(default, skip_serializing_if = "Option::is_none")]
102    pub composition_tax_expense: Option<String>,
103    /// Credited with the composition tax the seller owes the government; needed with
104    /// [`InvoiceAccounts::composition_tax_expense`]
105    #[serde(default, skip_serializing_if = "Option::is_none")]
106    pub composition_tax_payable: Option<String>,
107}

109impl InvoiceAccounts {
110    /// Name the account for each leg
111    pub fn new(
112        receivable: impl Into<String>,
113        revenue: impl Into<String>,
114        cgst_output: impl Into<String>,
115        sgst_output: impl Into<String>,
116        igst_output: impl Into<String>,
117    ) -> Self {
118        Self {
119            receivable: receivable.into(),
120            revenue: revenue.into(),
121            cgst_output: cgst_output.into(),
122            sgst_output: sgst_output.into(),
123            igst_output: igst_output.into(),
124            composition_tax_expense: None,
125            composition_tax_payable: None,
126        }
127    }
128
129    /// These accounts with the expense and payable accounts a composition seller's tax on its
130    /// turnover posts to
131    #[must_use]
132    pub fn with_composition_tax(
133        self,
134        expense: impl Into<String>,
135        payable: impl Into<String>,
136    ) -> Self {
137        Self {
138            composition_tax_expense: Some(expense.into()),
139            composition_tax_payable: Some(payable.into()),
140            ..self
141        }
142    }
143
144    /// Account id `leg` posts to; `None` for a tax other than CGST, SGST or IGST
145    #[must_use]
146    pub fn account_for(&self, leg: &PostingLeg) -> Option<&str> {
147        match leg {
148            PostingLeg::Receivable => Some(&self.receivable),
149            PostingLeg::Revenue => Some(&self.revenue),
150            PostingLeg::OutputTax(code) if code == CGST_CODE => Some(&self.cgst_output),
151            PostingLeg::OutputTax(code) if code == SGST_CODE => Some(&self.sgst_output),
152            PostingLeg::OutputTax(code) if code == IGST_CODE => Some(&self.igst_output),
153            PostingLeg::OutputTax(_) => None,
154        }
155    }
156}

158impl From<&InvoiceAccounts> for SalesAccounts {
159    fn from(accounts: &InvoiceAccounts) -> Self {
160        SalesAccounts::new(accounts.receivable.clone(), accounts.revenue.clone())
161            .with_output_tax(CGST_CODE, accounts.cgst_output.clone())
162            .with_output_tax(SGST_CODE, accounts.sgst_output.clone())
163            .with_output_tax(IGST_CODE, accounts.igst_output.clone())
164    }
165}

167impl GstInvoice {
168    /// Journal entries that record this invoice in the ledger
169    ///
170    /// The invoice must pass the error-severity compliance rules of
171    /// [`validate_invoice`](super::validate_invoice), checked as of its own date: every error rule
172    /// applies except the future-date check, which depends on when the caller posts it. Warnings
173    /// don't block the posting. The entries balance: the receivable debit equals the sales and tax
174    /// credits. Pass them to [`TransactionBuilder::entry`](crate::ledger::TransactionBuilder::entry)
175    /// to build the transaction.
176    ///
177    /// # Errors
178    ///
179    /// [`PostingError::NotCompliant`] with the error-severity issues if the invoice breaks a
180    /// compliance rule, [`PostingError::NothingToPost`] if its total is zero,
181    /// [`PostingError::MissingCompositionAccounts`] if a composition seller's invoice owes tax
182    /// but `accounts` name no composition tax accounts, or [`PostingError::Invoice`] for any
183    /// error from [`GstInvoice::breakdown`].
184    ///
185    /// # Example
186    ///
187    /// ```
188    /// use accounting_core::gst::invoice::{GstInvoice, GstLineItem, Gstin, InvoiceAccounts};
189    /// use accounting_core::EntryType;
190    /// use bigdecimal::BigDecimal;
191    /// use chrono::NaiveDate;
192    ///
193    /// # fn main() -> Result<(), accounting_core::Error> {
194    /// let item = GstLineItem::new("998314", "IT consulting", BigDecimal::from(1),
195    ///     BigDecimal::from(1000), BigDecimal::from(18))?;
196    /// let invoice = GstInvoice::new("INV-001", NaiveDate::from_ymd_opt(2024, 11, 15).unwrap(),
197    ///     Gstin::parse("27AAPFU0939F1ZV")?, Gstin::parse("29AAPFU0939F1ZR")?, vec![item])?;
198    ///
199    /// let accounts = InvoiceAccounts::new("receivable", "sales", "cgst", "sgst", "igst");
200    /// let entries = invoice.to_entries(&accounts)?;
201    ///
202    /// // Inter-state: receivable 1180 = sales 1000 + IGST 180
203    /// assert_eq!(entries.len(), 3);
204    /// assert_eq!(entries[0].entry_type, EntryType::Debit);
205    /// assert_eq!(entries[0].amount, BigDecimal::from(1180));
206    /// assert_eq!(entries[2].account_id, "igst");
207    /// # Ok(())
208    /// # }
209    /// ```
210    pub fn to_entries(&self, accounts: &InvoiceAccounts) -> Result<Vec<Entry>, PostingError> {
211        let errors = compliance_errors(self, self.invoice_date);
212        document_entries(self, errors, accounts, PostingLeg::side)
213    }
214}

216impl GstCreditNote {
217    /// Journal entries that record this credit note in the ledger
218    ///
219    /// The legs are an invoice's, on the opposite sides: sales and the output tax are debited,
220    /// and the receivable is credited with the note's total. Pass a sales returns account as
221    /// `accounts.revenue` to keep returns apart from sales. The note must pass the error-severity
222    /// rules of [`validate_credit_note`](super::validate_credit_note) as of its own date.
223    ///
224    /// # Errors
225    ///
226    /// As [`GstInvoice::to_entries`].
227    ///
228    /// # Example
229    ///
230    /// ```
231    /// use accounting_core::gst::invoice::{GstCreditNote, GstInvoice, GstLineItem, Gstin, InvoiceAccounts};
232    /// use accounting_core::EntryType;
233    /// use bigdecimal::BigDecimal;
234    /// use chrono::NaiveDate;
235    ///
236    /// # fn main() -> Result<(), accounting_core::Error> {
237    /// let item = GstLineItem::new("998314", "IT consulting", BigDecimal::from(1),
238    ///     BigDecimal::from(1000), BigDecimal::from(18))?;
239    /// let invoice = GstInvoice::new("INV-001", NaiveDate::from_ymd_opt(2024, 11, 15).unwrap(),
240    ///     Gstin::parse("27AAPFU0939F1ZV")?, Gstin::parse("29AAPFU0939F1ZR")?, vec![item.clone()])?;
241    /// let note = GstCreditNote::new(&invoice, "CN-001", NaiveDate::from_ymd_opt(2024, 11, 20).unwrap(),
242    ///     vec![item])?;
243    ///
244    /// let accounts = InvoiceAccounts::new("receivable", "sales returns", "cgst", "sgst", "igst");
245    /// let entries = note.to_entries(&accounts)?;
246    ///
247    /// // Inter-state: sales returns 1000 + IGST 180 = receivable 1180, reversed
248    /// assert_eq!(entries[0].entry_type, EntryType::Credit);
249    /// assert_eq!(entries[0].amount, BigDecimal::from(1180));
250    /// assert_eq!(entries[1].entry_type, EntryType::Debit);
251    /// assert_eq!(entries[2].account_id, "igst");
252    /// # Ok(())
253    /// # }
254    /// ```
255    pub fn to_entries(&self, accounts: &InvoiceAccounts) -> Result<Vec<Entry>, PostingError> {
256        let errors = credit_note_errors(self, self.note_date);
257        document_entries(self, errors, accounts, |leg| leg.side().opposite())
258    }
259}

261/// The entries for `document`'s legs, each on the side `side` gives it, unless it has compliance
262/// `errors` or nothing to post
263fn document_entries(
264    document: &dyn GstDocument,
265    errors: Vec<ComplianceIssue>,
266    accounts: &InvoiceAccounts,
267    side: impl Fn(&PostingLeg) -> EntryType,
268) -> Result<Vec<Entry>, PostingError> {
269    if !errors.is_empty() {
270        return Err(PostingError::NotCompliant(errors));
271    }
272    let breakdown = document.tax_breakdown()?;
273    let sales = sales_entries(&breakdown, &SalesAccounts::from(accounts), &side)?;
274    let composition = match document.composition_tax()? {
275        Some(amount) => composition_entries(amount, accounts, &side)?,
276        None => Vec::new(),
277    };
278    Ok(sales.into_iter().chain(composition).collect())
279}

281/// The composition tax pair: the expense on the receivable's side and the payable on the
282/// revenue's, so a credit note, which flips those, reverses it too; nothing for a zero amount
283fn composition_entries(
284    amount: BigDecimal,
285    accounts: &InvoiceAccounts,
286    side: impl Fn(&PostingLeg) -> EntryType,
287) -> Result<Vec<Entry>, PostingError> {
288    if amount.is_zero() {
289        return Ok(Vec::new());
290    }
291    let (Some(expense), Some(payable)) = (
292        &accounts.composition_tax_expense,
293        &accounts.composition_tax_payable,
294    ) else {
295        return Err(PostingError::MissingCompositionAccounts { amount });
296    };
297    let memo = || Some(COMPOSITION_TAX_MEMO.to_string());
298    Ok(vec![
299        Entry::new(
300            expense,
301            side(&PostingLeg::Receivable),
302            amount.clone(),
303            memo(),
304        ),
305        Entry::new(payable, side(&PostingLeg::Revenue), amount, memo()),
306    ])
