1//! How the buyer can pay an invoice: bank details and a UPI payment link
2//!
3//! [`PaymentDetails`] is applied to a printed invoice with
4//! [`InvoicePrint::with_payment`](super::print::InvoicePrint::with_payment). When it names a UPI
5//! ID, the print model also carries a [`upi_uri`] that the PDF renderer draws as a QR code any UPI
6//! app can scan. Everything here is pure.
7//!
8//! # Example
9//!
10//! ```
11//! use accounting_core::gst::invoice::upi_uri;
12//! use bigdecimal::BigDecimal;
13//!
14//! let uri = upi_uri("acme@okhdfc", "Acme Traders", &BigDecimal::from(1180), "INV0002");
15//! assert_eq!(uri, "upi://pay?pa=acme%40okhdfc&pn=Acme%20Traders&am=1180.00&cu=INR&tn=INV0002");
16//! ```
17
18use crate::gst::tax::{round_to_paise, PAISE_SCALE};
19use bigdecimal::BigDecimal;
20use serde::{Deserialize, Serialize};
21
22/// Currency code a UPI payment link asks for
23pub const UPI_CURRENCY: &str = "INR";
24
25/// Where the buyer can pay: bank account and UPI ID, each optional
26///
27/// A blank field counts as unset. With no field set, nothing is printed.
28#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
29pub struct PaymentDetails {
30    /// Account holder the payment goes to; the seller's name when unset
31    #[serde(default, skip_serializing_if = "Option::is_none")]
32    pub payee_name: Option<String>,
33    /// Name of the bank
34    #[serde(default, skip_serializing_if = "Option::is_none")]
35    pub bank_name: Option<String>,
36    /// Bank account number
37    #[serde(default, skip_serializing_if = "Option::is_none")]
38    pub account_number: Option<String>,
39    /// IFSC of the bank branch
40    #[serde(default, skip_serializing_if = "Option::is_none")]
41    pub ifsc: Option<String>,
42    /// UPI ID (virtual payment address), such as `acme@okhdfc`
43    #[serde(default, skip_serializing_if = "Option::is_none")]
44    pub upi_id: Option<String>,
45    /// Amount the UPI QR code asks for, such as the balance still owed on a part-paid invoice;
46    /// the invoice total when unset. At zero or below no QR code is printed
47    #[serde(default, skip_serializing_if = "Option::is_none")]
48    pub amount_due: Option<BigDecimal>,
49}
50
51impl PaymentDetails {
52    /// The details with every field trimmed and blank fields unset
53    #[must_use]
54    pub fn trimmed(self) -> Self {
55        Self {
56            payee_name: trim(self.payee_name),
57            bank_name: trim(self.bank_name),
58            account_number: trim(self.account_number),
59            ifsc: trim(self.ifsc),
60            upi_id: trim(self.upi_id),
61            amount_due: self.amount_due,
62        }
63    }
64
65    /// Whether no bank or UPI detail is set, ignoring blank fields, the payee name and the
66    /// amount due
67    #[must_use]
68    pub fn is_empty(&self) -> bool {
69        [
70            &self.bank_name,
71            &self.account_number,
72            &self.ifsc,
73            &self.upi_id,
74        ]
75        .into_iter()
76        .all(|field| field.as_deref().is_none_or(|text| text.trim().is_empty()))
77    }
78}
79
80/// `field` trimmed, or `None` when it is blank
81fn trim(field: Option<String>) -> Option<String> {
82    field.as_deref().and_then(non_blank)
83}
84
85/// `text` trimmed, or `None` when it is blank
86pub(super) fn non_blank(text: &str) -> Option<String> {
87    let text = text.trim();
88    (!text.is_empty()).then(|| text.to_string())
89}
90
91/// A UPI payment link asking `upi_id` for `amount` rupees, with `note` as the transaction note
92///
93/// The amount is rounded to the paisa and written with two decimals. Every value is
94/// percent-encoded, so the link is safe to put in a QR code as is.
95#[must_use]
96pub fn upi_uri(
97    upi_id: impl Into<String>,
98    payee_name: impl Into<String>,
99    amount: &BigDecimal,
100    note: impl Into<String>,
101) -> String {
102    let amount = round_to_paise(amount).with_scale(PAISE_SCALE);
103    format!(
104        "upi://pay?pa={}&pn={}&am={}&cu={UPI_CURRENCY}&tn={}",
105        percent_encode(&upi_id.into()),
106        percent_encode(&payee_name.into()),
107        amount.to_plain_string(),
108        percent_encode(&note.into()),
109    )
110}
111
112/// `text` with every byte other than an RFC 3986 unreserved character written as `%XX`
113fn percent_encode(text: &str) -> String {
114    text.bytes()
115        .map(|byte| {
116            if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
117                char::from(byte).to_string()
118            } else {
119                format!("%{byte:02X}")
120            }
121        })
122        .collect()
123}
