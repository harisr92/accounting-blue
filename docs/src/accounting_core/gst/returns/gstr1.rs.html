1//! GSTR-1: the return of outward supplies
2//!
3//! [`Gstr1Return::build`] aggregates a filer's invoices and credit notes for one [`ReturnPeriod`]
4//! into the
5//! sections of GSTR-1 this crate can fill: Table 4A (B2B supplies, grouped by buyer GSTIN with
6//! one item per rate), Table 5 (B2CL: large inter-state supplies to unregistered buyers,
7//! grouped by place of supply), Table 7 (B2CS: every other supply to unregistered buyers,
8//! summed per place of supply and rate), Table 8 (nil-rated, exempt and non-GST lines, by
9//! supply type), Table 9B (credit notes, see [`super::cdn`]), Table 12 (the HSN/SAC summary,
10//! grouped by code and rate) and Table 13 (documents issued).
11//! The builder is pure; [`Gstr1Return::to_json`] writes the result in the GST portal's
12//! offline-tool schema, with its short keys and amounts as JSON numbers rounded to paise, ready
13//! to upload.
14//!
15//! Table 12 is split into the two tabs the portal has used since 2025: rows for B2B supplies go
16//! under `hsn.hsn_b2b`, and rows for B2CL and B2CS supplies under `hsn.hsn_b2c`. Which table an
17//! invoice belongs to is [`GstInvoice::supply_kind`]. Table 7 reports only taxable supplies other
18//! than through an e-commerce operator (`typ` `OE`).
19//!
20//! A line that charges no GST ([`GstLineItem::charges_gst`]) goes to Table 8, never to Table
21//! 4A, 5 or 7, and stays in the HSN summary at rate 0. Its [`SupplyTreatment`] picks the
22//! column: a taxable line at 0% is nil-rated, the others are exempt or non-GST. A B2B or B2CL
23//! invoice keeps its whole value but lists only its taxed rates, and one with no taxed line
24//! appears only in Table 8.
25//!
26//! A [`GstCreditNote`] is reported under the supply kind of the invoice it corrects. Against a B2B
27//! or B2CL invoice it goes to Table 9B; against a B2CS invoice it is subtracted from the Table 7
28//! row for its place of supply and rate, which may then be negative. Either way its lines are
29//! subtracted from Table 8 and from the HSN summary tab of its supply kind, and Table 13 lists the
30//! notes as a series of their own.

92    /// Table 9B: credit notes to registered buyers, one entry per buyer GSTIN
93    #[serde(rename = "cdnr", skip_serializing_if = "Vec::is_empty")]
94    pub cdnr: Vec<CdnrParty>,
95    /// Table 9B: credit notes against B2CL invoices to unregistered buyers
96    #[serde(rename = "cdnur", skip_serializing_if = "Vec::is_empty")]
97    pub cdnur: Vec<CdnurNote>,
98    /// Table 12: HSN/SAC summary
99    #[serde(rename = "hsn", skip_serializing_if = "HsnSummary::is_empty")]
100    pub hsn: HsnSummary,
101    /// Table 13: documents issued in the period
102    #[serde(rename = "doc_issue", skip_serializing_if = "DocIssue::is_empty")]

104}

309impl B2csRow {
310    fn new(place_of_supply: &str, is_inter_state: bool, rate: BigDecimal, b: GstBreakdown) -> Self {
311        let (supply_type, igst, cgst, sgst) = if is_inter_state {
312            (SupplyType::Inter, Some(b.igst), None, None)
313        } else {
314            (SupplyType::Intra, None, Some(b.cgst), Some(b.sgst))
315        };
316        Self {
317            supply_type,
318            rate,
319            b2cs_type: B2csType::OtherThanEcommerce,
320            place_of_supply: place_of_supply.to_string(),
321            taxable_value: b.taxable_value,
322            igst,
323            cgst,
324            sgst,
325            cess: BigDecimal::zero(),
326        }
327    }
328}

535/// Why a GSTR-1 return could not be built or written
536#[derive(Debug, thiserror::Error)]
537pub enum Gstr1Error {
538    /// The month is not 1-12, or the year is outside the GST era
539    #[error("invalid return period: month {month} of {year}")]
540    InvalidPeriod {
541        /// Year given
542        year: i32,
543        /// Month given
544        month: u32,
545    },
546    /// The invoice was issued by someone other than the filer
547    #[error("invoice {invoice_number} was issued by {seller}, not the filer")]
548    SellerMismatch {
549        /// The invoice number
550        invoice_number: String,
551        /// GSTIN of its seller
552        seller: Gstin,
553    },
554    /// The invoice is dated outside the return period
555    #[error("invoice {invoice_number} dated {date} is outside the period {period}")]
556    OutsidePeriod {
557        /// The invoice number
558        invoice_number: String,
559        /// Its date
560        date: NaiveDate,
561        /// The period being filed
562        period: ReturnPeriod,
563    },
564    /// Two invoices share a number, compared case-insensitively
565    #[error("invoice number {0} appears more than once")]
566    DuplicateInvoiceNumber(String),
567    /// The invoice breaks an error-severity compliance rule
568    #[error("invoice {invoice_number} is not compliant: {} error(s)", issues.len())]
569    NotCompliant {
570        /// The invoice number
571        invoice_number: String,
572        /// The error-severity issues found
573        issues: Vec<ComplianceIssue>,
574    },
575    /// A credit note can't go in the return
576    #[error("credit note {note_number} {reason}")]
577    InvalidCreditNote {
578        /// The note number
579        note_number: String,
580        /// The first rule it breaks
581        reason: DocumentProblem,
582    },
583    /// An invoice's tax could not be computed
584    #[error(transparent)]
585    Invoice(#[from] InvoiceError),
586    /// The return could not be written as JSON
587    #[error("could not write GSTR-1 JSON: {0}")]
588    Json(#[from] serde_json::Error),
589}
590
591impl Gstr1Error {
592    /// The error for the invoice numbered `invoice_number` that has `problem`
593    fn for_invoice(invoice_number: String, problem: DocumentProblem) -> Self {
594        match problem {
595            DocumentProblem::SellerMismatch(seller) => Self::SellerMismatch {
596                invoice_number,
597                seller,
598            },
599            DocumentProblem::OutsidePeriod { date, period } => Self::OutsidePeriod {
600                invoice_number,
601                date,
602                period,
603            },
604            DocumentProblem::DuplicateNumber => Self::DuplicateInvoiceNumber(invoice_number),
605            DocumentProblem::NotCompliant(issues) => Self::NotCompliant {
606                invoice_number,
607                issues,
608            },
609        }
610    }
611}

613/// Why an invoice or credit note can't go in the return
614#[derive(Debug, thiserror::Error)]
615pub enum DocumentProblem {
616    /// It was issued by someone other than the filer
617    #[error("was issued by {0}, not the filer")]
618    SellerMismatch(Gstin),
619    /// It is dated outside the return period
620    #[error("dated {date} is outside the period {period}")]
621    OutsidePeriod {
622        /// Its date
623        date: NaiveDate,
624        /// The period being filed
625        period: ReturnPeriod,
626    },
627    /// Another document of its kind has the same number, compared case-insensitively
628    #[error("appears more than once")]
629    DuplicateNumber,
630    /// It breaks an error-severity compliance rule; carries the issues
631    #[error("is not compliant: {} error(s)", .0.len())]
632    NotCompliant(Vec<ComplianceIssue>),
633}

635/// A document with the tax breakdown of each of its lines, their sum and the supply kind it is
636/// reported under, computed once
637///
638/// The amounts are signed by their effect on the period's supplies: a credit note's are negative,
639/// so summing invoices and notes together nets them.
640pub(super) struct PricedDocument<'a> {
641    pub(super) document: &'a dyn GstDocument,
642    pub(super) lines: Vec<GstBreakdown>,
643    pub(super) total: GstBreakdown,
644    pub(super) kind: SupplyKind,
645    reduces_supply: bool,
646}
647
648impl<'a> PricedDocument<'a> {
649    /// Price an invoice's lines and classify it by supply kind
650    fn invoice(invoice: &'a GstInvoice) -> Result<Self, InvoiceError> {
651        let lines = invoice.line_breakdowns()?;
652        let total: GstBreakdown = lines.iter().sum();
653        Ok(Self {
654            document: invoice,
655            kind: supply_kind_for(invoice, &total.total),
656            lines,
657            total,
658            reduces_supply: false,
659        })
660    }
661
662    /// Price a credit note's lines as negative amounts, under the supply kind of its invoice
663    fn credit_note(note: &'a GstCreditNote) -> Result<Self, InvoiceError> {
664        let lines: Vec<_> = note
665            .line_breakdowns()?
666            .iter()
667            .map(GstBreakdown::negated)
668            .collect();
669        Ok(Self {
670            document: note,
671            total: lines.iter().sum(),
672            lines,
673            kind: note.original.kind,
674            reduces_supply: true,
675        })
676    }
677
678    /// A line's quantity, signed like its amounts
679    fn quantity(&self, item: &GstLineItem) -> BigDecimal {
680        if self.reduces_supply {
681            -&item.quantity
682        } else {
683            item.quantity.clone()
684        }
685    }
686}

688impl Gstr1Return {
689    /// Aggregate a filer's invoices and credit notes for a period into GSTR-1
690    ///
691    /// Every invoice must be issued by `filer`, dated in `period`, carry a number no other
692    /// invoice has (ignoring case), pass the error-severity rules of
693    /// [`validate_invoice`](crate::gst::invoice::validate_invoice) as of its own date.
694    /// [`GstInvoice::supply_kind`] picks its table. Every credit note meets the same rules, with
695    /// [`validate_credit_note`](crate::gst::invoice::validate_credit_note), its number unique among the
696    /// notes; the supply kind of its invoice picks its table. `master` supplies the HSN/SAC summary's
697    /// descriptions, usually [`HsnMaster::global`]. Output is ordered by buyer GSTIN or place of
698    /// supply, then invoice date and number, then rate, whatever the input order.
699    ///
700    /// # Errors
701    ///
702    /// [`Gstr1Error::SellerMismatch`], [`Gstr1Error::OutsidePeriod`],
703    /// [`Gstr1Error::DuplicateInvoiceNumber`] or [`Gstr1Error::NotCompliant`] for the first
704    /// invoice that fails, [`Gstr1Error::InvalidCreditNote`] for the first credit note that fails,
705    /// or [`Gstr1Error::Invoice`] if a line's tax can't be computed.
706    ///
707    /// # Example
708    ///
709    /// ```
710    /// use accounting_core::gst::invoice::{GstInvoice, GstLineItem, Gstin, HsnMaster};
711    /// use accounting_core::gst::returns::{Gstr1Return, ReturnPeriod};
712    /// use bigdecimal::BigDecimal;
713    /// use chrono::NaiveDate;
714    ///
715    /// # fn main() -> Result<(), accounting_core::Error> {
716    /// let seller = Gstin::parse("27AAPFU0939F1ZV")?;
717    /// let item = GstLineItem::new("998314", "IT consulting", BigDecimal::from(1),
718    ///     BigDecimal::from(1000), BigDecimal::from(18))?;
719    /// let invoice = GstInvoice::new("INV-001", NaiveDate::from_ymd_opt(2024, 11, 15).unwrap(),
720    ///     seller.clone(), Gstin::parse("29AAPFU0939F1ZR")?, vec![item])?;
721    ///
722    /// let period = ReturnPeriod::new(2024, 11)?;
723    /// let gstr1 = Gstr1Return::build(&seller, period, &[invoice], &[], HsnMaster::global())?;
724    /// assert_eq!(gstr1.b2b[0].invoices[0].invoice_value, BigDecimal::from(1180));
725    ///
726    /// let json = gstr1.to_json()?;
727    /// assert!(json.contains(r#""fp": "112024""#));
728    /// assert!(json.contains(r#""iamt": 180.0"#));
729    /// # Ok(())
730    /// # }
731    /// ```
732    pub fn build(
733        filer: &Gstin,
734        period: ReturnPeriod,
735        invoices: &[GstInvoice],
736        credit_notes: &[GstCreditNote],
737        master: &HsnMaster,
738    ) -> Result<Self, Gstr1Error> {
739        let (invoices, notes) = checked_and_priced(filer, period, invoices, credit_notes)?;
740        Ok(Self::from_priced(filer, period, &invoices, &notes, master))
741    }
742
743    /// Lay the checked, priced and ordered documents out in the return's tables
744    fn from_priced(
745        filer: &Gstin,
746        period: ReturnPeriod,
747        invoices: &[PricedDocument],
748        notes: &[PricedDocument],
749        master: &HsnMaster,
750    ) -> Self {
751        let all = || invoices.iter().chain(notes);
752        let b2c_kinds = [SupplyKind::B2cl, SupplyKind::B2cs];
753        Self {
754            filer_gstin: filer.clone(),
755            period,
756            b2b: b2b_section(&of_kind(invoices, &[SupplyKind::B2b])),
757            b2cl: b2cl_section(&of_kind(invoices, &[SupplyKind::B2cl])),
758            b2cs: b2cs_section(&of_kind(all(), &[SupplyKind::B2cs])),
759            nil: nil_section(all()),
760            cdnr: cdnr_section(&of_kind(notes, &[SupplyKind::B2b])),
761            cdnur: cdnur_section(&of_kind(notes, &[SupplyKind::B2cl])),
762            hsn: HsnSummary {
763                b2b: hsn_rows(&of_kind(all(), &[SupplyKind::B2b]), master),
764                b2c: hsn_rows(&of_kind(all(), &b2c_kinds), master),
765            },
766            doc_issue: DocIssue {
767                documents: [
768                    doc_summary(
769                        OUTWARD_INVOICES_DOC_NUM,
770                        OUTWARD_INVOICES_DOC_TYPE,
771                        invoices,
772                    ),
773                    doc_summary(CREDIT_NOTES_DOC_NUM, CREDIT_NOTES_DOC_TYPE, notes),
774                ]
775                .into_iter()
776                .flatten()
777                .collect(),
778            },
779        }
780    }
781
782    /// The return as pretty-printed JSON in the portal's offline-tool schema
783    ///
784    /// # Errors
785    ///
786    /// [`Gstr1Error::Json`] if an amount is too large to write as a JSON number.
787    pub fn to_json(&self) -> Result<String, Gstr1Error> {
788        Ok(serde_json::to_string_pretty(self)?)
789    }
790}

792/// Documents by date, then number with its trailing digits compared as a number, so that
793/// `INV-9` comes before `INV-10`
794fn document_order(a: &dyn GstDocument, b: &dyn GstDocument) -> Ordering {
795    (a.date(), number_key(a.number()))
796        .cmp(&(b.date(), number_key(b.number())))
797        .then_with(|| a.number().cmp(b.number()))
798}
799
800/// An invoice number as its prefix and its trailing digits, the digits keyed by their
801/// significant length then text so that they compare as a number of any size
802fn number_key(number: &str) -> (&str, usize, &str) {
803    let prefix = number.trim_end_matches(|c: char| c.is_ascii_digit());
804    let digits = number
805        .strip_prefix(prefix)
806        .unwrap_or_default()
807        .trim_start_matches('0');
808    (prefix, digits.len(), digits)
809}

811/// The invoices and credit notes once each is checked as a return takes it (see
812/// [`Gstr1Return::build`]), priced and ordered by date then number
813pub(super) fn checked_and_priced<'a>(
814    filer: &Gstin,
815    period: ReturnPeriod,
816    invoices: &'a [GstInvoice],
817    credit_notes: &'a [GstCreditNote],
818) -> Result<(Vec<PricedDocument<'a>>, Vec<PricedDocument<'a>>), Gstr1Error> {
819    check_documents(filer, period, invoices, |invoice| {
820        compliance_errors(invoice, invoice.invoice_date)
821    })
822    .map_err(|(number, problem)| Gstr1Error::for_invoice(number, problem))?;
823    check_documents(filer, period, credit_notes, |note| {
824        credit_note_errors(note, note.note_date)
825    })
826    .map_err(|(note_number, reason)| Gstr1Error::InvalidCreditNote {
827        note_number,
828        reason,
829    })?;
830    Ok((
831        priced(invoices, PricedDocument::invoice)?,
832        priced(credit_notes, PricedDocument::credit_note)?,
833    ))
834}
835
836/// Check every document, and that no two share a number ignoring case; the first failure comes
837/// back with the number of the document that failed
838fn check_documents<D: GstDocument>(
839    filer: &Gstin,
840    period: ReturnPeriod,
841    documents: &[D],
842    errors: impl Fn(&D) -> Vec<ComplianceIssue>,
843) -> Result<(), (String, DocumentProblem)> {
844    documents
845        .iter()
846        .try_fold(HashSet::new(), |mut seen, document| {
847            let problem = document_problem(filer, period, document, &errors).or_else(|| {
848                (!seen.insert(document.number().to_ascii_uppercase()))
849                    .then_some(DocumentProblem::DuplicateNumber)
850            });
851            match problem {
852                Some(problem) => Err((document.number().to_string(), problem)),
853                None => Ok(seen),
854            }
855        })
856        .map(drop)
857}
858
859/// The first rule one document breaks on its own, if any
860fn document_problem<D: GstDocument>(
861    filer: &Gstin,
862    period: ReturnPeriod,
863    document: &D,
864    errors: impl Fn(&D) -> Vec<ComplianceIssue>,
865) -> Option<DocumentProblem> {
866    if document.seller_gstin() != filer {
867        return Some(DocumentProblem::SellerMismatch(
868            document.seller_gstin().clone(),
869        ));
870    }
871    if !period.contains(document.date()) {
872        return Some(DocumentProblem::OutsidePeriod {
873            date: document.date(),
874            period,
875        });
876    }
877    let issues = errors(document);
878    (!issues.is_empty()).then_some(DocumentProblem::NotCompliant(issues))
879}

881/// Every document priced with `price`, by date then number
882fn priced<'a, D>(
883    documents: &'a [D],
884    price: impl Fn(&'a D) -> Result<PricedDocument<'a>, InvoiceError>,
885) -> Result<Vec<PricedDocument<'a>>, InvoiceError> {
886    let mut priced = documents.iter().map(price).collect::<Result<Vec<_>, _>>()?;
887    priced.sort_by(|a, b| document_order(a.document, b.document));
888    Ok(priced)
889}

902/// Values grouped under their keys, in key order, keeping their order within each key
903pub(super) fn grouped<K: Ord, V>(pairs: impl Iterator<Item = (K, V)>) -> BTreeMap<K, Vec<V>> {
904    pairs.fold(BTreeMap::new(), |mut groups, (key, value)| {
905        groups.entry(key).or_insert_with(Vec::new).push(value);
906        groups
907    })
908}

983/// Table 7: every taxed line summed per place of supply and rate; untaxed lines go to Table 8
984///
985/// The filer is the seller of every invoice, so one place of supply is either always inside its
986/// state or always outside it, and the supply type is the same for every line of a row.
987fn b2cs_section(priced: &[&PricedDocument]) -> Vec<B2csRow> {
988    priced
989        .iter()
990        .flat_map(|p| {
991            let key = (
992                p.document.buyer().place_of_supply(),
993                p.document.is_inter_state(),
994            );
995            taxable_rates(p).map(move |(rate, b)| (key, rate, b))
996        })
997        .fold(
998            BTreeMap::new(),
999            |mut rows, ((pos, inter), rate, breakdown)| {
1000                *rows
1001                    .entry((pos, rate, inter))
1002                    .or_insert_with(GstBreakdown::default) += &breakdown;
1003                rows
1004            },
1005        )
1006        .into_iter()
1007        .map(|((pos, rate, inter), breakdown)| B2csRow::new(pos, inter, rate, breakdown))
1008        .collect()

1011/// Table 8: every line that charges no GST, summed per supply type into the column of its
1012/// treatment
1013fn nil_section<'p, 'a: 'p>(priced: impl Iterator<Item = &'p PricedDocument<'a>>) -> NilSupplies {
1014    let rows = priced
1015        .flat_map(|p| {
1016            let supply_type = nil_supply_type(p.document);
1017            p.document
1018                .line_items()
1019                .iter()
1020                .zip(&p.lines)
1021                .filter(|(item, _)| !item.charges_gst())
1022                .map(move |(item, b)| (supply_type, item.treatment, &b.taxable_value))
1023        })
1024        .fold(
1025            BTreeMap::new(),
1026            |mut rows, (supply_type, treatment, value)| {
1027                *rows
1028                    .entry(supply_type)
1029                    .or_insert_with(|| NilRow::empty(supply_type))
1030                    .column_mut(treatment) += value;
1031                rows
1032            },
1033        );
1034    NilSupplies {
1035        rows: rows.into_values().collect(),
1036    }
1037}

1057/// The breakdowns of the lines `keep` accepts, summed per rate so that 18 and 18.00 are one rate
1058fn by_rate_where(
1059    p: &PricedDocument,
1060    keep: impl Fn(&GstLineItem) -> bool,

1064        .iter()

1073/// The lines of every invoice with one HSN/SAC code, unit and rate, before numbering
1074#[derive(Default)]
1075struct HsnTotals {
1076    quantity: BigDecimal,
1077    breakdown: GstBreakdown,
1078    first_description: String,
1079}
1080
1081/// Grouping key of a Table 12 row: HSN/SAC code, unit quantity code and rate
1082type HsnKey = (String, String, BigDecimal);
1083
1084/// One tab of Table 12: every line of `priced` grouped by HSN/SAC code, unit and rate, so
1085/// quantities in different units are never added together
1086fn hsn_rows(priced: &[&PricedDocument], master: &HsnMaster) -> Vec<HsnRow> {
1087    let totals = priced
1088        .iter()
1089        .flat_map(|p| {
1090            p.document
1091                .line_items()
1092                .iter()
1093                .zip(&p.lines)
1094                .map(|(item, breakdown)| (item, breakdown, p.quantity(item)))
1095        })
1096        .fold(BTreeMap::new(), |mut rows, (item, breakdown, quantity)| {
1097            let totals: &mut HsnTotals = rows
1098                .entry((item.hsn_sac.clone(), uqc(item), item.gst_rate.normalized()))
1099                .or_default();
1100            if totals.first_description.is_empty() {
1101                totals.first_description.clone_from(&item.description);
1102            }
1103            totals.quantity += quantity;
1104            totals.breakdown += breakdown;
1105            rows
1106        });
1107
1108    totals
1109        .into_iter()
1110        .enumerate()
1111        .map(|(i, (key, totals))| hsn_row(i + 1, key, totals, master))
1112        .collect()
1113}

1115/// The unit quantity code a line reports: its unit in upper case, as the portal's UQC codes
1116/// are, or [`GOODS_UQC`], for goods, and [`SERVICES_UQC`] for services
1117fn uqc(item: &GstLineItem) -> String {
1118    match HsnSacKind::of_code(&item.hsn_sac) {
1119        HsnSacKind::Hsn => item
1120            .unit
1121            .as_deref()
1122            .map(str::trim)
1123            .filter(|unit| !unit.is_empty())
1124            .map_or_else(|| GOODS_UQC.to_string(), str::to_ascii_uppercase),
1125        HsnSacKind::Sac => SERVICES_UQC.to_string(),
1126    }
1127}

1129/// A description cut to the whole words that fit in [`HSN_DESCRIPTION_MAX`] characters, or cut
1130/// mid-word when its first word alone is longer
1131pub(super) fn fit_description(text: &str) -> String {
1132    let text = text.split_whitespace().collect::<Vec<_>>().join(" ");
1133    if text.chars().count() <= HSN_DESCRIPTION_MAX {
1134        return text;
1135    }
1136    let cut: String = text.chars().take(HSN_DESCRIPTION_MAX + 1).collect();
1137    match cut.rfind(' ') {
1138        Some(space) if space > 0 => cut[..space].trim_end().to_string(),
1139        _ => cut.chars().take(HSN_DESCRIPTION_MAX).collect(),
1140    }
1141}

1143fn hsn_row(number: usize, key: HsnKey, totals: HsnTotals, master: &HsnMaster) -> HsnRow {
1144    let (hsn_sac, uqc, rate) = key;
1145    let description = master
1146        .lookup(&hsn_sac)
1147        .map_or(totals.first_description, |entry| entry.description.clone());
1148    let description = fit_description(&description);
1149    let quantity = match HsnSacKind::of_code(&hsn_sac) {
1150        HsnSacKind::Hsn => totals.quantity,
1151        HsnSacKind::Sac => BigDecimal::zero(),
1152    };
1153    let breakdown = totals.breakdown;
1154    HsnRow {
1155        number,
1156        hsn_sac,
1157        description,
1158        uqc,
1159        quantity,
1160        total_value: breakdown.total,
1161        taxable_value: breakdown.taxable_value,
1162        igst: breakdown.igst,
1163        cgst: breakdown.cgst,
1164        sgst: breakdown.sgst,
1165        cess: BigDecimal::zero(),
1166        rate,
1167    }
1168}

1170/// One kind of document in Table 13: the documents as one series, from the first to the last by
1171/// date then number, or `None` when none were issued
1172fn doc_summary(doc_num: usize, doc_type: &str, priced: &[PricedDocument]) -> Option<DocSummary> {
