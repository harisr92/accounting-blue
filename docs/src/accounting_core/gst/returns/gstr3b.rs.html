1//! GSTR-3B's outward supplies: Table 3.1 from the period's invoices and credit notes
2//!
3//! GSTR-3B is the monthly summary return a regular taxpayer files with the tax paid. Table 3.1
4//! reports the month's outward supplies in five rows; this builds the four that sales decide:
5//!
6//! - (a) taxable supplies other than zero-rated, nil-rated and exempt: the lines that charge GST;
7//! - (b) zero-rated supplies (exports and supplies to SEZs): none, since exports aren't covered;
8//! - (c) other outward supplies, nil-rated and exempt: taxable lines at 0% and exempt lines;
9//! - (d) inward supplies liable to reverse charge: purchases, not built from sales, so zero;
10//! - (e) non-GST outward supplies.
11//!
12//! A credit note is subtracted from the row its lines fall in. Building is pure, and the
13//! documents are checked as [`Gstr1Return::build`](super::Gstr1Return::build) checks them.
14
15use super::gstr1::{checked_and_priced, Gstr1Error, PricedDocument};
16use super::period::ReturnPeriod;
17use crate::gst::invoice::SupplyTreatment;
18use crate::gst::invoice::{GstBreakdown, GstCreditNote, GstInvoice, GstLineItem, Gstin};
19use bigdecimal::BigDecimal;
20use serde::Serialize;
21
22/// One row of Table 3.1: the value of the supplies and the tax on them
23#[derive(Debug, Clone, Default, PartialEq, Serialize)]
24pub struct OutwardRow {
25    /// Taxable value
26    pub taxable_value: BigDecimal,
27    /// Integrated tax
28    pub igst: BigDecimal,
29    /// Central tax
30    pub cgst: BigDecimal,
31    /// State or union territory tax
32    pub sgst: BigDecimal,
33    /// Compensation cess; not charged by these documents, so always zero
34    pub cess: BigDecimal,
35}
36
37impl OutwardRow {
38    /// The row with `breakdown` added
39    fn plus(self, breakdown: &GstBreakdown) -> Self {
40        Self {
41            taxable_value: self.taxable_value + &breakdown.taxable_value,
42            igst: self.igst + &breakdown.igst,
43            cgst: self.cgst + &breakdown.cgst,
44            sgst: self.sgst + &breakdown.sgst,
45            cess: self.cess,
46        }
47    }
48}

50/// Table 3.1 of GSTR-3B: the period's outward supplies
51#[derive(Debug, Clone, Default, PartialEq, Serialize)]
52pub struct OutwardSupplies {
53    /// (a) Outward taxable supplies (other than zero-rated, nil-rated and exempted)
54    pub taxable: OutwardRow,
55    /// (b) Outward taxable supplies (zero-rated)
56    pub zero_rated: OutwardRow,
57    /// (c) Other outward supplies (nil-rated, exempted)
58    pub nil_rated_exempt: OutwardRow,
59    /// (d) Inward supplies (liable to reverse charge)
60    pub reverse_charge: OutwardRow,
61    /// (e) Non-GST outward supplies
62    pub non_gst: OutwardRow,
63}
64
65/// The rows of Table 3.1 a sales line can fall in
66#[derive(Debug, Clone, Copy, PartialEq, Eq)]
67enum OutwardKind {
68    Taxable,
69    NilRatedExempt,
70    NonGst,
71}
72
73/// The row a line falls in, by whether it charges GST and its treatment
74fn outward_kind(item: &GstLineItem) -> OutwardKind {
75    match item.treatment {
76        SupplyTreatment::NonGst => OutwardKind::NonGst,
77        _ if item.charges_gst() => OutwardKind::Taxable,
78        SupplyTreatment::Taxable | SupplyTreatment::Exempt => OutwardKind::NilRatedExempt,
79    }
80}
81
82impl OutwardSupplies {
83    fn row_mut(&mut self, kind: OutwardKind) -> &mut OutwardRow {
84        match kind {
85            OutwardKind::Taxable => &mut self.taxable,
86            OutwardKind::NilRatedExempt => &mut self.nil_rated_exempt,
87            OutwardKind::NonGst => &mut self.non_gst,
88        }
89    }
90
91    /// Every line of the priced documents added to its row; a credit note's amounts are negative
92    fn from_priced<'p, 'a: 'p>(priced: impl Iterator<Item = &'p PricedDocument<'a>>) -> Self {
93        priced
94            .flat_map(|p| p.document.line_items().iter().zip(&p.lines))
95            .fold(Self::default(), |mut supplies, (item, breakdown)| {
96                let row = supplies.row_mut(outward_kind(item));
97                *row = std::mem::take(row).plus(breakdown);
98                supplies
99            })
100    }
101}

103/// GSTR-3B's outward supplies for one filer and period
104#[derive(Debug, Clone, PartialEq, Serialize)]
105pub struct Gstr3bOutward {
106    /// The filer's GSTIN
107    pub filer_gstin: Gstin,
108    /// The month filed
109    pub period: ReturnPeriod,
110    /// Table 3.1
111    pub outward: OutwardSupplies,
112}
113
114impl Gstr3bOutward {
115    /// Sum a filer's invoices and credit notes for a period into Table 3.1
116    ///
117    /// # Errors
118    ///
119    /// The same as [`Gstr1Return::build`](super::Gstr1Return::build): the documents are checked
120    /// the same way, so a month that files as GSTR-1 summarises as GSTR-3B.
121    pub fn build(
122        filer: &Gstin,
123        period: ReturnPeriod,
124        invoices: &[GstInvoice],
125        credit_notes: &[GstCreditNote],
126    ) -> Result<Self, Gstr1Error> {
127        let (invoices, notes) = checked_and_priced(filer, period, invoices, credit_notes)?;
128        Ok(Self {
129            filer_gstin: filer.clone(),
130            period,
131            outward: OutwardSupplies::from_priced(invoices.iter().chain(&notes)),
132        })
133    }
134}
