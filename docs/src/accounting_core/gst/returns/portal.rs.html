1//! How GST returns write their values in the GST portal's JSON: amounts rounded to paise and
2//! written as JSON numbers, shared by GSTR-1 and GSTR-3B
3
4use crate::gst::tax::round_to_paise;
5use bigdecimal::{BigDecimal, ToPrimitive};
6use serde::ser::Error as _;
7use serde::Serializer;
8
9/// An amount rounded to paise, written as a JSON number
10pub(super) fn portal_amount<S: Serializer>(
11    amount: &BigDecimal,
12    serializer: S,
13) -> Result<S::Ok, S::Error> {
14    portal_number(&round_to_paise(amount), serializer)
15}
16
17/// A decimal written as a JSON number
18pub(super) fn portal_number<S: Serializer>(
19    value: &BigDecimal,
20    serializer: S,
21) -> Result<S::Ok, S::Error> {
22    let number = value
23        .to_f64()
24        .filter(|number| number.is_finite())
25        .ok_or_else(|| S::Error::custom(format!("{value} does not fit a JSON number")))?;
26    serializer.serialize_f64(number)
27}
