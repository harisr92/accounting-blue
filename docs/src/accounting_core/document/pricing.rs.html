1//! Pricing a document's lines under a tax regime
2//!
3//! A [`Document`] is a numbered, dated list of lines with an optional discount off the
4//! subtotal, taxed under a [`TaxRegime`]. [`price_lines`] is the one pricing engine every
5//! document shares: it rounds each line, shares the document discount across the lines by value
6//! with [`apportion`], and asks the regime for the tax on what is left.
7
8use crate::document::discount::{apportion, Discount, DiscountError};
9use crate::document::line::DocumentLine;
10use crate::document::tax::{TaxBreakdown, TaxRegime};
11use crate::money::round_money;
12use bigdecimal::{BigDecimal, Zero};
13use chrono::NaiveDate;
14
15/// A numbered, dated document of lines, taxed under one regime
16///
17/// The provided methods hold the arithmetic every document shares: the subtotal, the discount
18/// off it, and the tax breakdown of each line and of the whole document.
19pub trait Document {
20    /// The document's line type
21    type Line: DocumentLine;
22
23    /// The regime that taxes the document's lines
24    type Regime: TaxRegime;
25
26    /// Why the document could not be priced: a discount that no longer fits, or the regime's
27    /// own error
28    type Error: From<DiscountError> + From<<Self::Regime as TaxRegime>::Error>;
29
30    /// Document number
31    fn number(&self) -> &str;
32
33    /// Date of issue
34    fn date(&self) -> NaiveDate;
35
36    /// The document's lines
37    fn lines(&self) -> &[Self::Line];
38
39    /// Discount off the subtotal, before tax; none unless the document gives one
40    fn document_discount(&self) -> Option<&Discount> {
41        None
42    }
43
44    /// The regime the document's lines are taxed under
45    fn regime(&self) -> Self::Regime;
46
47    /// Sum of the lines' taxable values after their own discounts, each rounded, before the
48    /// [`Document::document_discount`]
49    fn subtotal(&self) -> BigDecimal {
50        line_values(self.lines()).iter().sum()
51    }
52
53    /// Amount the [`Document::document_discount`] takes off the subtotal; zero when there is
54    /// none
55    fn discount_amount(&self) -> BigDecimal {
56        self.document_discount()
57            .map_or_else(BigDecimal::zero, |discount| {
58                discount.amount_off(&self.subtotal())
59            })
60    }
61
62    /// Check the [`Document::document_discount`] can come off the subtotal
63    ///
64    /// # Errors
65    ///
66    /// The [`DiscountError`] naming the rule it breaks.
67    fn check_discount(&self) -> Result<(), DiscountError> {
68        self.document_discount()
69            .map_or(Ok(()), |discount| discount.check(&self.subtotal()))
70    }
71
72    /// Tax breakdown for each line, in order, after its share of the document discount
73    ///
74    /// # Errors
75    ///
76    /// A [`DiscountError`] if the document discount no longer fits the subtotal, such as after
77    /// its lines were edited, or the regime's error for the first line it can't tax.
78    fn line_tax_breakdowns(&self) -> Result<Vec<TaxBreakdown>, Self::Error> {
79        Ok(price_lines(self)?.breakdowns)
80    }
81
82    /// Tax breakdown summed across all lines, one component per tax
83    ///
84    /// # Errors
85    ///
86    /// As [`Document::line_tax_breakdowns`].
87    fn tax_breakdown(&self) -> Result<TaxBreakdown, Self::Error> {
88        Ok(self.line_tax_breakdowns()?.iter().sum())
89    }
90}
91
92/// Each line's taxable value after its own discount, rounded
93#[must_use]
94pub fn line_values<L: DocumentLine>(lines: &[L]) -> Vec<BigDecimal> {
95    lines
96        .iter()
97        .map(|line| round_money(&line.taxable_value()))
98        .collect()
99}

101/// A document's lines priced once: the subtotal, the discount off it, and each line's breakdown
102/// after its share of that discount
103#[derive(Debug, Clone, PartialEq, Eq)]
104pub struct PricedLines {
105    /// Sum of the lines after their own discounts, before the document discount
106    pub subtotal: BigDecimal,
107    /// Amount the document discount takes off the subtotal; zero when there is none
108    pub discount: BigDecimal,
109    /// Each line's breakdown, in order
110    pub breakdowns: Vec<TaxBreakdown>,
111}
112
113/// Price `document`'s lines: share its discount across them by value and tax what is left
114///
115/// The breakdown methods of [`Document`] and any print model read this, so the line values and
116/// the subtotal are worked out once.
117///
118/// # Errors
119///
120/// As [`Document::line_tax_breakdowns`].
121pub fn price_lines<D: Document + ?Sized>(document: &D) -> Result<PricedLines, D::Error> {
122    let values = line_values(document.lines());
123    let subtotal: BigDecimal = values.iter().sum();
124    let discount = document
125        .document_discount()
126        .map_or(Ok(BigDecimal::zero()), |discount| {
127            discount
128                .check(&subtotal)
129                .map(|()| discount.amount_off(&subtotal))
130        })?;
131    let regime = document.regime();
132    let breakdowns = document
133        .lines()
134        .iter()
135        .zip(values.iter().zip(apportion(&discount, &values)))
136        .map(|(line, (value, share))| regime.line_tax(&(value - share), line.tax_rate()))
137        .collect::<Result<_, _>>()?;
138    Ok(PricedLines {
139        subtotal,
140        discount,
141        breakdowns,
142    })
143}
