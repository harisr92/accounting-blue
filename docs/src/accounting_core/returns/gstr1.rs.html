1//! GSTR-1: the return of outward supplies
2//!
3//! [`Gstr1Return::build`] aggregates a filer's invoices for one [`ReturnPeriod`] into the
4//! sections of GSTR-1 this crate can fill: Table 4A (B2B supplies, grouped by buyer GSTIN with
5//! one item per rate), Table 5 (B2CL: large inter-state supplies to unregistered buyers,
6//! grouped by place of supply), Table 7 (B2CS: every other supply to unregistered buyers,
7//! summed per place of supply and rate), Table 8 (nil-rated, exempt and non-GST lines, by
8//! supply type), Table 12 (the HSN/SAC summary, grouped by code and rate) and Table 13
9//! (documents issued).
10//! The builder is pure; [`Gstr1Return::to_json`] writes the result in the GST portal's
11//! offline-tool schema, with its short keys and amounts as JSON numbers rounded to paise, ready
12//! to upload.
13//!
14//! Table 12 is split into the two tabs the portal has used since 2025: rows for B2B supplies go
15//! under `hsn.hsn_b2b`, and rows for B2CL and B2CS supplies under `hsn.hsn_b2c`. Which table an
16//! invoice belongs to is [`GstInvoice::supply_kind`]. Table 7 reports only taxable supplies other
17//! than through an e-commerce operator (`typ` `OE`).
18//!
19//! A line that charges no GST ([`GstLineItem::charges_gst`]) goes to Table 8, never to Table
20//! 4A, 5 or 7, and stays in the HSN summary at rate 0. Its [`SupplyTreatment`] picks the
21//! column: a taxable line at 0% is nil-rated, the others are exempt or non-GST. A B2B or B2CL
22//! invoice keeps its whole value but lists only its taxed rates, and one with no taxed line
23//! appears only in Table 8.

286impl B2csRow {
287    fn new(place_of_supply: &str, is_inter_state: bool, rate: BigDecimal, b: GstBreakdown) -> Self {
288        let (supply_type, igst, cgst, sgst) = if is_inter_state {
289            (SupplyType::Inter, Some(b.igst), None, None)
290        } else {
291            (SupplyType::Intra, None, Some(b.cgst), Some(b.sgst))
292        };
293        Self {
294            supply_type,
295            rate,
296            b2cs_type: B2csType::OtherThanEcommerce,
297            place_of_supply: place_of_supply.to_string(),
298            taxable_value: b.taxable_value,
299            igst,
300            cgst,
301            sgst,
302            cess: BigDecimal::zero(),
303        }
304    }

510/// Why a GSTR-1 return could not be built or written
511#[derive(Debug, thiserror::Error)]
512pub enum Gstr1Error {
513    /// The month is not 1-12, or the year is outside the GST era
514    #[error("invalid return period: month {month} of {year}")]
515    InvalidPeriod {
516        /// Year given
517        year: i32,
518        /// Month given
519        month: u32,
520    },
521    /// The invoice was issued by someone other than the filer
522    #[error("invoice {invoice_number} was issued by {seller}, not the filer")]
523    SellerMismatch {
524        /// The invoice number
525        invoice_number: String,
526        /// GSTIN of its seller
527        seller: Gstin,
528    },
529    /// The invoice is dated outside the return period
530    #[error("invoice {invoice_number} dated {date} is outside the period {period}")]
531    OutsidePeriod {
532        /// The invoice number
533        invoice_number: String,
534        /// Its date
535        date: NaiveDate,
536        /// The period being filed
537        period: ReturnPeriod,
538    },
539    /// Two invoices share a number, compared case-insensitively
540    #[error("invoice number {0} appears more than once")]
541    DuplicateInvoiceNumber(String),
542    /// The invoice breaks an error-severity compliance rule
543    #[error("invoice {invoice_number} is not compliant: {} error(s)", issues.len())]
544    NotCompliant {
545        /// The invoice number
546        invoice_number: String,
547        /// The error-severity issues found
548        issues: Vec<ComplianceIssue>,
549    },
550    /// An invoice's tax could not be computed
551    #[error(transparent)]
552    Invoice(#[from] InvoiceError),
553    /// The return could not be written as JSON
554    #[error("could not write GSTR-1 JSON: {0}")]
555    Json(#[from] serde_json::Error),
556}

558/// An invoice with the tax breakdown of each of its lines, their sum and its supply kind,
559/// computed once
560struct PricedInvoice<'a> {
561    invoice: &'a GstInvoice,
562    lines: Vec<GstBreakdown>,
563    total: GstBreakdown,
564    kind: SupplyKind,
565}
566
567impl<'a> PricedInvoice<'a> {
568    /// Price an invoice's lines and classify it by supply kind
569    fn new(invoice: &'a GstInvoice) -> Result<Self, InvoiceError> {
570        let lines = invoice.line_breakdowns()?;
571        let total: GstBreakdown = lines.iter().sum();
572        Ok(Self {
573            invoice,
574            kind: supply_kind_for(invoice, &total.total),
575            lines,
576            total,
577        })
578    }
579}

581impl Gstr1Return {
582    /// Aggregate a filer's invoices for a period into GSTR-1
583    ///
584    /// Every invoice must be issued by `filer`, dated in `period`, carry a number no other
585    /// invoice has (ignoring case), pass the error-severity rules of
586    /// [`validate_invoice`](crate::invoice::validate_invoice) as of its own date.
587    /// [`GstInvoice::supply_kind`] picks its table. `master` supplies the HSN/SAC summary's
588    /// descriptions, usually [`HsnMaster::global`]. Output is ordered by buyer GSTIN or place of
589    /// supply, then invoice date and number, then rate, whatever the input order.
590    ///
591    /// # Errors
592    ///
593    /// [`Gstr1Error::SellerMismatch`], [`Gstr1Error::OutsidePeriod`],
594    /// [`Gstr1Error::DuplicateInvoiceNumber`] or [`Gstr1Error::NotCompliant`] for the first
595    /// invoice that fails, or [`Gstr1Error::Invoice`] if a line's tax can't be computed.
596    ///
597    /// # Example
598    ///
599    /// ```
600    /// use accounting_core::invoice::{GstInvoice, GstLineItem, Gstin, HsnMaster};
601    /// use accounting_core::returns::{Gstr1Return, ReturnPeriod};
602    /// use bigdecimal::BigDecimal;
603    /// use chrono::NaiveDate;
604    ///
605    /// # fn main() -> Result<(), accounting_core::Error> {
606    /// let seller = Gstin::parse("27AAPFU0939F1ZV")?;
607    /// let item = GstLineItem::new("998314", "IT consulting", BigDecimal::from(1),
608    ///     BigDecimal::from(1000), BigDecimal::from(18))?;
609    /// let invoice = GstInvoice::new("INV-001", NaiveDate::from_ymd_opt(2024, 11, 15).unwrap(),
610    ///     seller.clone(), Gstin::parse("29AAPFU0939F1ZR")?, vec![item])?;
611    ///
612    /// let period = ReturnPeriod::new(2024, 11)?;
613    /// let gstr1 = Gstr1Return::build(&seller, period, &[invoice], HsnMaster::global())?;
614    /// assert_eq!(gstr1.b2b[0].invoices[0].invoice_value, BigDecimal::from(1180));
615    ///
616    /// let json = gstr1.to_json()?;
617    /// assert!(json.contains(r#""fp": "112024""#));
618    /// assert!(json.contains(r#""iamt": 180.0"#));
619    /// # Ok(())
620    /// # }
621    /// ```
622    pub fn build(
623        filer: &Gstin,
624        period: ReturnPeriod,
625        invoices: &[GstInvoice],
626        master: &HsnMaster,
627    ) -> Result<Self, Gstr1Error> {
628        check_invoices(filer, period, invoices)?;
629
630        let mut priced = invoices
631            .iter()
632            .map(PricedInvoice::new)
633            .collect::<Result<Vec<_>, _>>()?;
634        priced.sort_by(|a, b| invoice_order(a.invoice, b.invoice));
635        let b2b = of_kind(&priced, SupplyKind::B2b);
636        let b2cl = of_kind(&priced, SupplyKind::B2cl);
637        let b2cs = of_kind(&priced, SupplyKind::B2cs);
638        let b2c: Vec<_> = b2cl.iter().chain(&b2cs).copied().collect();
639
640        Ok(Self {
641            filer_gstin: filer.clone(),
642            period,
643            b2b: b2b_section(&b2b),
644            b2cl: b2cl_section(&b2cl),
645            b2cs: b2cs_section(&b2cs),
646            nil: nil_section(&priced),
647            hsn: HsnSummary {
648                b2b: hsn_rows(&b2b, master),
649                b2c: hsn_rows(&b2c, master),
650            },
651            doc_issue: doc_issue_section(&priced),
652        })
653    }
654
655    /// The return as pretty-printed JSON in the portal's offline-tool schema
656    ///
657    /// # Errors
658    ///
659    /// [`Gstr1Error::Json`] if an amount is too large to write as a JSON number.
660    pub fn to_json(&self) -> Result<String, Gstr1Error> {
661        Ok(serde_json::to_string_pretty(self)?)
662    }
663}

665/// Invoices by date, then number with its trailing digits compared as a number, so that
666/// `INV-9` comes before `INV-10`
667fn invoice_order(a: &GstInvoice, b: &GstInvoice) -> Ordering {
668    (a.invoice_date, number_key(&a.invoice_number))
669        .cmp(&(b.invoice_date, number_key(&b.invoice_number)))
670        .then_with(|| a.invoice_number.cmp(&b.invoice_number))
671}
672
673/// An invoice number as its prefix and its trailing digits, the digits keyed by their
674/// significant length then text so that they compare as a number of any size
675fn number_key(number: &str) -> (&str, usize, &str) {
676    let prefix = number.trim_end_matches(|c: char| c.is_ascii_digit());
677    let digits = number
678        .strip_prefix(prefix)
679        .unwrap_or_default()
680        .trim_start_matches('0');
681    (prefix, digits.len(), digits)
682}
683
684/// Check every invoice, and that no two share a number
685fn check_invoices(
686    filer: &Gstin,
687    period: ReturnPeriod,
688    invoices: &[GstInvoice],
689) -> Result<(), Gstr1Error> {
690    invoices
691        .iter()
692        .try_fold(HashSet::new(), |mut seen, invoice| {
693            check_invoice(filer, period, invoice)?;
694            if seen.insert(invoice.invoice_number.to_ascii_uppercase()) {
695                Ok(seen)
696            } else {
697                Err(Gstr1Error::DuplicateInvoiceNumber(
698                    invoice.invoice_number.clone(),
699                ))
700            }
701        })
702        .map(drop)
703}
704
705/// The rules one invoice must meet on its own
706fn check_invoice(
707    filer: &Gstin,
708    period: ReturnPeriod,
709    invoice: &GstInvoice,
710) -> Result<(), Gstr1Error> {
711    let invoice_number = || invoice.invoice_number.clone();
712    if invoice.seller_gstin != *filer {
713        return Err(Gstr1Error::SellerMismatch {
714            invoice_number: invoice_number(),
715            seller: invoice.seller_gstin.clone(),
716        });
717    }
718    if !period.contains(invoice.invoice_date) {
719        return Err(Gstr1Error::OutsidePeriod {
720            invoice_number: invoice_number(),
721            date: invoice.invoice_date,
722            period,
723        });
724    }
725    let issues = compliance_errors(invoice, invoice.invoice_date);
726    if issues.is_empty() {
727        Ok(())
728    } else {
729        Err(Gstr1Error::NotCompliant {
730            invoice_number: invoice_number(),
731            issues,
732        })
733    }
734}

736/// The priced invoices of one supply kind, keeping their order
737fn of_kind<'p, 'a>(
738    priced: &'p [PricedInvoice<'a>],
739    kind: SupplyKind,
740) -> Vec<&'p PricedInvoice<'a>> {
741    priced.iter().filter(|p| p.kind == kind).collect()

744/// Table 4A: invoices grouped by buyer GSTIN, keeping their order within each buyer
745///
746/// An invoice with no taxed line is left out: its lines are all in Table 8.
747fn b2b_section(priced: &[&PricedInvoice]) -> Vec<B2bParty> {
748    priced
749        .iter()
750        .filter_map(|p| Some((p.invoice.buyer.gstin()?, b2b_invoice(p)?)))
751        .fold(
752            BTreeMap::<&Gstin, Vec<B2bInvoice>>::new(),
753            |mut parties, (buyer_gstin, invoice)| {
754                parties.entry(buyer_gstin).or_default().push(invoice);
755                parties
756            },
757        )
758        .into_iter()
759        .map(|(buyer_gstin, invoices)| B2bParty {
760            buyer_gstin: buyer_gstin.clone(),
761            invoices,
762        })
763        .collect()
764}

766/// One Table 4A invoice with an item per taxed rate, or `None` when no line charges GST; its
767/// value stays the whole invoice's
768fn b2b_invoice(p: &PricedInvoice) -> Option<B2bInvoice> {
769    let items: Vec<_> = taxable_rates(p)
770        .enumerate()
771        .map(|(i, (rate, breakdown))| B2bItem {
772            number: i + 1,
773            detail: ItemDetail::new(rate, &breakdown),
774        })
775        .collect();
776    (!items.is_empty()).then(|| B2bInvoice {
777        invoice_number: p.invoice.invoice_number.clone(),
778        invoice_date: p.invoice.invoice_date,
779        invoice_value: p.total.total.clone(),
780        place_of_supply: p.invoice.buyer.place_of_supply().to_string(),
781        reverse_charge: false,
782        invoice_type: B2bInvoiceType::Regular,
783        items,
784    })
785}

787/// Table 5: invoices grouped by place of supply, keeping their order within each place
788///
789/// An invoice with no taxable line is left out: its 0% lines are all in Table 8.
790fn b2cl_section(priced: &[&PricedInvoice]) -> Vec<B2clPlace> {
791    priced
792        .iter()
793        .filter_map(|p| Some((p.invoice.buyer.place_of_supply(), b2cl_invoice(p)?)))
794        .fold(
795            BTreeMap::<&str, Vec<B2clInvoice>>::new(),
796            |mut places, (place_of_supply, invoice)| {
797                places.entry(place_of_supply).or_default().push(invoice);
798                places
799            },

801        .into_iter()
802        .map(|(place_of_supply, invoices)| B2clPlace {
803            place_of_supply: place_of_supply.to_string(),
804            invoices,
805        })
806        .collect()

809/// One Table 5 invoice with an item per taxed rate, or `None` when no line charges GST; its
810/// value stays the whole invoice's, untaxed lines included
811fn b2cl_invoice(p: &PricedInvoice) -> Option<B2clInvoice> {

813        .enumerate()
814        .map(|(i, (rate, breakdown))| B2clItem {
815            number: i + 1,
816            detail: B2clItemDetail::new(rate, &breakdown),
817        })
818        .collect();
819    (!items.is_empty()).then(|| B2clInvoice {
820        invoice_number: p.invoice.invoice_number.clone(),
821        invoice_date: p.invoice.invoice_date,
822        invoice_value: p.total.total.clone(),
823        items,
824    })

827/// Table 7: every taxed line summed per place of supply and rate; untaxed lines go to Table 8
828///
829/// The filer is the seller of every invoice, so one place of supply is either always inside its
830/// state or always outside it, and the supply type is the same for every line of a row.
831fn b2cs_section(priced: &[&PricedInvoice]) -> Vec<B2csRow> {
832    priced
833        .iter()
834        .flat_map(|p| {
835            let key = (
836                p.invoice.buyer.place_of_supply(),
837                p.invoice.is_inter_state(),
838            );
839            taxable_rates(p).map(move |(rate, b)| (key, rate, b))
840        })
841        .fold(
842            BTreeMap::new(),
843            |mut rows, ((pos, inter), rate, breakdown)| {
844                *rows
845                    .entry((pos, rate, inter))
846                    .or_insert_with(GstBreakdown::default) += &breakdown;
847                rows
848            },
849        )
850        .into_iter()
851        .map(|((pos, rate, inter), breakdown)| B2csRow::new(pos, inter, rate, breakdown))
852        .collect()
853}
854
855/// Table 8: every line that charges no GST, summed per supply type into the column of its
856/// treatment
857fn nil_section(priced: &[PricedInvoice]) -> NilSupplies {
858    let rows = priced
859        .iter()
860        .flat_map(|p| {
861            let supply_type = nil_supply_type(p.invoice);
862            p.invoice
863                .line_items
864                .iter()
865                .zip(&p.lines)
866                .filter(|(item, _)| !item.charges_gst())
867                .map(move |(item, b)| (supply_type, item.treatment, &b.taxable_value))
868        })
869        .fold(
870            BTreeMap::new(),
871            |mut rows, (supply_type, treatment, value)| {
872                *rows
873                    .entry(supply_type)
874                    .or_insert_with(|| NilRow::empty(supply_type))
875                    .column_mut(treatment) += value;
876                rows
877            },
878        );
879    NilSupplies {
880        rows: rows.into_values().collect(),
881    }
882}
883
884/// The Table 8 row an invoice's untaxed lines belong to
885fn nil_supply_type(invoice: &GstInvoice) -> NilSupplyType {
886    match (invoice.buyer.is_registered(), invoice.is_inter_state()) {
887        (true, true) => NilSupplyType::InterB2b,
888        (true, false) => NilSupplyType::IntraB2b,
889        (false, true) => NilSupplyType::InterB2c,
890        (false, false) => NilSupplyType::IntraB2c,
891    }
892}

894/// An invoice's breakdowns per rate over the lines that charge GST: what Tables 4A, 5 and 7
895/// report, Table 8 taking the rest (the same split [`nil_section`] makes)
896fn taxable_rates(p: &PricedInvoice) -> impl Iterator<Item = (BigDecimal, GstBreakdown)> {
897    by_rate_where(p, GstLineItem::charges_gst).into_iter()
898}
899
900/// The breakdowns of the lines `keep` accepts, summed per rate so that 18 and 18.00 are one rate
901fn by_rate_where(
902    p: &PricedInvoice,
903    keep: impl Fn(&GstLineItem) -> bool,
904) -> BTreeMap<BigDecimal, GstBreakdown> {
905    p.invoice
906        .line_items
907        .iter()
908        .zip(&p.lines)
909        .filter(|(item, _)| keep(item))
910        .fold(BTreeMap::new(), |mut rates, (item, breakdown)| {
911            *rates.entry(item.gst_rate.normalized()).or_default() += breakdown;
912            rates
913        })
914}

916/// The lines of every invoice with one HSN/SAC code and rate, before numbering
917#[derive(Default)]
918struct HsnTotals {
919    quantity: BigDecimal,
920    breakdown: GstBreakdown,
921    first_description: String,
922}
923
924/// One tab of Table 12: every line of `priced` grouped by HSN/SAC code and rate
925fn hsn_rows(priced: &[&PricedInvoice], master: &HsnMaster) -> Vec<HsnRow> {
926    let totals = priced
927        .iter()
928        .flat_map(|p| p.invoice.line_items.iter().zip(&p.lines))
929        .fold(BTreeMap::new(), |mut rows, (item, breakdown)| {
930            let totals: &mut HsnTotals = rows
931                .entry((item.hsn_sac.clone(), item.gst_rate.normalized()))
932                .or_default();
933            if totals.first_description.is_empty() {
934                totals.first_description.clone_from(&item.description);
935            }
936            totals.quantity += &item.quantity;
937            totals.breakdown += breakdown;
938            rows
939        });
940
941    totals
942        .into_iter()
943        .enumerate()
944        .map(|(i, ((hsn_sac, rate), totals))| hsn_row(i + 1, hsn_sac, rate, totals, master))
945        .collect()
946}
947
948fn hsn_row(
949    number: usize,
950    hsn_sac: String,
951    rate: BigDecimal,
952    totals: HsnTotals,
953    master: &HsnMaster,
954) -> HsnRow {
955    let description = master
956        .lookup(&hsn_sac)
957        .map_or(totals.first_description, |entry| entry.description.clone());
958    let (uqc, quantity) = match HsnSacKind::of_code(&hsn_sac) {
959        HsnSacKind::Hsn => (GOODS_UQC, totals.quantity),
960        HsnSacKind::Sac => (SERVICES_UQC, BigDecimal::zero()),
961    };
962    let breakdown = totals.breakdown;
963    HsnRow {
964        number,
965        hsn_sac,
966        description,
967        uqc: uqc.to_string(),
968        quantity,
969        total_value: breakdown.total,
970        taxable_value: breakdown.taxable_value,
971        igst: breakdown.igst,
972        cgst: breakdown.cgst,
973        sgst: breakdown.sgst,
974        cess: BigDecimal::zero(),
975        rate,
976    }
977}

1017/// A decimal written as a JSON number
1018fn portal_number<S: Serializer>(value: &BigDecimal, serializer: S) -> Result<S::Ok, S::Error> {
1019    let number = value
1020        .to_f64()
1021        .filter(|number| number.is_finite())
1022        .ok_or_else(|| S::Error::custom(format!("{value} does not fit a JSON number")))?;
1023    serializer.serialize_f64(number)
1024}
