1//! GSTR-1 Table 9B: credit notes
2//!
3//! A credit note is reported by the supply kind of the invoice it corrects. Notes against B2B
4//! invoices go to `cdnr`, grouped by buyer GSTIN like Table 4A. Notes against B2CL invoices go to
5//! `cdnur` with the type `B2CL`. Notes against B2CS invoices have no Table 9B entry: the GSTR-1
6//! builder subtracts them from the Table 7 rows instead.
7//!
8//! The amounts here are positive, as the portal expects; the note type `C` says they reduce the
9//! supply. Like an invoice, a note lists only the rates it charges GST at, keeps its whole value,
10//! and is left out when no line charges GST, its lines then being netted in Table 8 alone.
11

21/// Every credit note to one registered buyer
22#[derive(Debug, Clone, PartialEq, Serialize)]
23pub struct CdnrParty {
24    /// Buyer's GSTIN
25    #[serde(rename = "ctin")]
26    pub buyer_gstin: Gstin,
27    /// The buyer's notes, by date then number
28    #[serde(rename = "nt")]
29    pub notes: Vec<CdnrNote>,
30}

32/// One credit note to a registered buyer
33#[derive(Debug, Clone, PartialEq, Serialize)]
34pub struct CdnrNote {
35    /// Kind of note
36    #[serde(rename = "ntty")]
37    pub note_type: NoteType,
38    /// Note number
39    #[serde(rename = "nt_num")]
40    pub note_number: String,
41    /// Date of issue, written `dd-mm-yyyy`
42    #[serde(rename = "nt_dt", serialize_with = "portal_date")]
43    pub note_date: NaiveDate,
44    /// Place of supply: the buyer's state code
45    #[serde(rename = "pos")]
46    pub place_of_supply: String,
47    /// Whether tax is payable on reverse charge, written `Y` or `N`
48    #[serde(rename = "rchrg", serialize_with = "yes_no")]
49    pub reverse_charge: bool,
50    /// Kind of the invoice the note corrects
51    #[serde(rename = "inv_typ")]
52    pub invoice_type: B2bInvoiceType,
53    /// Note value: taxable value plus tax
54    #[serde(rename = "val", serialize_with = "portal_amount")]
55    pub note_value: BigDecimal,
56    /// One item per GST rate on the note, by rate
57    #[serde(rename = "itms")]
58    pub items: Vec<B2bItem>,
59}

61/// One credit note to an unregistered buyer, against a B2CL invoice
62#[derive(Debug, Clone, PartialEq, Serialize)]

69    pub note_type: NoteType,

71    #[serde(rename = "nt_num")]
72    pub note_number: String,

75    pub note_date: NaiveDate,
76    /// Place of supply: a state code
77    #[serde(rename = "pos")]
78    pub place_of_supply: String,

80    #[serde(rename = "val", serialize_with = "portal_amount")]

87/// Kind of a Table 9B note
88#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize)]
89pub enum NoteType {
90    /// A credit note, which reduces the supply
91    #[serde(rename = "C")]
92    Credit,
93}

95/// Kind of invoice a `cdnur` note corrects
96#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize)]
97pub enum CdnurType {
98    /// A large inter-state supply to an unregistered buyer (Table 5)
99    #[serde(rename = "B2CL")]
100    B2cl,
101}

103/// Table 9B `cdnr`: notes grouped by buyer GSTIN, keeping their order within each buyer
104pub(super) fn cdnr_section(notes: &[&PricedDocument]) -> Vec<CdnrParty> {
105    grouped(
106        notes
107            .iter()
108            .filter_map(|p| Some((p.document.buyer().gstin()?, cdnr_note(p)?))),
109    )
110    .into_iter()
111    .map(|(buyer_gstin, notes)| CdnrParty {
112        buyer_gstin: buyer_gstin.clone(),
113        notes,
114    })
115    .collect()
116}

118/// One `cdnr` note with an item per taxed rate, or `None` when no line charges GST
119fn cdnr_note(p: &PricedDocument) -> Option<CdnrNote> {
120    let items: Vec<_> = issued_rates(p)
121        .enumerate()
122        .map(|(i, (rate, breakdown))| B2bItem {
123            number: i + 1,
124            detail: ItemDetail::new(rate, &breakdown),
125        })
126        .collect();
127    (!items.is_empty()).then(|| CdnrNote {
128        note_type: NoteType::Credit,
129        note_number: p.document.number().to_string(),
130        note_date: p.document.date(),
131        place_of_supply: p.document.buyer().place_of_supply().to_string(),
132        reverse_charge: false,
133        invoice_type: B2bInvoiceType::Regular,
134        note_value: -&p.total.total,
135        items,
136    })
137}

139/// Table 9B `cdnur`: the notes against B2CL invoices, by date then number
140pub(super) fn cdnur_section(notes: &[&PricedDocument]) -> Vec<CdnurNote> {
141    notes.iter().filter_map(|p| cdnur_note(p)).collect()
142}
143
144/// One `cdnur` note with an item per taxed rate, or `None` when no line charges GST
145fn cdnur_note(p: &PricedDocument) -> Option<CdnurNote> {
146    let items: Vec<_> = issued_rates(p)
147        .enumerate()
148        .map(|(i, (rate, breakdown))| B2clItem {
149            number: i + 1,
150            detail: B2clItemDetail::new(rate, &breakdown),
151        })
152        .collect();
153    (!items.is_empty()).then(|| CdnurNote {
154        original_type: CdnurType::B2cl,
155        note_type: NoteType::Credit,
156        note_number: p.document.number().to_string(),
157        note_date: p.document.date(),
158        place_of_supply: p.document.buyer().place_of_supply().to_string(),
159        note_value: -&p.total.total,
160        items,
161    })
162}

164/// A note's breakdowns per taxed rate as issued: positive, where the priced note nets them
165/// negative
166fn issued_rates(p: &PricedDocument) -> impl Iterator<Item = (BigDecimal, GstBreakdown)> {
167    taxable_rates(p).map(|(rate, breakdown)| (rate, breakdown.negated()))
168}
