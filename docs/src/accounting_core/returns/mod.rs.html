1//! GST returns built from invoices
2//!
3//! [`Gstr1Return::build`] aggregates a filer's [`GstInvoice`](crate::invoice::GstInvoice)s for one
4//! [`ReturnPeriod`] into GSTR-1, the return of outward supplies: Table 4A (B2B supplies by buyer
5//! and rate), Table 5 (B2CL: large inter-state supplies to unregistered buyers, by place of
6//! supply), Table 7 (B2CS: other supplies to unregistered buyers, summed by place of supply and
7//! rate), Table 8 (the nil-rated lines of B2C invoices), Table 12 (the HSN/SAC summary) and Table 13
8//! (documents issued).
9//! [`Gstr1Return::to_json`] writes it in the GST portal's offline-tool schema for upload.
10//!
11//! Building is pure: the caller passes the invoices and the HSN/SAC master, and gets a value
12//! back. Supplies through an e-commerce operator, exports, credit and debit notes and amendments
13//! are not covered yet. Failures are typed as [`Gstr1Error`].
14//!
15//! # Example
16//!
17//! ```
18//! use accounting_core::invoice::{GstInvoice, GstLineItem, Gstin, HsnMaster};
19//! use accounting_core::returns::{Gstr1Return, ReturnPeriod};
20//! use bigdecimal::BigDecimal;
21//! use chrono::NaiveDate;
22//!
23//! # fn main() -> Result<(), accounting_core::Error> {
24//! let seller = Gstin::parse("27AAPFU0939F1ZV")?;
25//! let buyer = Gstin::parse("27AAPFU0939F2ZU")?;
26//! let date = NaiveDate::from_ymd_opt(2024, 11, 15).unwrap();
27//! let item = |rate| GstLineItem::new("998314", "IT consulting", BigDecimal::from(1),
28//!     BigDecimal::from(1000), BigDecimal::from(rate));
29//! let invoices = [
30//!     GstInvoice::new("INV-001", date, seller.clone(), buyer.clone(), vec![item(18)?])?,
31//!     GstInvoice::new("INV-002", date, seller.clone(), buyer, vec![item(18)?, item(5)?])?,
32//! ];
33//!
34//! let gstr1 = Gstr1Return::build(&seller, ReturnPeriod::new(2024, 11)?, &invoices,
35//!     HsnMaster::global())?;
36//!
37//! // One buyer with two invoices; the second has one item per rate
38//! assert_eq!(gstr1.b2b.len(), 1);
39//! assert_eq!(gstr1.b2b[0].invoices[1].items.len(), 2);
40//! // The HSN summary has one row per code and rate
41//! assert_eq!(gstr1.hsn.b2b.len(), 2);
42//! assert_eq!(gstr1.doc_issue.documents[0].series[0].total, 2);
43//! # Ok(())
44//! # }
45//! ```
